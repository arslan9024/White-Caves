# **Login/Signup Redesign with Your Requirements**

## **1. Redux State Structure**
```javascript
// authSlice.js
import { createSlice, createAsyncThunk } from '@reduxjs/toolkit';

const initialState = {
  authMethod: 'social', // 'social', 'email', 'phone', 'uae_pass'
  email: '',
  phone: '',
  otp: '',
  isOtpSent: false,
  loading: false,
  error: null,
};

export const verifyOtp = createAsyncThunk(
  'auth/verifyOtp',
  async ({ type, value, otp }, { rejectWithValue }) => {
    try {
      // This would be your actual API call
      const response = await fetch('/api/verify-otp', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ type, value, otp }),
      });
      
      const data = await response.json();
      
      // Using your JSON response structure
      if (data.verified) {
        return { type, value, verified: true };
      } else {
        return rejectWithValue(data.error || 'OTP verification failed');
      }
    } catch (error) {
      return rejectWithValue(error.message);
    }
  }
);

export const sendOtp = createAsyncThunk(
  'auth/sendOtp',
  async ({ type, value }) => {
    // Your API call to send OTP
    const response = await fetch('/api/send-otp', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ type, value }),
    });
    return await response.json();
  }
);

const authSlice = createSlice({
  name: 'auth',
  initialState,
  reducers: {
    setAuthMethod: (state, action) => {
      state.authMethod = action.payload;
      state.isOtpSent = false;
      state.otp = '';
      state.error = null;
    },
    setEmail: (state, action) => {
      state.email = action.payload;
    },
    setPhone: (state, action) => {
      state.phone = action.payload;
    },
    setOtp: (state, action) => {
      state.otp = action.payload;
    },
    resetAuthState: (state) => {
      state.authMethod = 'social';
      state.email = '';
      state.phone = '';
      state.otp = '';
      state.isOtpSent = false;
      state.error = null;
    },
  },
  extraReducers: (builder) => {
    builder
      .addCase(verifyOtp.pending, (state) => {
        state.loading = true;
        state.error = null;
      })
      .addCase(verifyOtp.fulfilled, (state, action) => {
        state.loading = false;
        state.isOtpSent = false;
        state.otp = '';
        state.error = null;
      })
      .addCase(verifyOtp.rejected, (state, action) => {
        state.loading = false;
        state.error = action.payload;
      })
      .addCase(sendOtp.fulfilled, (state, action) => {
        state.isOtpSent = true;
        state.loading = false;
      });
  },
});

export const { setAuthMethod, setEmail, setPhone, setOtp, resetAuthState } = authSlice.actions;
export default authSlice.reducer;
```

## **2. Main Auth Component**
```jsx
// AuthContainer.jsx
import React, { useState } from 'react';
import { useSelector, useDispatch } from 'react-redux';
import { setAuthMethod, resetAuthState } from './authSlice';
import SocialAuth from './SocialAuth';
import EmailForm from './EmailForm';
import PhoneForm from './PhoneForm';
import UaePassAuth from './UaePassAuth';

const AuthContainer = () => {
  const dispatch = useDispatch();
  const { authMethod } = useSelector((state) => state.auth);
  const [isLogin, setIsLogin] = useState(true);

  const handleMethodChange = (method) => {
    dispatch(setAuthMethod(method));
  };

  const handleBack = () => {
    dispatch(setAuthMethod('social'));
  };

  const toggleAuthMode = () => {
    setIsLogin(!isLogin);
    dispatch(resetAuthState());
  };

  return (
    <div className="auth-container">
      <div className="auth-header">
        <h2>{isLogin ? 'Welcome Back' : 'Create Account'}</h2>
        <p>
          {isLogin 
            ? 'Sign in to access your account' 
            : 'Join our community today'
          }
        </p>
      </div>

      {/* UAE Pass - Always Visible */}
      <UaePassAuth isLogin={isLogin} />

      {/* Toggle for Email/Phone */}
      {(authMethod === 'social' || authMethod === 'email' || authMethod === 'phone') && (
        <div className="method-toggle">
          <button
            className={`toggle-btn ${authMethod === 'email' ? 'active' : ''}`}
            onClick={() => handleMethodChange('email')}
          >
            <i className="icon-email"></i> Email
          </button>
          <button
            className={`toggle-btn ${authMethod === 'phone' ? 'active' : ''}`}
            onClick={() => handleMethodChange('phone')}
          >
            <i className="icon-phone"></i> Phone
          </button>
        </div>
      )}

      {/* Conditional Rendering */}
      {authMethod === 'social' ? (
        <>
          <div className="divider">
            <span>or continue with</span>
          </div>
          <SocialAuth />
          
          {/* Signup Toggle */}
          <div className="auth-switch">
            {isLogin ? "Don't have an account?" : "Already have an account?"}
            <button onClick={toggleAuthMode} className="switch-btn">
              {isLogin ? 'Sign up' : 'Sign in'}
            </button>
          </div>
        </>
      ) : authMethod === 'email' ? (
        <EmailForm isLogin={isLogin} onBack={handleBack} />
      ) : authMethod === 'phone' ? (
        <PhoneForm isLogin={isLogin} onBack={handleBack} />
      ) : null}
    </div>
  );
};

export default AuthContainer;
```

## **4. Email Form Component**
```jsx
// EmailForm.jsx
import React, { useState } from 'react';
import { useSelector, useDispatch } from 'react-redux';
import { setEmail, sendOtp, verifyOtp } from './authSlice';

const EmailForm = ({ isLogin, onBack }) => {
  const dispatch = useDispatch();
  const { email, isOtpSent, loading, error } = useSelector((state) => state.auth);
  const [otp, setLocalOtp] = useState('');

  const handleSubmit = async (e) => {
    e.preventDefault();
    
    if (isOtpSent) {
      dispatch(verifyOtp({ type: 'email', value: email, otp }));
    } else {
      dispatch(sendOtp({ type: 'email', value: email }));
    }
  };

  const handleResendOtp = () => {
    dispatch(sendOtp({ type: 'email', value: email }));
  };

  return (
    <div className="auth-form">
      <button className="back-btn" onClick={onBack}>
        ← Back to all options
      </button>

      <form onSubmit={handleSubmit}>
        <h3>{isLogin ? 'Sign in with Email' : 'Sign up with Email'}</h3>
        
        {!isOtpSent ? (
          <>
            <div className="form-group">
              <label htmlFor="email">Email Address</label>
              <input
                type="email"
                id="email"
                value={email}
                onChange={(e) => dispatch(setEmail(e.target.value))}
                placeholder="Enter your email"
                required
              />
            </div>
            
            {isLogin && (
              <div className="form-group">
                <label htmlFor="password">Password</label>
                <input
                  type="password"
                  id="password"
                  placeholder="Enter your password"
                  required
                />
              </div>
            )}
          </>
        ) : (
          <>
            <div className="otp-section">
              <p>Enter the OTP sent to <strong>{email}</strong></p>
              <div className="otp-inputs">
                {[...Array(6)].map((_, index) => (
                  <input
                    key={index}
                    type="text"
                    maxLength="1"
                    value={otp[index] || ''}
                    onChange={(e) => {
                      const newOtp = otp.split('');
                      newOtp[index] = e.target.value;
                      setLocalOtp(newOtp.join(''));
                      dispatch(setOtp(newOtp.join('')));
                      
                      // Auto-focus next input
                      if (e.target.value && index < 5) {
                        document.getElementById(`otp-${index + 1}`)?.focus();
                      }
                    }}
                    id={`otp-${index}`}
                    className="otp-input"
                  />
                ))}
              </div>
              
              <button
                type="button"
                className="resend-btn"
                onClick={handleResendOtp}
              >
                Resend OTP
              </button>
            </div>
          </>
        )}

        {error && <div className="error-message">{error}</div>}

        <button type="submit" className="submit-btn" disabled={loading}>
          {loading ? 'Processing...' : 
           isOtpSent ? 'Verify OTP' : 
           isLogin ? 'Sign In' : 'Continue'}
        </button>
      </form>
    </div>
  );
};

export default EmailForm;
```

## **9. Store Configuration**
```javascript
// store.js
import { configureStore } from '@reduxjs/toolkit';
import authReducer from './authSlice';

export const store = configureStore({
  reducer: {
    auth: authReducer,
  },
  middleware: (getDefaultMiddleware) =>
    getDefaultMiddleware({
      serializableCheck: false,
    }),
});
```

This implementation:
1. **Doesn't change your color scheme** - Uses your existing brand colors
2. **Shows social components on priority** - Grid layout prominently displayed
3. **Has email/phone toggle** - Toggle replaces social with forms using Redux state
4. **Includes JSON API responses** - Mock API structure for OTP verification

You can install dependencies:
```bash
npm install react-redux @reduxjs/toolkit react-phone-input-2
```

The flow:
1. User sees UAE Pass + Social buttons
2. Toggles email/phone → Social buttons disappear, form appears
3. Back button returns to social view
4. All state managed through Redux
5. JSON API structure ready for backend integration
