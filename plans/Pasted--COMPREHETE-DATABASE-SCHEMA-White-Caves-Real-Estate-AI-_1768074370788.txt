## **2. POSTGRESQL SCHEMA (Financial & Transactional Data)**

### **2.1 Revenue Tables**

```sql
-- revenue_transactions table
CREATE TABLE revenue_transactions (
    transaction_id VARCHAR(50) PRIMARY KEY,
    external_transaction_id VARCHAR(50) NOT NULL,
```

## **8. DATA RETENTION POLICIES**

### **8.1 Retention Periods**

```sql
-- PostgreSQL retention policies
-- Audit logs: 7 years
CREATE TABLE audit_logs_archive (LIKE audit_logs INCLUDING ALL);
-- Move data older than 7 years to archive
INSERT INTO audit_logs_archive 
SELECT * FROM audit_logs 
WHERE timestamp < NOW() - INTERVAL '7 years';

-- Financial transactions: 10 years
CREATE TABLE revenue_transactions_archive (LIKE revenue_transactions INCLUDING ALL);
-- Move completed transactions older than 10 years to archive
INSERT INTO revenue_transactions_archive 
SELECT * FROM revenue_transactions 
WHERE status = 'completed' 
  AND payment_date < NOW() - INTERVAL '10 years';
```

```javascript
// Migration script for version updates
async function migrateDatabase(version) {
  const migrations = {
    "1.0": async () => {
      // Initial migration
      await db.users.updateMany(
        {},
        { $set: { "tier.autoUpgradeEnabled": true } }
      );
    },
    "1.1": async () => {
      // Add UAE PASS verification field
      await db.users.updateMany(
        {},
        { $set: { "verification.uaePassVerified": false } }
      );
    },
    "1.2": async () => {
      // Add AI assistant performance metrics
      await db.ai_assistants.updateMany(
        {},
        { $set: { "performanceMetrics": defaultPerformanceMetrics } }
      );
    }
  };
  
  // Execute migrations in order
  for (const [migrationVersion, migrationFunction] of Object.entries(migrations)) {
    if (compareVersions(migrationVersion, version) > 0) {
      console.log(`Running migration ${migrationVersion}`);
      await migrationFunction();
      await db.meta.updateOne(
        { key: "database_version" },
        { $set: { value: migrationVersion } },
        { upsert: true }
      );
    }
  }
}
```

```javascript
// Optimized query patterns
const optimizedQueries = {
  // Get user with permissions and tier
  getUserWithAccess: async (userId) => {
    return db.users.aggregate([
      { $match: { userId } },
      { $lookup: {
          from: "permissions",

          foreignField: "userId",
          as: "permissions"
        }
      },
      { $lookup: {
          from: "tiers",
          localField: "tier.currentTier",
          foreignField: "tierId",
          as: "tierDetails"
        }
      },
      { $project: {
          userId: 1,
          email: 1,
          userType: 1,
          tier: 1,
          permissions: 1,
          accessibleServices: 1,
          tierDetails: { $arrayElemAt: ["$tierDetails", 0] }
        }
      }
    ]).toArray();
  },
  
  // Get properties with filters and pagination
  searchProperties: async (filters, page, limit) => {
    const pipeline = [
      { $match: buildPropertyFilters(filters) },
      { $sort: { "metadata.score": -1, "audit.createdAt": -1 } },
      { $skip: (page - 1) * limit },
      { $limit: limit },
      { $lookup: {
          from: "properties_media",
          localField: "propertyId",
          foreignField: "propertyId",
          as: "media"
        }
      },
      { $project: {
          propertyId: 1,
          title: 1,
          description: 1,
          location: 1,
          basicInfo: 1,
          status: 1,
          financial: 1,
          media: { $slice: ["$media", 5] } // Limit to 5 images
        }
      }
    ];
    
    return db.properties.aggregate(pipeline).toArray();
  }
};
```
