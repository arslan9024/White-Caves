## 🔐 1. Google Sheets Service (Contact Manager)

This service securely fetches and validates contact numbers from your Google Sheets.

### **`google-sheets-service.js`**
```javascript
const { google } = require('googleapis');
const { authenticate } = require('@google-cloud/local-auth');
const path = require('path');
require('dotenv').config();

class GoogleSheetsService {
    constructor() {
        this.SCOPES = ['https://www.googleapis.com/auth/spreadsheets.readonly'];
        this.CREDENTIALS_PATH = path.join(process.cwd(), 'credentials.json');
        this.sheets = null;
        this.initialize();
    }

    async initialize() {
        try {
            const auth = await authenticate({
                scopes: this.SCOPES,
                keyfilePath: this.CREDENTIALS_PATH,
            });
            this.sheets = google.sheets({ version: 'v4', auth });
            console.log('Google Sheets service initialized');
        } catch (error) {
            console.error('Failed to initialize Google Sheets:', error);
            throw error;
        }
    }

    /**
     * Reads and validates contact numbers from a Google Sheet
     * @param {string} spreadsheetId - ID of the Google Sheet
     * @param {string} range - A1 notation range (e.g., 'Contacts!A2:B')
     * @returns {Promise<Array>} - Array of validated contact objects
     */
    async getValidatedContacts(spreadsheetId, range) {
        if (!this.sheets) {
            await this.initialize();
        }

        try {
            // Fetch data from Google Sheets[reference:0]
            const result = await this.sheets.spreadsheets.values.get({
                spreadsheetId,
                range,
            });

            const rows = result.data.values;
            if (!rows || rows.length === 0) {
                console.log('No data found in sheet');
                return [];
            }

            // Process and validate each row
            const validatedContacts = rows.map((row, index) => {
                const [name, rawNumber, tags, lastContacted] = row;
                const validation = this.validateAndFormatNumber(rawNumber);
                
                return {
                    id: `contact-${index + 1}`,
                    name: name || `Contact ${index + 1}`,
                    rawNumber,
                    validatedNumber: validation.correctedNumber,
                    isValid: validation.isValid,
                    countryCode: validation.countryCode,
                    tags: tags ? tags.split(',').map(t => t.trim()) : [],
                    lastContacted: lastContacted || null,
                    sheetRow: index + 2, // For reference back to sheet
                    validationErrors: validation.errors || []
                };
            });

            // Separate valid and invalid contacts
            const validContacts = validatedContacts.filter(c => c.isValid);
            const invalidContacts = validatedContacts.filter(c => !c.isValid);

            console.log(`Loaded ${validContacts.length} valid, ${invalidContacts.length} invalid contacts`);
            
            return {
                validContacts,
                invalidContacts,
                total: validatedContacts.length,
                timestamp: new Date().toISOString()
            };

        } catch (error) {
            console.error('Error reading Google Sheet:', error);
            throw new Error(`Failed to read contacts: ${error.message}`);
        }
    }

    /**
     * Validates and formats phone numbers for WhatsApp
     */
    validateAndFormatNumber(rawNumber) {
        // Remove all non-digit characters
        let cleaned = rawNumber.toString().replace(/\D/g, '');
        const errors = [];
        
        // Check minimum length
        if (cleaned.length < 10) {
            errors.push('Number too short');
            return { isValid: false, correctedNumber: null, errors };
        }
        
        // Country code detection and formatting
        let correctedNumber;
        let countryCode = '';
        
        // Handle common country codes
        if (cleaned.startsWith('1') && cleaned.length === 11) {
            countryCode = '1';
            correctedNumber = cleaned + '@c.us';
        } else if (cleaned.startsWith('91') && cleaned.length === 12) {
            countryCode = '91';
            correctedNumber = cleaned + '@c.us';
        } else if (cleaned.startsWith('44') && cleaned.length === 12) {
            countryCode = '44';
            correctedNumber = cleaned + '@c.us';
        } else if (cleaned.length === 10) {
            // Assume Indian number if 10 digits
            countryCode = '91';
            correctedNumber = '91' + cleaned + '@c.us';
        } else if (cleaned.length >= 11 && cleaned.length <= 15) {
            // Generic international format
            correctedNumber = cleaned + '@c.us';
        } else {
            errors.push('Unrecognized number format');
            return { isValid: false, correctedNumber: null, errors };
        }
        
        return {
            isValid: true,
            correctedNumber,
            countryCode,
            original: rawNumber,
            errors
        };
    }

    /**
     * Updates contact status in the sheet (e.g., last contacted)
     */
    async updateContactStatus(spreadsheetId, range, rowIndex, status) {
        try {
            await this.sheets.spreadsheets.values.update({
                spreadsheetId,
                range: `${range.split('!')[0]}!D${rowIndex}`, // Update column D
                valueInputOption: 'RAW',
                resource: {
                    values: [[status]]
                }
            });
            console.log(`Updated contact status for row ${rowIndex}`);
        } catch (error) {
            console.error('Failed to update sheet:', error);
        }
    }
}

module.exports = GoogleSheetsService;
```

## 🛡️ 2. Enhanced Campaign Manager with Strict Limits

This manager enforces multi-level limits and behavioral patterns.

### **`enhanced-campaign-manager.js`**
```javascript
const CircuitBreaker = require('opossum');
const GoogleSheetsService = require('./google-sheets-service');

class EnhancedCampaignManager {
    constructor(whatsappClient) {
        this.client = whatsappClient;
        this.sheetsService = new GoogleSheetsService();
        
        // Strict multi-level limits
        this.limits = {
            daily: parseInt(process.env.MAX_DAILY_CAMPAIGN_MESSAGES) || 80,
            hourly: 12,
            perContactDaily: 2,
            perCampaign: 50,
            minDelayBetweenMessages: 3000, // 3 seconds minimum
            maxDelayBetweenMessages: 10000, // 10 seconds maximum
            campaignCooldownHours: 6
        };
        
        // Trackers
        this.counters = {
            daily: 0,
            hourly: {},
            contactDaily: {},
            campaignHistory: []
        };
        
        // Circuit breaker for sending messages[reference:1]
        this.breaker = new CircuitBreaker(async (chatId, message) => {
            return await this.client.sendMessage(chatId, message);
        }, {
            timeout: 10000,
            errorThresholdPercentage: 40, // Trip if 40% of sends fail
            resetTimeout: 30000,
            rollingCountTimeout: 60000,
            rollingCountBuckets: 10
        });
        
        this.setupBreakerEvents();
        this.resetCounters();
    }
    
    setupBreakerEvents() {
        this.breaker.on('open', () => {
            console.warn('🚨 CIRCUIT BREAKER OPEN: Message sending suspended');
            this.logAnomaly('circuit_breaker_opened', {
                timestamp: new Date().toISOString(),
                failureRate: this.breaker.stats.failures / this.breaker.stats.requests * 100
            });
        });
        
        this.breaker.on('halfOpen', () => {
            console.log('🔶 Circuit breaker half-open: Testing recovery');
        });
        
        this.breaker.on('close', () => {
            console.log('✅ Circuit breaker closed: Service recovered');
        });
        
        this.breaker.fallback(async (chatId, message) => {
            return {
                success: false,
                error: 'Service temporarily unavailable due to high failure rate',
                fallback: true,
                timestamp: new Date().toISOString()
            };
        });
    }
    
    /**
     * Main campaign execution with enhanced protections
     */
    async executeCampaign(campaignConfig) {
        const {
            spreadsheetId,
            range,
            messageTemplate,
            campaignId = `campaign-${Date.now()}`
        } = campaignConfig;
        
        // 1. Validate campaign schedule
        const scheduleCheck = this.checkCampaignSchedule();
        if (!scheduleCheck.allowed) {
            return {
                success: false,
                error: `Schedule restriction: ${scheduleCheck.reason}`,
                waitMinutes: scheduleCheck.waitMinutes
            };
        }
        
        // 2. Load contacts from Google Sheets
        let contacts;
        try {
            contacts = await this.sheetsService.getValidatedContacts(spreadsheetId, range);
        } catch (error) {
            return {
                success: false,
                error: `Failed to load contacts: ${error.message}`
            };
        }
        
        // 3. Apply contact filters
        const filteredContacts = this.filterContacts(contacts.validContacts, campaignConfig);
        
        // 4. Check campaign limits
        const limitCheck = this.checkCampaignLimits(filteredContacts.length, campaignId);
        if (!limitCheck.allowed) {
            return {
                success: false,
                error: `Campaign limit exceeded: ${limitCheck.reason}`,
                details: limitCheck.details
            };
        }
        
        // 5. Execute with strict pacing
        const results = await this.executeWithPacing(filteredContacts, messageTemplate, campaignId);
        
        // 6. Record campaign completion
        this.recordCampaignCompletion(campaignId, results);
        
        return {
            success: true,
            campaignId,
            summary: {
                totalContacts: filteredContacts.length,
                sent: results.sent,
                failed: results.failed,
                dailyRemaining: this.limits.daily - this.counters.daily,
                estimatedCompletion: new Date(Date.now() + (filteredContacts.length * this.limits.minDelayBetweenMessages))
            },
            details: results.details
        };
    }
    
    /**
     * Check if campaign can run based on schedule
     */
    checkCampaignSchedule() {
        const now = new Date();
        const hour = now.getHours();
        const day = now.getDay();
        
        // Night off (10 PM to 8 AM)
        if (hour >= 22 || hour < 8) {
            const waitHours = hour >= 22 ? (32 - hour) : (8 - hour);
            return {
                allowed: false,
                reason: 'Night off period',
                waitMinutes: waitHours * 60
            };
        }
        
        // Lunch break (12 PM to 1:30 PM)
        if (hour === 12 || (hour === 13 && now.getMinutes() < 30)) {
            const waitMinutes = hour === 12 ? 90 : (30 - now.getMinutes());
            return {
                allowed: false,
                reason: 'Lunch break',
                waitMinutes
            };
        }
        
        // Dinner break (6 PM to 7:30 PM)
        if (hour === 18 || (hour === 19 && now.getMinutes() < 30)) {
            const waitMinutes = hour === 18 ? 90 : (30 - now.getMinutes());
            return {
                allowed: false,
                reason: 'Dinner break',
                waitMinutes
            };
        }
        
        // Weekend restriction
        if (day === 0 || day === 6) { // Sunday or Saturday
            const waitDays = day === 6 ? 2 : 1; // Wait until Monday
            return {
                allowed: false,
                reason: 'Weekend restriction',
                waitMinutes: waitDays * 24 * 60
            };
        }
        
        return { allowed: true };
    }
    
    /**
     * Filter contacts based on campaign rules
     */
    filterContacts(contacts, campaignConfig) {
        const { excludeRecentlyContacted = true, minDaysBetweenMessages = 7 } = campaignConfig;
        
        return contacts.filter(contact => {
            // Check daily contact limit
            const contactKey = contact.validatedNumber;
            const today = new Date().toDateString();
            const contactDailyCount = this.counters.contactDaily[contactKey] || 0;
            
            if (contactDailyCount >= this.limits.perContactDaily) {
                return false;
            }
            
            // Check if recently contacted
            if (excludeRecentlyContacted && contact.lastContacted) {
                const lastContact = new Date(contact.lastContacted);
                const daysSince = (new Date() - lastContact) / (1000 * 60 * 60 * 24);
                if (daysSince < minDaysBetweenMessages) {
                    return false;
                }
            }
            
            return true;
        });
    }
    
    /**
     * Execute campaign with human-like pacing
     */
    async executeWithPacing(contacts, messageTemplate, campaignId) {
        const results = {
            sent: 0,
            failed: 0,
            details: []
        };
        
        for (const contact of contacts) {
            // Check real-time limits before each message
            const realTimeCheck = this.checkRealTimeLimits();
            if (!realTimeCheck.allowed) {
                results.details.push({
                    contact: contact.validatedNumber,
                    success: false,
                    error: realTimeCheck.reason,
                    skipped: true
                });
                results.failed++;
                continue;
            }
            
            // Personalize message
            const personalizedMessage = this.personalizeMessage(messageTemplate, contact);
            
            // Calculate dynamic delay
            const delay = this.calculateDynamicDelay(results.sent);
            await this.delay(delay);
            
            // Send message through circuit breaker
            try {
                const sendResult = await this.breaker.fire(contact.validatedNumber, personalizedMessage);
                
                if (sendResult.success !== false) {
                    results.sent++;
                    this.incrementCounters(contact.validatedNumber);
                    
                    // Update contact status in sheet
                    await this.sheetsService.updateContactStatus(
                        process.env.SPREADSHEET_ID,
                        process.env.RANGE,
                        contact.sheetRow,
                        new Date().toISOString()
                    );
                } else {
                    results.failed++;
                }
                
                results.details.push({
                    contact: contact.validatedNumber,
                    success: sendResult.success !== false,
                    messageId: sendResult.id?.id,
                    timestamp: new Date().toISOString(),
                    error: sendResult.error
                });
                
            } catch (error) {
                results.failed++;
                results.details.push({
                    contact: contact.validatedNumber,
                    success: false,
                    error: error.message,
                    timestamp: new Date().toISOString()
                });
                
                // Log anomaly for ML detection
                this.logAnomaly('send_message_failed', {
                    contact: contact.validatedNumber,
                    error: error.message,
                    campaignId
                });
            }
            
            // Check if circuit breaker is open
            if (this.breaker.opened) {
                console.warn('Campaign paused: Circuit breaker is open');
                break;
            }
        }
        
        return results;
    }
    
    /**
     * Calculate dynamic delay that mimics human behavior
     */
    calculateDynamicDelay(messageCount) {
        // Base delay increases as we send more messages
        const baseDelay = this.limits.minDelayBetweenMessages + 
                         (Math.log(messageCount + 1) * 1000);
        
        // Add random variation (30-70% of base delay)
        const randomFactor = 0.3 + Math.random() * 0.4;
        const randomDelay = baseDelay * randomFactor;
        
        // Occasional longer pauses (simulating human breaks)
        const pauseChance = 0.05;
        const pause = Math.random() < pauseChance ? 
                      (30000 + Math.random() * 60000) : 0; // 30-90 second pause
        
        return Math.min(
            this.limits.maxDelayBetweenMessages,
            baseDelay + randomDelay + pause
        );
    }
    
    /**
     * Personalize message with contact data
     */
    personalizeMessage(template, contact) {
        return template
            .replace(/{name}/g, contact.name)
            .replace(/{number}/g, contact.rawNumber)
            .replace(/{tags}/g, contact.tags.join(', '))
            .replace(/{time}/g, new Date().toLocaleTimeString());
    }
    
    /**
     * Check real-time limits
     */
    checkRealTimeLimits() {
        const now = new Date();
        const hour = now.getHours();
        const today = now.toDateString();
        
        // Daily limit
        if (this.counters.daily >= this.limits.daily) {
            return {
                allowed: false,
                reason: `Daily limit reached (${this.limits.daily})`,
                resetTime: 'Tomorrow 00:00'
            };
        }
        
        // Hourly limit
        const hourCount = this.counters.hourly[hour] || 0;
        if (hourCount >= this.limits.hourly) {
            const nextHour = hour === 23 ? 0 : hour + 1;
            return {
                allowed: false,
                reason: `Hourly limit reached (${this.limits.hourly})`,
                waitMinutes: 60 - now.getMinutes()
            };
        }
        
        return { allowed: true };
    }
    
    /**
     * Increment counters after successful send
     */
    incrementCounters(contactNumber) {
        const now = new Date();
        const hour = now.getHours();
        const today = now.toDateString();
        
        // Daily counter
        this.counters.daily++;
        
        // Hourly counter
        if (!this.counters.hourly[hour]) {
            this.counters.hourly[hour] = 0;
        }
        this.counters.hourly[hour]++;
        
        // Contact daily counter
        if (!this.counters.contactDaily[contactNumber]) {
            this.counters.contactDaily[contactNumber] = 0;
        }
        this.counters.contactDaily[contactNumber]++;
    }
    
    /**
     * Log anomalies for ML detection analysis
     */
    logAnomaly(type, data) {
        const anomalyLog = {
            type,
            timestamp: new Date().toISOString(),
            data,
            counters: {
                daily: this.counters.daily,
                hourly: this.counters.hourly[new Date().getHours()] || 0
            },
            breakerStatus: this.breaker.stats
        };
        
        // In production, send to monitoring service
        console.warn('ANOMALY DETECTED:', JSON.stringify(anomalyLog, null, 2));
        
        // Auto-adjust limits if anomalies are frequent
        if (type === 'circuit_breaker_opened') {
            this.adjustLimits('reduce');
        }
    }
    
    /**
     * Dynamically adjust limits based on anomalies
     */
    adjustLimits(direction) {
        if (direction === 'reduce') {
            this.limits.daily = Math.max(20, Math.floor(this.limits.daily * 0.7));
            this.limits.hourly = Math.max(3, Math.floor(this.limits.hourly * 0.7));
            console.warn(`Limits reduced due to anomalies: ${this.limits.daily} daily, ${this.limits.hourly} hourly`);
        }
    }
    
    /**
     * Reset counters daily
     */
    resetCounters() {
        // Reset at midnight
        const now = new Date();
        const night = new Date(now);
        night.setHours(24, 0, 0, 0);
        const msUntilMidnight = night - now;
        
        setTimeout(() => {
            this.counters.daily = 0;
            this.counters.hourly = {};
            this.counters.contactDaily = {};
            console.log('Daily counters reset');
            
            // Schedule next reset
            this.resetCounters();
        }, msUntilMidnight);
    }
    
    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

module.exports = EnhancedCampaignManager;
```

## ⚠️ 5. Critical Protection Layers Summary

| Layer | Protection | Implementation |
|-------|------------|----------------|
| **Schedule Enforcement** | Avoids sending during off-hours | Time-based blocking with configurable breaks |
| **Multi-Level Rate Limiting** | Prevents pattern detection | Daily, hourly, per-contact limits with dynamic adjustment |
| **Circuit Breaker** | Stops sending when failures increase | Opossum circuit breaker with automatic recovery[reference:2] |
| **Behavioral Pacing** | Mimics human typing patterns | Dynamic delays with random variation and occasional pauses |
| **Anomaly Detection** | Logs suspicious patterns | Automatic limit reduction when anomalies detected |
| **Contact Validation** | Ensures valid numbers | Google Sheets integration with strict validation |
| **ML Evasion** | Reduces detection risk | Irregular timing, personalized messages, limited volumes |

## ⚠️ **Final Warning**

Even with these advanced protections, using `whatsapp-web.js` remains high-risk:

1. **WhatsApp's ML systems** continuously evolve to detect automation
2. **Phone number bans are permanent** and cannot be appealed
3. **Legal risks** exist for violating WhatsApp's Terms of Service
4. **No official support** from Meta for unofficial integrations

**For business or production use, the official WhatsApp Business API is the only safe, reliable, and legal option.**

This enhanced system should only be used for **personal, low-volume testing** with full awareness of the risks. Always start with extremely conservative limits (e.g., 10-20 messages per day) and monitor closely for any signs of detection.
