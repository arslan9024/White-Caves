2. Fast Property Search (Prisma-like)ts

async function findProperties(f: {
  maxPrice?: number
  minBeds?: number
  community?: string
}) {
  return prisma.property.findMany({
    where: {
      priceAED: { lte: f.maxPrice },
      bedrooms: { gte: f.minBeds },
      community: f.community,
      status: 'available'
    },
    orderBy: { priceAED: 'asc' },
    take: 20
  })
}

3. Very Simple AML Risk Scorets

function quickAMLRisk(tx: {
  amount: number
  isCash: boolean
  isPEP: boolean
  priceDeviation: number // %
}) {
  let score = 0
  if (tx.amount > 2e6)      score += 30
  if (tx.isCash)            score += 40
  if (tx.isPEP)             score += 25
  if (Math.abs(tx.priceDeviation) > 25) score += 20
  return Math.min(100, score)
}

6. Quick Lead Hotness Scorets

function leadScore(lead: {
  budget: number
  urgency: 'now' | 'soon' | 'later'
  views: number
}) {
  let score = 0
  if (lead.budget > 15e6) score += 40
  if (lead.urgency === 'now') score += 35
  if (lead.views > 12) score += 25
  return Math.min(100, score)
}

7. Status Change (atomic)ts

async function changeStatus(id: string, status: string) {
  await prisma.property.update({
    where: { id },
    data: {
      status,
      updatedAt: new Date(),
      ...(status === 'sold' && { soldAt: new Date() })
    }
  })
}

8. Basic Commission Calc (very common)ts

function calculateCommission(
  salePriceAED: number,
  agentSharePercent: number = 50
) {
  const companyCommission = salePriceAED * 0.02 // typical 2%
  const agentCommission = companyCommission * (agentSharePercent / 100)
  return { company: companyCommission, agent: agentCommission }
}
