## 🎯 **Daily Random Campaign Manager**

### **`daily-campaign-manager.js`**
```javascript
const GoogleSheetsService = require('./google-sheets-service');
const moment = require('moment');
const { shuffle } = require('lodash');

class DailyCampaignManager {
    constructor(whatsappClient) {
        this.client = whatsappClient;
        this.sheetsService = new GoogleSheetsService();
        
        // Ultra-strict daily limits
        this.limits = {
            daily: parseInt(process.env.MAX_DAILY_MESSAGES) || 50, // Very conservative
            minBetweenMessagesMs: 5000, // 5 seconds minimum
            maxBetweenMessagesMs: 15000, // 15 seconds maximum
            randomizationFactor: 0.7, // Only use 70% of daily limit for randomization
            minDaysBetweenMessages: 30 // Don't message same person within 30 days
        };
        
        // Tracking
        this.dailyStats = {
            date: moment().format('YYYY-MM-DD'),
            sent: 0,
            failed: 0,
            skippedDueToRecency: 0,
            contactsMessaged: [],
            lastReset: Date.now()
        };
        
        this.resetDailyStats();
    }
    
    /**
     * Main function to run daily random campaign
     */
    async runDailyRandomCampaign(campaignConfig) {
        const {
            spreadsheetId = process.env.GOOGLE_SHEETS_SPREADSHEET_ID,
            range = process.env.GOOGLE_SHEETS_RANGE,
            messageTemplate,
            campaignName = `daily-${moment().format('YYYY-MM-DD')}`,
            excludeRecentlyContacted = true
        } = campaignConfig;
        
        console.log(`🚀 Starting Daily Campaign: ${campaignName}`);
        
        // 1. Check if we've already reached daily limit
        if (this.dailyStats.sent >= this.limits.daily) {
            return {
                success: false,
                error: `Daily limit already reached (${this.limits.daily} messages)`,
                stats: this.dailyStats
            };
        }
        
        // 2. Load and validate contacts from Google Sheets
        let contactsData;
        try {
            contactsData = await this.sheetsService.getValidatedContacts(spreadsheetId, range);
        } catch (error) {
            return {
                success: false,
                error: `Failed to load contacts: ${error.message}`
            };
        }
        
        // 3. Apply 30-day restriction filter
        const eligibleContacts = this.filterByLastContactDate(
            contactsData.validContacts, 
            excludeRecentlyContacted
        );
        
        console.log(`📊 After 30-day filter: ${eligibleContacts.length} eligible contacts`);
        
        // 4. Randomly select contacts (limited by daily remaining quota)
        const selectedContacts = this.selectRandomContacts(
            eligibleContacts,
            this.getAvailableDailyQuota()
        );
        
        if (selectedContacts.length === 0) {
            return {
                success: false,
                error: 'No eligible contacts found after filtering',
                stats: {
                    totalContacts: contactsData.validContacts.length,
                    eligibleAfterFilter: eligibleContacts.length,
                    dailyStats: this.dailyStats
                }
            };
        }
        
        console.log(`🎯 Randomly selected ${selectedContacts.length} contacts for today's campaign`);
        
        // 5. Execute campaign with ultra-conservative pacing
        const results = await this.executeWithConservativePacing(
            selectedContacts,
            messageTemplate,
            campaignName
        );
        
        // 6. Update campaign stats
        this.updateDailyStats(results);
        
        // 7. Update Google Sheets with last contacted dates
        await this.updateContactHistory(selectedContacts, spreadsheetId, range);
        
        return {
            success: true,
            campaignName,
            summary: {
                totalContacts: contactsData.validContacts.length,
                eligibleAfter30Days: eligibleContacts.length,
                randomlySelected: selectedContacts.length,
                successfullySent: results.sent,
                failed: results.failed,
                skippedDueToRecency: this.dailyStats.skippedDueToRecency,
                dailyRemaining: this.limits.daily - this.dailyStats.sent,
                nextAvailableDate: this.getNextAvailableDate(),
                estimatedNextCampaign: this.getNextCampaignTime()
            },
            details: results.details,
            dailyStats: this.dailyStats
        };
    }
    
    /**
     * Filter contacts by last contacted date (30-day restriction)
     */
    filterByLastContactDate(contacts, enforceRestriction = true) {
        if (!enforceRestriction) {
            return contacts;
        }
        
        const thirtyDaysAgo = moment().subtract(this.limits.minDaysBetweenMessages, 'days');
        
        return contacts.filter(contact => {
            // If no last contacted date, assume never contacted
            if (!contact.lastContacted) {
                return true;
            }
            
            const lastContactDate = moment(contact.lastContacted);
            const daysSinceLastContact = moment().diff(lastContactDate, 'days');
            
            // Strict: Must be at least 30 days
            if (daysSinceLastContact < this.limits.minDaysBetweenMessages) {
                // Log skipped contact for debugging
                console.log(`⏸️ Skipping ${contact.name || contact.validatedNumber}: Contacted ${daysSinceLastContact} days ago`);
                this.dailyStats.skippedDueToRecency++;
                return false;
            }
            
            // Extra safety: Also check if we've messaged them today
            const contactedToday = this.dailyStats.contactsMessaged.includes(contact.validatedNumber);
            if (contactedToday) {
                console.log(`⚠️ Skipping ${contact.validatedNumber}: Already messaged today`);
                return false;
            }
            
            return true;
        });
    }
    
    /**
     * Randomly select contacts with distribution control
     */
    selectRandomContacts(contacts, maxToSelect) {
        if (contacts.length <= maxToSelect) {
            // If we have fewer contacts than quota, use all
            return shuffle(contacts);
        }
        
        // Use weighted random selection for better distribution
        return this.weightedRandomSelection(contacts, maxToSelect);
    }
    
    /**
     * Weighted random selection to avoid patterns
     */
    weightedRandomSelection(contacts, n) {
        const shuffled = shuffle([...contacts]);
        
        // Assign weights based on last contacted (older contacts get higher weight)
        const weightedContacts = shuffled.map(contact => {
            let weight = 1;
            
            // Higher weight for contacts never contacted
            if (!contact.lastContacted) {
                weight = 5;
            } else {
                const daysSince = moment().diff(moment(contact.lastContacted), 'days');
                if (daysSince > 90) weight = 4; // > 3 months
                else if (daysSince > 60) weight = 3; // 2-3 months
                else if (daysSince > 30) weight = 2; // 1-2 months
            }
            
            return { contact, weight };
        });
        
        // Normalize weights and select
        const totalWeight = weightedContacts.reduce((sum, wc) => sum + wc.weight, 0);
        const selected = [];
        
        while (selected.length < n && weightedContacts.length > 0) {
            const random = Math.random() * totalWeight;
            let cumulative = 0;
            
            for (let i = 0; i < weightedContacts.length; i++) {
                cumulative += weightedContacts[i].weight;
                if (random <= cumulative) {
                    selected.push(weightedContacts[i].contact);
                    totalWeight -= weightedContacts[i].weight;
                    weightedContacts.splice(i, 1);
                    break;
                }
            }
        }
        
        return selected;
    }
    
    /**
     * Get available daily quota (with randomization factor)
     */
    getAvailableDailyQuota() {
        const remaining = this.limits.daily - this.dailyStats.sent;
        
        // Apply randomization factor (only use 70% of remaining to avoid patterns)
        const randomizedQuota = Math.floor(remaining * this.limits.randomizationFactor);
        
        // Ensure minimum of 1 if we have quota remaining
        return Math.max(1, randomizedQuota);
    }
    
    /**
     * Execute with ultra-conservative pacing
     */
    async executeWithConservativePacing(contacts, messageTemplate, campaignName) {
        const results = {
            sent: 0,
            failed: 0,
            details: []
        };
        
        let consecutiveSuccesses = 0;
        let consecutiveFailures = 0;
        
        for (let i = 0; i < contacts.length; i++) {
            const contact = contacts[i];
            const contactId = contact.validatedNumber;
            
            // Check real-time quota
            if (this.dailyStats.sent >= this.limits.daily) {
                console.log(`⏹️ Daily limit reached. Stopping campaign.`);
                break;
            }
            
            // Dynamic delay calculation
            const delay = this.calculateDynamicDelay(i, consecutiveSuccesses, consecutiveFailures);
            
            console.log(`⏳ Waiting ${delay/1000}s before next message...`);
            await this.delay(delay);
            
            // Personalize message
            const personalizedMessage = this.personalizeMessage(messageTemplate, contact);
            
            try {
                // Send message with timeout
                const sendResult = await this.sendMessageWithTimeout(
                    contactId,
                    personalizedMessage,
                    15000 // 15 second timeout
                );
                
                if (sendResult.success) {
                    results.sent++;
                    this.dailyStats.sent++;
                    this.dailyStats.contactsMessaged.push(contactId);
                    consecutiveSuccesses++;
                    consecutiveFailures = 0;
                    
                    // Update contact success rate for future weighting
                    contact.successRate = (contact.successRate || 0) * 0.9 + 0.1;
                    
                    console.log(`✅ Sent to ${contact.name || contactId} (${results.sent}/${contacts.length})`);
                } else {
                    results.failed++;
                    consecutiveFailures++;
                    consecutiveSuccesses = 0;
                    
                    console.log(`❌ Failed to send to ${contact.name || contactId}: ${sendResult.error}`);
                }
                
                results.details.push({
                    contact: contact.name || contactId,
                    number: contact.validatedNumber,
                    success: sendResult.success,
                    timestamp: new Date().toISOString(),
                    messageId: sendResult.messageId,
                    error: sendResult.error,
                    delayUsed: delay,
                    campaignName
                });
                
                // Adaptive throttling based on failures
                if (consecutiveFailures >= 3) {
                    console.warn('⚠️ Multiple consecutive failures. Pausing for 30 seconds...');
                    await this.delay(30000);
                    consecutiveFailures = 0;
                }
                
                // Success-based pacing (slow down after successes to appear human)
                if (consecutiveSuccesses >= 5) {
                    console.log('🔄 5 consecutive successes. Taking a short break...');
                    await this.delay(10000);
                    consecutiveSuccesses = 0;
                }
                
            } catch (error) {
                results.failed++;
                results.details.push({
                    contact: contact.name || contactId,
                    success: false,
                    error: error.message,
                    timestamp: new Date().toISOString()
                });
                
                console.error(`💥 Error sending to ${contactId}:`, error.message);
                
                // Exponential backoff on critical errors
                if (error.message.includes('blocked') || error.message.includes('banned')) {
                    console.error('🚨 CRITICAL: Potential detection. Stopping campaign immediately.');
                    break;
                }
            }
        }
        
        return results;
    }
    
    /**
     * Calculate dynamic delay that avoids patterns
     */
    calculateDynamicDelay(messageIndex, consecutiveSuccesses, consecutiveFailures) {
        // Base delay increases throughout the day
        const timeOfDay = new Date().getHours();
        let baseDelay;
        
        if (timeOfDay >= 22 || timeOfDay < 8) {
            baseDelay = 20000; // Slower at night
        } else if (timeOfDay >= 8 && timeOfDay < 12) {
            baseDelay = 8000; // Morning pace
        } else if (timeOfDay >= 12 && timeOfDay < 14) {
            baseDelay = 12000; // Slower during lunch
        } else if (timeOfDay >= 14 && timeOfDay < 18) {
            baseDelay = 6000; // Afternoon pace
        } else {
            baseDelay = 10000; // Evening
        }
        
        // Add randomness (30-50% variation)
        const randomFactor = 0.3 + Math.random() * 0.2;
        const randomDelay = baseDelay * randomFactor;
        
        // Increase delay as we send more messages (simulating fatigue)
        const fatigueFactor = Math.log(messageIndex + 1) * 1000;
        
        // Adjust based on recent performance
        let performanceAdjustment = 0;
        if (consecutiveFailures > 0) {
            performanceAdjustment = consecutiveFailures * 2000; // Slow down on failures
        } else if (consecutiveSuccesses > 3) {
            performanceAdjustment = -1000; // Speed up a bit on success streak
        }
        
        // Final delay calculation
        let finalDelay = baseDelay + randomDelay + fatigueFactor + performanceAdjustment;
        
        // Apply limits
        finalDelay = Math.max(this.limits.minBetweenMessagesMs, 
                            Math.min(this.limits.maxBetweenMessagesMs, finalDelay));
        
        // Add occasional long pause (simulating human breaks)
        if (Math.random() < 0.02) { // 2% chance
            const longPause = 45000 + Math.random() * 90000; // 45-135 seconds
            console.log(`🧘 Taking human-like break: ${Math.round(longPause/1000)}s`);
            finalDelay += longPause;
        }
        
        return Math.round(finalDelay);
    }
    
    /**
     * Send message with timeout protection
     */
    async sendMessageWithTimeout(chatId, message, timeoutMs) {
        return new Promise(async (resolve) => {
            const timeoutId = setTimeout(() => {
                resolve({
                    success: false,
                    error: 'Send timeout',
                    messageId: null
                });
            }, timeoutMs);
            
            try {
                const result = await this.client.sendMessage(chatId, message);
                clearTimeout(timeoutId);
                resolve({
                    success: true,
                    messageId: result.id?.id,
                    timestamp: result.timestamp
                });
            } catch (error) {
                clearTimeout(timeoutId);
                resolve({
                    success: false,
                    error: error.message,
                    messageId: null
                });
            }
        });
    }
    
    /**
     * Personalize message with contact data
     */
    personalizeMessage(template, contact) {
        const variables = {
            name: contact.name || 'there',
            firstName: contact.name?.split(' ')[0] || 'there',
            day: moment().format('dddd'),
            date: moment().format('MMMM Do'),
            time: moment().format('h:mm a'),
            company: contact.tags?.find(t => t.includes('company')) || '',
            lastContacted: contact.lastContacted ? 
                moment(contact.lastContacted).fromNow() : 'never'
        };
        
        let personalized = template;
        Object.entries(variables).forEach(([key, value]) => {
            personalized = personalized.replace(new RegExp(`{${key}}`, 'gi'), value);
        });
        
        return personalized;
    }
    
    /**
     * Update contact history in Google Sheets
     */
    async updateContactHistory(contacts, spreadsheetId, range) {
        const batchUpdates = [];
        const today = new Date().toISOString().split('T')[0];
        
        for (const contact of contacts) {
            if (contact.sheetRow) {
                batchUpdates.push({
                    range: `${range.split('!')[0]}!D${contact.sheetRow}`,
                    values: [[today]]
                });
            }
        }
        
        // Update in batches of 10
        for (let i = 0; i < batchUpdates.length; i += 10) {
            const batch = batchUpdates.slice(i, i + 10);
            
            try {
                await this.sheetsService.batchUpdateContactStatus(spreadsheetId, batch);
                console.log(`📝 Updated ${batch.length} contacts in sheet`);
                await this.delay(2000); // Delay between batch updates
            } catch (error) {
                console.error('Failed to update sheet batch:', error.message);
            }
        }
    }
    
    /**
     * Update daily statistics
     */
    updateDailyStats(results) {
        this.dailyStats.failed += results.failed;
        
        // Log to file for tracking
        this.logCampaignStats(results);
        
        // Check if we need to reset (after 24 hours)
        if (Date.now() - this.dailyStats.lastReset > 24 * 60 * 60 * 1000) {
            this.resetDailyStats();
        }
    }
    
    /**
     * Reset daily statistics
     */
    resetDailyStats() {
        const today = moment().format('YYYY-MM-DD');
        
        if (this.dailyStats.date !== today) {
            this.dailyStats = {
                date: today,
                sent: 0,
                failed: 0,
                skippedDueToRecency: 0,
                contactsMessaged: [],
                lastReset: Date.now()
            };
            
            console.log(`🔄 Daily stats reset for ${today}`);
        }
    }
    
    /**
     * Get next available campaign date
     */
    getNextAvailableDate() {
        const today = moment();
        const messagesSent = this.dailyStats.sent;
        
        if (messagesSent < this.limits.daily) {
            return 'Today (quota available)';
        }
        
        // If we've used our quota, next available is tomorrow
        return today.add(1, 'day').format('YYYY-MM-DD');
    }
    
    /**
     * Get next campaign execution time
     */
    getNextCampaignTime() {
        const now = moment();
        const hour = now.hour();
        
        // Optimal times for sending (based on WhatsApp usage patterns)
        const optimalTimes = [
            { start: 9, end: 11, weight: 0.3 },  // Morning
            { start: 14, end: 16, weight: 0.4 }, // Afternoon
            { start: 18, end: 20, weight: 0.3 }  // Evening
        ];
        
        // Find next optimal time window
        for (const window of optimalTimes) {
            if (hour < window.start) {
                return moment().set({ hour: window.start, minute: 0, second: 0 }).format();
            }
        }
        
        // Default to tomorrow morning
        return moment().add(1, 'day').set({ hour: 9, minute: 0, second: 0 }).format();
    }
    
    /**
     * Log campaign statistics for analysis
     */
    logCampaignStats(results) {
        const logEntry = {
            date: moment().format('YYYY-MM-DD HH:mm:ss'),
            campaign: 'daily-random',
            stats: {
                sent: results.sent,
                failed: results.failed,
                dailyTotal: this.dailyStats.sent,
                dailyLimit: this.limits.daily,
                remainingQuota: this.limits.daily - this.dailyStats.sent
            },
            performance: {
                successRate: results.sent / (results.sent + results.failed) || 0,
                averageDelay: this.limits.minBetweenMessagesMs,
                skippedDueToRecency: this.dailyStats.skippedDueToRecency
            }
        };
        
        // In production, write to file or send to monitoring service
        console.log('📈 Campaign Stats:', JSON.stringify(logEntry, null, 2));
    }
    
    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}
```

## 🚀 **Scheduled Daily Campaign Runner**

### **`daily-scheduler.js`**
```javascript
const cron = require('node-cron');
const DailyCampaignManager = require('./daily-campaign-manager');
const moment = require('moment');
const fs = require('fs').promises;
const path = require('path');

class DailyCampaignScheduler {
    constructor(whatsappClient) {
        this.client = whatsappClient;
        this.campaignManager = new DailyCampaignManager(whatsappClient);
        this.campaigns = [];
        this.logFile = path.join(__dirname, 'logs', 'daily-campaigns.log');
        this.initializeLogs();
    }
    
    async initializeLogs() {
        const logDir = path.join(__dirname, 'logs');
        try {
            await fs.mkdir(logDir, { recursive: true });
        } catch (error) {
            // Directory already exists
        }
    }
    
    /**
     * Schedule daily random campaign
     */
    scheduleDailyCampaign(config) {
        const {
            name = 'Daily Random Campaign',
            schedule = '0 9 * * 1-5', // Weekdays at 9 AM
            spreadsheetId,
            range,
            messageTemplates = [] // Multiple templates for variation
        } = config;
        
        console.log(`📅 Scheduling daily campaign: ${name} (${schedule})`);
        
        const job = cron.schedule(schedule, async () => {
            console.log(`⏰ ${moment().format('YYYY-MM-DD HH:mm:ss')} - Running ${name}`);
            
            try {
                // Randomly select a message template
                const randomTemplate = messageTemplates[
                    Math.floor(Math.random() * messageTemplates.length)
                ] || "Hello {firstName}! Hope you're having a great {day}.";
                
                const result = await this.campaignManager.runDailyRandomCampaign({
                    spreadsheetId,
                    range,
                    messageTemplate: randomTemplate,
                    campaignName: name,
                    excludeRecentlyContacted: true
                });
                
                await this.logCampaignResult(name, result);
                
                if (result.success) {
                    console.log(`✅ ${name} completed successfully:`, result.summary);
                } else {
                    console.error(`❌ ${name} failed:`, result.error);
                }
                
            } catch (error) {
                console.error(`💥 Error in ${name}:`, error);
                await this.logCampaignError(name, error);
            }
        }, {
            scheduled: true,
            timezone: "Asia/Kolkata" // Adjust to your timezone
        });
        
        this.campaigns.push({ name, job, config });
        return job;
    }
    
    /**
     * Schedule multiple time slots for better randomization
     */
    scheduleMultipleDailySlots(config) {
        const timeSlots = [
            '0 9 * * 1-5',   // 9:00 AM on weekdays
            '0 14 * * 1-5',  // 2:00 PM on weekdays
            '0 17 * * 1-5',  // 5:00 PM on weekdays
        ];
        
        timeSlots.forEach((schedule, index) => {
            this.scheduleDailyCampaign({
                ...config,
                name: `${config.name} - Slot ${index + 1}`,
                schedule
            });
        });
    }
    
    /**
     * Manual trigger for daily campaign
     */
    async triggerDailyCampaignNow(config) {
        console.log('🚀 Manually triggering daily campaign...');
        
        try {
            const result = await this.campaignManager.runDailyRandomCampaign({
                ...config,
                campaignName: 'Manual-Daily-' + moment().format('YYYY-MM-DD-HHmm')
            });
            
            await this.logCampaignResult('Manual Trigger', result);
            return result;
            
        } catch (error) {
            console.error('Manual trigger failed:', error);
            await this.logCampaignError('Manual Trigger', error);
            throw error;
        }
    }
    
    /**
     * Log campaign results
     */
    async logCampaignResult(campaignName, result) {
        const logEntry = {
            timestamp: moment().format('YYYY-MM-DD HH:mm:ss'),
            campaign: campaignName,
            result: result.success ? 'SUCCESS' : 'FAILED',
            details: {
                sent: result.summary?.successfullySent || 0,
                failed: result.summary?.failed || 0,
                dailyRemaining: result.summary?.dailyRemaining || 0,
                skippedDueToRecency: result.summary?.skippedDueToRecency || 0
            },
            error: result.error,
            nextAvailable: result.summary?.nextAvailableDate
        };
        
        try {
            await fs.appendFile(
                this.logFile, 
                JSON.stringify(logEntry) + '\n'
            );
        } catch (error) {
            console.error('Failed to write log:', error);
        }
    }
    
    async logCampaignError(campaignName, error) {
        const logEntry = {
            timestamp: moment().format('YYYY-MM-DD HH:mm:ss'),
            campaign: campaignName,
            result: 'ERROR',
            error: error.message,
            stack: error.stack
        };
        
        try {
            await fs.appendFile(
                this.logFile, 
                JSON.stringify(logEntry) + '\n'
            );
        } catch (error) {
            console.error('Failed to write error log:', error);
        }
    }
    
    /**
     * Get campaign statistics
     */
    async getCampaignStats(days = 7) {
        try {
            const logs = await fs.readFile(this.logFile, 'utf-8');
            const entries = logs.split('\n')
                .filter(line => line.trim())
                .map(line => JSON.parse(line));
            
            const recentEntries = entries.filter(entry => 
                moment(entry.timestamp).isAfter(moment().subtract(days, 'days'))
            );
            
            const stats = {
                totalCampaigns: recentEntries.length,
                successful: recentEntries.filter(e => e.result === 'SUCCESS').length,
                failed: recentEntries.filter(e => e.result === 'FAILED').length,
                errors: recentEntries.filter(e => e.result === 'ERROR').length,
                totalMessages: recentEntries.reduce((sum, e) => sum + (e.details?.sent || 0), 0),
                averagePerDay: {},
                recentActivities: recentEntries.slice(-5)
            };
            
            // Calculate daily averages
            for (let i = 0; i < days; i++) {
                const date = moment().subtract(i, 'days').format('YYYY-MM-DD');
                const dayEntries = recentEntries.filter(e => 
                    e.timestamp.startsWith(date)
                );
                
                stats.averagePerDay[date] = {
                    campaigns: dayEntries.length,
                    messages: dayEntries.reduce((sum, e) => sum + (e.details?.sent || 0), 0)
                };
            }
            
            return stats;
        } catch (error) {
            console.error('Failed to read stats:', error);
            return { error: 'No stats available' };
        }
    }
    
    /**
     * Stop all scheduled campaigns
     */
    stopAllCampaigns() {
        this.campaigns.forEach(campaign => {
            campaign.job.stop();
            console.log(`🛑 Stopped campaign: ${campaign.name}`);
        });
        
        this.campaigns = [];
    }
    
    /**
     * List all scheduled campaigns
     */
    listScheduledCampaigns() {
        return this.campaigns.map(c => ({
            name: c.name,
            schedule: c.config.schedule,
            nextRun: this.getNextRunTime(c.job),
            status: 'ACTIVE'
        }));
    }
    
    getNextRunTime(job) {
        const nextDates = job.nextDates(1);
        return nextDates && nextDates.length > 0 
            ? moment(nextDates[0]).format('YYYY-MM-DD HH:mm:ss')
            : 'Not scheduled';
    }
}

module.exports = DailyCampaignScheduler;
```

## 🔐 1. Google Sheets Service (Contact Manager)

This service securely fetches and validates contact numbers from your Google Sheets.

### **`google-sheets-service.js`**
```javascript
const { google } = require('googleapis');
const { authenticate } = require('@google-cloud/local-auth');
const path = require('path');
require('dotenv').config();

class GoogleSheetsService {
    constructor() {
        this.SCOPES = ['https://www.googleapis.com/auth/spreadsheets.readonly'];
        this.CREDENTIALS_PATH = path.join(process.cwd(), 'credentials.json');
        this.sheets = null;
        this.initialize();
    }

    async initialize() {
        try {
            const auth = await authenticate({
                scopes: this.SCOPES,
                keyfilePath: this.CREDENTIALS_PATH,
            });
            this.sheets = google.sheets({ version: 'v4', auth });
            console.log('Google Sheets service initialized');
        } catch (error) {
            console.error('Failed to initialize Google Sheets:', error);
            throw error;
        }
    }

    /**
     * Reads and validates contact numbers from a Google Sheet
     * @param {string} spreadsheetId - ID of the Google Sheet
     * @param {string} range - A1 notation range (e.g., 'Contacts!A2:B')
     * @returns {Promise<Array>} - Array of validated contact objects
     */
    async getValidatedContacts(spreadsheetId, range) {
        if (!this.sheets) {
            await this.initialize();
        }

        try {
            // Fetch data from Google Sheets[reference:0]
            const result = await this.sheets.spreadsheets.values.get({
                spreadsheetId,
                range,
            });

            const rows = result.data.values;
            if (!rows || rows.length === 0) {
                console.log('No data found in sheet');
                return [];
            }

            // Process and validate each row
            const validatedContacts = rows.map((row, index) => {
                const [name, rawNumber, tags, lastContacted] = row;
                const validation = this.validateAndFormatNumber(rawNumber);
                
                return {
                    id: `contact-${index + 1}`,
                    name: name || `Contact ${index + 1}`,
                    rawNumber,
                    validatedNumber: validation.correctedNumber,
                    isValid: validation.isValid,
                    countryCode: validation.countryCode,
                    tags: tags ? tags.split(',').map(t => t.trim()) : [],
                    lastContacted: lastContacted || null,
                    sheetRow: index + 2, // For reference back to sheet
                    validationErrors: validation.errors || []
                };
            });

            // Separate valid and invalid contacts
            const validContacts = validatedContacts.filter(c => c.isValid);
            const invalidContacts = validatedContacts.filter(c => !c.isValid);

            console.log(`Loaded ${validContacts.length} valid, ${invalidContacts.length} invalid contacts`);
            
            return {
                validContacts,
                invalidContacts,
                total: validatedContacts.length,
                timestamp: new Date().toISOString()
            };

        } catch (error) {
            console.error('Error reading Google Sheet:', error);
            throw new Error(`Failed to read contacts: ${error.message}`);
        }
    }

    /**
     * Validates and formats phone numbers for WhatsApp
     */
    validateAndFormatNumber(rawNumber) {
        // Remove all non-digit characters
        let cleaned = rawNumber.toString().replace(/\D/g, '');
        const errors = [];
        
        // Check minimum length
        if (cleaned.length < 10) {
            errors.push('Number too short');
            return { isValid: false, correctedNumber: null, errors };
        }
        
        // Country code detection and formatting
        let correctedNumber;
        let countryCode = '';
        
        // Handle common country codes
        if (cleaned.startsWith('1') && cleaned.length === 11) {
            countryCode = '1';
            correctedNumber = cleaned + '@c.us';
        } else if (cleaned.startsWith('91') && cleaned.length === 12) {
            countryCode = '91';
            correctedNumber = cleaned + '@c.us';
        } else if (cleaned.startsWith('44') && cleaned.length === 12) {
            countryCode = '44';
            correctedNumber = cleaned + '@c.us';
        } else if (cleaned.length === 10) {
            // Assume Indian number if 10 digits
            countryCode = '91';
            correctedNumber = '91' + cleaned + '@c.us';
        } else if (cleaned.length >= 11 && cleaned.length <= 15) {
            // Generic international format
            correctedNumber = cleaned + '@c.us';
        } else {
            errors.push('Unrecognized number format');
            return { isValid: false, correctedNumber: null, errors };
        }
        
        return {
            isValid: true,
            correctedNumber,
            countryCode,
            original: rawNumber,
            errors
        };
    }

    /**
     * Updates contact status in the sheet (e.g., last contacted)
     */
    async updateContactStatus(spreadsheetId, range, rowIndex, status) {
        try {
            await this.sheets.spreadsheets.values.update({
                spreadsheetId,
                range: `${range.split('!')[0]}!D${rowIndex}`, // Update column D
                valueInputOption: 'RAW',
                resource: {
                    values: [[status]]
                }
            });
            console.log(`Updated contact status for row ${rowIndex}`);
        } catch (error) {
            console.error('Failed to update sheet:', error);
        }
    }
}
```

### **`enhanced-campaign-manager.js`**
```javascript
const CircuitBreaker = require('opossum');
const GoogleSheetsService = require('./google-sheets-service');

class EnhancedCampaignManager {
    constructor(whatsappClient) {
        this.client = whatsappClient;
        this.sheetsService = new GoogleSheetsService();
        
        // Strict multi-level limits
        this.limits = {
            daily: parseInt(process.env.MAX_DAILY_CAMPAIGN_MESSAGES) || 80,
            hourly: 12,
            perContactDaily: 2,
            perCampaign: 50,
            minDelayBetweenMessages: 3000, // 3 seconds minimum
            maxDelayBetweenMessages: 10000, // 10 seconds maximum
            campaignCooldownHours: 6
        };
        
        // Trackers
        this.counters = {
            daily: 0,
            hourly: {},
            contactDaily: {},
            campaignHistory: []
        };
        
        // Circuit breaker for sending messages[reference:1]
        this.breaker = new CircuitBreaker(async (chatId, message) => {
            return await this.client.sendMessage(chatId, message);
        }, {
            timeout: 10000,
            errorThresholdPercentage: 40, // Trip if 40% of sends fail
            resetTimeout: 30000,
            rollingCountTimeout: 60000,
            rollingCountBuckets: 10
        });
        
        this.setupBreakerEvents();
        this.resetCounters();
    }
    
    setupBreakerEvents() {
        this.breaker.on('open', () => {
            console.warn('🚨 CIRCUIT BREAKER OPEN: Message sending suspended');
            this.logAnomaly('circuit_breaker_opened', {
                timestamp: new Date().toISOString(),
                failureRate: this.breaker.stats.failures / this.breaker.stats.requests * 100
            });
        });
        
        this.breaker.on('halfOpen', () => {
            console.log('🔶 Circuit breaker half-open: Testing recovery');
        });
        
        this.breaker.on('close', () => {
            console.log('✅ Circuit breaker closed: Service recovered');
        });
        
        this.breaker.fallback(async (chatId, message) => {
            return {
                success: false,
                error: 'Service temporarily unavailable due to high failure rate',
                fallback: true,
                timestamp: new Date().toISOString()
            };
        });
    }
    
    /**
     * Main campaign execution with enhanced protections
     */
    async executeCampaign(campaignConfig) {
        const {
            spreadsheetId,
            range,
            messageTemplate,
            campaignId = `campaign-${Date.now()}`
        } = campaignConfig;
        
        // 1. Validate campaign schedule
        const scheduleCheck = this.checkCampaignSchedule();
        if (!scheduleCheck.allowed) {
            return {
                success: false,
                error: `Schedule restriction: ${scheduleCheck.reason}`,
                waitMinutes: scheduleCheck.waitMinutes
            };
        }
        
        // 2. Load contacts from Google Sheets
        let contacts;
        try {
            contacts = await this.sheetsService.getValidatedContacts(spreadsheetId, range);
        } catch (error) {
            return {
                success: false,
                error: `Failed to load contacts: ${error.message}`
            };
        }
        
        // 3. Apply contact filters
        const filteredContacts = this.filterContacts(contacts.validContacts, campaignConfig);
        
        // 4. Check campaign limits
        const limitCheck = this.checkCampaignLimits(filteredContacts.length, campaignId);
        if (!limitCheck.allowed) {
            return {
                success: false,
                error: `Campaign limit exceeded: ${limitCheck.reason}`,
                details: limitCheck.details
            };
        }
        
        // 5. Execute with strict pacing
        const results = await this.executeWithPacing(filteredContacts, messageTemplate, campaignId);
        
        // 6. Record campaign completion
        this.recordCampaignCompletion(campaignId, results);
        
        return {
            success: true,
            campaignId,
            summary: {
                totalContacts: filteredContacts.length,
                sent: results.sent,
                failed: results.failed,
                dailyRemaining: this.limits.daily - this.counters.daily,
                estimatedCompletion: new Date(Date.now() + (filteredContacts.length * this.limits.minDelayBetweenMessages))
            },
            details: results.details
        };
    }
    
    /**
     * Check if campaign can run based on schedule
     */
    checkCampaignSchedule() {
        const now = new Date();
        const hour = now.getHours();
        const day = now.getDay();
        
        // Night off (10 PM to 8 AM)
        if (hour >= 22 || hour < 8) {
            const waitHours = hour >= 22 ? (32 - hour) : (8 - hour);
            return {
                allowed: false,
                reason: 'Night off period',
                waitMinutes: waitHours * 60
            };
        }
        
        // Lunch break (12 PM to 1:30 PM)
        if (hour === 12 || (hour === 13 && now.getMinutes() < 30)) {
            const waitMinutes = hour === 12 ? 90 : (30 - now.getMinutes());
            return {
                allowed: false,
                reason: 'Lunch break',
                waitMinutes
            };
        }
        
        // Dinner break (6 PM to 7:30 PM)
        if (hour === 18 || (hour === 19 && now.getMinutes() < 30)) {
            const waitMinutes = hour === 18 ? 90 : (30 - now.getMinutes());
            return {
                allowed: false,
                reason: 'Dinner break',
                waitMinutes
            };
        }
        
        // Weekend restriction
        if (day === 0 || day === 6) { // Sunday or Saturday
            const waitDays = day === 6 ? 2 : 1; // Wait until Monday
            return {
                allowed: false,
                reason: 'Weekend restriction',
                waitMinutes: waitDays * 24 * 60
            };
        }
        
        return { allowed: true };
    }
    
    /**
     * Filter contacts based on campaign rules
     */
    filterContacts(contacts, campaignConfig) {
        const { excludeRecentlyContacted = true, minDaysBetweenMessages = 7 } = campaignConfig;
        
        return contacts.filter(contact => {
            // Check daily contact limit
            const contactKey = contact.validatedNumber;
            const today = new Date().toDateString();
            const contactDailyCount = this.counters.contactDaily[contactKey] || 0;
            
            if (contactDailyCount >= this.limits.perContactDaily) {
                return false;
            }
            
            // Check if recently contacted
            if (excludeRecentlyContacted && contact.lastContacted) {
                const lastContact = new Date(contact.lastContacted);
                const daysSince = (new Date() - lastContact) / (1000 * 60 * 60 * 24);
                if (daysSince < minDaysBetweenMessages) {
                    return false;
                }
            }
            
            return true;
        });
    }
    
    /**
     * Execute campaign with human-like pacing
     */
    async executeWithPacing(contacts, messageTemplate, campaignId) {
        const results = {
            sent: 0,
            failed: 0,
            details: []
        };
        
        for (const contact of contacts) {
            // Check real-time limits before each message
            const realTimeCheck = this.checkRealTimeLimits();
            if (!realTimeCheck.allowed) {
                results.details.push({
                    contact: contact.validatedNumber,
                    success: false,
                    error: realTimeCheck.reason,
                    skipped: true
                });
                results.failed++;
                continue;
            }
            
            // Personalize message
            const personalizedMessage = this.personalizeMessage(messageTemplate, contact);
            
            // Calculate dynamic delay
            const delay = this.calculateDynamicDelay(results.sent);
            await this.delay(delay);
            
            // Send message through circuit breaker
            try {
                const sendResult = await this.breaker.fire(contact.validatedNumber, personalizedMessage);
                
                if (sendResult.success !== false) {
                    results.sent++;
                    this.incrementCounters(contact.validatedNumber);
                    
                    // Update contact status in sheet
                    await this.sheetsService.updateContactStatus(
                        process.env.SPREADSHEET_ID,
                        process.env.RANGE,
                        contact.sheetRow,
                        new Date().toISOString()
                    );
                } else {
                    results.failed++;
                }
                
                results.details.push({
                    contact: contact.validatedNumber,
                    success: sendResult.success !== false,
                    messageId: sendResult.id?.id,
                    timestamp: new Date().toISOString(),
                    error: sendResult.error
                });
                
            } catch (error) {
                results.failed++;
                results.details.push({
                    contact: contact.validatedNumber,
                    success: false,
                    error: error.message,
                    timestamp: new Date().toISOString()
                });
                
                // Log anomaly for ML detection
                this.logAnomaly('send_message_failed', {
                    contact: contact.validatedNumber,
                    error: error.message,
                    campaignId
                });
            }
            
            // Check if circuit breaker is open
            if (this.breaker.opened) {
                console.warn('Campaign paused: Circuit breaker is open');
                break;
            }
        }
        
        return results;
    }
    
    /**
     * Calculate dynamic delay that mimics human behavior
     */
    calculateDynamicDelay(messageCount) {
        // Base delay increases as we send more messages
        const baseDelay = this.limits.minDelayBetweenMessages + 
                         (Math.log(messageCount + 1) * 1000);
        
        // Add random variation (30-70% of base delay)
        const randomFactor = 0.3 + Math.random() * 0.4;
        const randomDelay = baseDelay * randomFactor;
        
        // Occasional longer pauses (simulating human breaks)
        const pauseChance = 0.05;
        const pause = Math.random() < pauseChance ? 
                      (30000 + Math.random() * 60000) : 0; // 30-90 second pause
        
        return Math.min(
            this.limits.maxDelayBetweenMessages,
            baseDelay + randomDelay + pause
        );
    }
    
    /**
     * Personalize message with contact data
     */
    personalizeMessage(template, contact) {
        return template
            .replace(/{name}/g, contact.name)
            .replace(/{number}/g, contact.rawNumber)
            .replace(/{tags}/g, contact.tags.join(', '))
            .replace(/{time}/g, new Date().toLocaleTimeString());
    }
    
    /**
     * Check real-time limits
     */
    checkRealTimeLimits() {
        const now = new Date();
        const hour = now.getHours();
        const today = now.toDateString();
        
        // Daily limit
        if (this.counters.daily >= this.limits.daily) {
            return {
                allowed: false,
                reason: `Daily limit reached (${this.limits.daily})`,
                resetTime: 'Tomorrow 00:00'
            };
        }
        
        // Hourly limit
        const hourCount = this.counters.hourly[hour] || 0;
        if (hourCount >= this.limits.hourly) {
            const nextHour = hour === 23 ? 0 : hour + 1;
            return {
                allowed: false,
                reason: `Hourly limit reached (${this.limits.hourly})`,
                waitMinutes: 60 - now.getMinutes()
            };
        }
        
        return { allowed: true };
    }
    
    /**
     * Increment counters after successful send
     */
    incrementCounters(contactNumber) {
        const now = new Date();
        const hour = now.getHours();
        const today = now.toDateString();
        
        // Daily counter
        this.counters.daily++;
        
        // Hourly counter
        if (!this.counters.hourly[hour]) {
            this.counters.hourly[hour] = 0;
        }
        this.counters.hourly[hour]++;
        
        // Contact daily counter
        if (!this.counters.contactDaily[contactNumber]) {
            this.counters.contactDaily[contactNumber] = 0;
        }
        this.counters.contactDaily[contactNumber]++;
    }
    
    /**
     * Log anomalies for ML detection analysis
     */
    logAnomaly(type, data) {
        const anomalyLog = {
            type,
            timestamp: new Date().toISOString(),
            data,
            counters: {
                daily: this.counters.daily,
                hourly: this.counters.hourly[new Date().getHours()] || 0
            },
            breakerStatus: this.breaker.stats
        };
        
        // In production, send to monitoring service
        console.warn('ANOMALY DETECTED:', JSON.stringify(anomalyLog, null, 2));
        
        // Auto-adjust limits if anomalies are frequent
        if (type === 'circuit_breaker_opened') {
            this.adjustLimits('reduce');
        }
    }
    
    /**
     * Dynamically adjust limits based on anomalies
     */
    adjustLimits(direction) {
        if (direction === 'reduce') {
            this.limits.daily = Math.max(20, Math.floor(this.limits.daily * 0.7));
            this.limits.hourly = Math.max(3, Math.floor(this.limits.hourly * 0.7));
            console.warn(`Limits reduced due to anomalies: ${this.limits.daily} daily, ${this.limits.hourly} hourly`);
        }
    }
    
    /**
     * Reset counters daily
     */
    resetCounters() {
        // Reset at midnight
        const now = new Date();
        const night = new Date(now);
        night.setHours(24, 0, 0, 0);
        const msUntilMidnight = night - now;
        
        setTimeout(() => {
            this.counters.daily = 0;
            this.counters.hourly = {};
            this.counters.contactDaily = {};
            console.log('Daily counters reset');
            
            // Schedule next reset
            this.resetCounters();
        }, msUntilMidnight);
    }
    
    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

module.exports = EnhancedCampaignManager;
```

```javascript
                { time: '14:00-18:00', maxPerHour: 25, label: 'Afternoon' },
                { time: '18:00-22:00', maxPerHour: 15, label: 'Evening' }
            ]

        this.dailyCounts = {};

    isWithinSchedule(start, end) {
        const now = new Date();
        const currentTime = now.getHours() * 100 + now.getMinutes();
        const startTime = parseInt(start.replace(':', ''));
        const endTime = parseInt(end.replace(':',''));
        
        if (startTime <= endTime) {
            return currentTime >= startTime && currentTime <= endTime;
        } else {
            //跨越午夜的时间段
            return currentTime >= startTime || currentTime <= endTime;
        }
    }
    
    isOffHour() {
        const now = new Date();
        const dayOfWeek = now.getDay();
        
        // 检查周末
        if ((dayOfWeek === 6 && this.schedules.weekends.saturday.enabled === false) ||
            (dayOfWeek === 0 && this.schedules.weekends.sunday.enabled === false)) {
            return { isOff: true, reason: 'Weekend' };
        }
        
        // 检查休息时间
        for (const period of this.schedules.offHours) {
            if (this.isWithinSchedule(period.start, period.end)) {
                return { 
                    isOff: true, 
                    reason: period.label,
                    nextAvailable: this.getNextAvailableTime(period.end)
                };
            }
        }
        
        return { isOff: false };
    }
    
    canSendMessage() {
        const offCheck = this.isOffHour();
        if (offCheck.isOff) {
            return {
                allowed: false,
                reason: `Schedule restriction: ${offCheck.reason}`,
                waitMinutes: offCheck.nextAvailable || 0
            };
        }
        
        // 检查小时限制
        const hour = new Date().getHours();
        const currentLimit = this.getCurrentTimeLimit();
        
        if (!this.hourlyCounts[hour]) this.hourlyCounts[hour] = 0;
        if (this.hourlyCounts[hour] >= currentLimit.maxPerHour) {
            const nextHour = hour === 23 ? 0 : hour + 1;
            const waitMinutes = 60 - new Date().getMinutes();
            return {
                allowed: false,
                reason: `Hourly limit reached (${currentLimit.maxPerHour}/hour)`,
                waitMinutes: waitMinutes
            };
        }
        
        // 检查每日限制
        const today = new Date().toDateString();
        if (!this.dailyCounts[today]) this.dailyCounts[today] = 0;
        
        const dailyLimit = parseInt(process.env.MAX_DAILY_MESSAGES || 150);
        if (this.dailyCounts[today] >= dailyLimit) {
            return {
                allowed: false,
                reason: `Daily limit reached (${dailyLimit}/day)`,
                waitMinutes: 1440 // 24小时
            };
        }
        
        return { allowed: true };
    }
    
    recordMessageSent() {
        const hour = new Date().getHours();
        const today = new Date().toDateString();
        
        this.hourlyCounts[hour] = (this.hourlyCounts[hour] || 0) + 1;
        this.dailyCounts[today] = (this.dailyCounts[today] || 0) + 1;
    }

        const now = new Date();
        const currentTime = now.getHours() * 100 + now.getMinutes();
        
        for (const limit of this.schedules.timeBasedLimits) {
            const [start, end] = limit.time.split('-');
            const startTime = parseInt(start.replace(':', ''));
            const endTime = parseInt(end.replace(':',''));
            
            if (currentTime >= startTime && currentTime <= endTime) {
                return limit;
            }
        }
        
        return { maxPerHour: 15, label: 'Default' };
    }
    
    getNextAvailableTime(endTime) {
        const now = new Date();
        const [endHour, endMinute] = endTime.split(':').map(Number);
        const endDate = new Date(now);
        endDate.setHours(endHour, endMinute, 0, 0);
        
        if (endDate <= now) {
            endDate.setDate(endDate.getDate() + 1);
        }
        
        return Math.ceil((endDate - now) / (1000 * 60));
    }
    
    resetCounters() {
        // 每小时重置小时计数器
        setInterval(() => {
            const currentHour = new Date().getHours();
            Object.keys(this.hourlyCounts).forEach(hour => {
                if (parseInt(hour) < currentHour) {

                }
            });

        // 每天午夜重置每日计数器
        const now = new Date();
        const night = new Date(now);
        night.setHours(24, 0, 0, 0);
        setTimeout(() => {
            this.dailyCounts = {};
            setInterval(() => {
                this.dailyCounts = {};
            }, 24 * 60 * 60 * 1000);
        }, night - now);
    }
```

```javascript
            maxMessages: pattern.messagesPerSession.min + 
                        Math.random() * (pattern.messagesPerSession.max - pattern.messagesPerSession.min)
        };
        
        console.log(`Started new ${selectedPattern} session. Planned duration: ${Math.round(this.currentSession.plannedDuration)}min`);
        
        // 安排会话结束
        this.scheduleSessionEnd();
        
        return this.currentSession;
    }
    
    scheduleSessionEnd() {
        const durationMs = this.currentSession.plannedDuration * 60 * 1000;
        
        setTimeout(() => {
            console.log('Session time limit reached. Preparing to disconnect...');
            this.endSession('time_limit');
        }, durationMs);
    }
    
    recordMessage() {
        this.currentSession.messageCount++;
        
        // 检查是否达到消息限制
        if (this.currentSession.messageCount >= this.currentSession.maxMessages) {
            console.log('Message limit reached for this session');
            this.endSession('message_limit');
            return false;
        }
        
        // 检查是否需要休息
        if (Date.now() >= this.currentSession.nextBreak) {
            console.log('Break time reached');
            this.takeBreak();
            return false;
        }
        
        return true;
    }
    
    async takeBreak() {
        const breakDuration = 5 + Math.random() * 15; // 5-20分钟
        
        console.log(`Taking a break for ${Math.round(breakDuration)} minutes`);
        
        // 模拟离开
        await this.simulateAwayTime(breakDuration * 60 * 1000);
        
        // 恢复会话
        this.currentSession.nextBreak = Date.now() + (
            this.sessionPatterns[this.currentSession.pattern].breaks.min + 
            Math.random() * (this.sessionPatterns[this.currentSession.pattern].breaks.max - 
            this.sessionPatterns[this.currentSession.pattern].breaks.min)
        ) * 60 * 1000;
        
        console.log('Break ended, resuming session');
    }
    
    async simulateAwayTime(duration) {
        // 这里可以添加实际的不活动模拟
        // 例如：最小化浏览器、减少活动等
        
        return new Promise(resolve => setTimeout(resolve, duration));
    }
    
    endSession(reason = 'normal') {
        const sessionData = {
            ...this.currentSession,
            endTime: Date.now(),
            duration: (Date.now() - this.currentSession.startTime) / (1000 * 60),
            reason
        };
        
        this.sessionHistory.push(sessionData);
        
        // 保留最近100个会话
        if (this.sessionHistory.length > 100) {
            this.sessionHistory.shift();
        }
        
        console.log(`Session ended. Reason: ${reason}, Duration: ${Math.round(sessionData.duration)}min, Messages: ${sessionData.messageCount}`);
        
        // 开始新会话前等待
        const waitTime = this.getInterSessionWaitTime();
        console.log(`Waiting ${Math.round(waitTime / (1000 * 60))} minutes before next session`);
        
        setTimeout(() => {
            this.startNewSession();
        }, waitTime);
    }
    
    getInterSessionWaitTime() {
        // 基于一天中的时间计算等待时间
        const hour = new Date().getHours();
        let baseWait;
        
        if (hour >= 22 || hour < 6) {
            baseWait = 4 * 60 * 60 * 1000; // 夜间等待4小时
        } else if (hour >= 6 && hour < 9) {
            baseWait = 30 * 60 * 1000; // 早晨等待30分钟
        } else if (hour >= 9 && hour < 18) {
            baseWait = (60 + Math.random() * 120) * 60 * 1000; // 工作时间等待1-3小时
        } else {
            baseWait = 2 * 60 * 60 * 1000; // 晚上等待2小时
        }
        
        // 添加随机变化
        const variation = baseWait * 0.3;
        return baseWait + (Math.random() * variation * 2) - variation;
    }
    
    getAnomalyScore() {
        // 计算当前会话是否异常
        if (this.sessionHistory.length < 3) return 0;
        
        const recentSessions = this.sessionHistory.slice(-3);
        const avgDuration = recentSessions.reduce((sum, s) => sum + s.duration, 0) / recentSessions.length;
        const avgMessages = recentSessions.reduce((sum, s) => sum + s.messageCount, 0) / recentSessions.length;
        
        const currentDuration = (Date.now() - this.currentSession.startTime) / (1000 * 60);
        const durationDiff = Math.abs(currentDuration - avgDuration) / avgDuration;
        const messageDiff = Math.abs(this.currentSession.messageCount - avgMessages) / avgMessages;
        
        return Math.max(durationDiff, messageDiff);
    }
}
```

```javascript
class SecureWhatsAppSystem {

            this.scheduleManager.canSendMessage(),
            this.checkMessageLimits(chatId),
            this.checkContactLimits(chatId),
            this.checkConversationLimits(chatId)
        ];
        
        for (const check of checks) {
            if (!check.allowed) {
                return check;
            }
        }
        
        return { allowed: true };
    }
    
    checkMessageLimits(chatId) {

        if (dailyCount >= this.messageLimits.daily) {

                allowed: false,
                reason: `Daily limit reached (${this.messageLimits.daily})`
            };
        }

        if (hourlyCount >= this.messageLimits.hourly) {
            return {
                allowed: false,
                reason: `Hourly limit reached (${this.messageLimits.hourly})`
            };
        }
        
        return { allowed: true };
    }
    
    setupAPI() {

        this.app.post('/api/v1/secure/send', async (req, res) => {
            try {

                });

            }
        });

    }

    }

    }
}
```

```javascript
class MessageQueue {
    constructor() {
        this.queue = [];
        this.processing = false;
        this.delayBetweenMessages = 2000; // Minimum 2 seconds between messages
        this.dailyLimit = 250; // WhatsApp's unofficial daily limit
        this.sentToday = 0;
        this.resetInterval = setInterval(() => {
            this.sentToday = 0;
            console.log('Daily counter reset');
        }, 24 * 60 * 60 * 1000);
    }

    async add(message) {
        if (this.sentToday >= this.dailyLimit) {
            throw new Error('Daily message limit reached');
        }
        
        return new Promise((resolve) => {
            this.queue.push({ message, resolve });
            this.processQueue();
        });
    }

    async processQueue() {
        if (this.processing || this.queue.length === 0) return;
        
        this.processing = true;
        const item = this.queue.shift();
        
        try {
            const result = await sendMessage(item.message.number, item.message.content);
            this.sentToday++;
            item.resolve(result);
        } catch (error) {
            console.error('Queue processing error:', error);
            item.resolve({ success: false, error: error.message });
        }

        // Wait before processing next message
        setTimeout(() => {
            this.processing = false;
            this.processQueue();
        }, this.delayBetweenMessages + Math.random() * 1000); // Random delay
    }
}
```

```javascript
class SecureMessageSender {
    constructor() {

        this.dailySent = 0;

            this.processQueue();
        });
    }

        this.isSending = true;

        try {

            const result = await client.sendMessage(chatId, message);

                to: number,
                dailyRemaining: this.maxDaily - this.dailySent
            });
        } catch (error) {
            logger.error(`Failed to send to ${number}:`, error.message);
            reject(error);
        } finally {
            this.isSending = false;
            this.processQueue();
        }
    }
```

```javascript
/**
 * Validate and correct phone number format
 * @param {string} number - Phone number to validate
 * @returns {object} {isValid: boolean, correctedNumber: string, countryCode: string}
 */
function validateAndCorrectNumber(number) {
    // Remove all non-digit characters
    let cleaned = number.replace(/\D/g, '');
    
    // Common country codes for validation
    const countryCodes = {
        '1': 'US/Canada',
        '44': 'UK',
        '91': 'India',
        '86': 'China',
        '33': 'France',
        '49': 'Germany',
        '81': 'Japan',
        '55': 'Brazil',
        '7': 'Russia',
        '61': 'Australia'
    };
    
    // Check if number already has country code
    let countryCode = '';
    let correctedNumber = '';
    let isValid = false;
    
    // Check for known country codes
    for (const [code, country] of Object.entries(countryCodes)) {
        if (cleaned.startsWith(code)) {
            countryCode = code;
            correctedNumber = cleaned;
            isValid = true;
            break;
        }
    }
    
    // If no country code found, try to detect
    if (!isValid) {
        // For demonstration, assuming Indian numbers if 10 digits
        if (cleaned.length === 10) {
            countryCode = '91';
            correctedNumber = '91' + cleaned;
            isValid = true;
        } else if (cleaned.length >= 10 && cleaned.length <= 15) {
            // Generic formatting
            correctedNumber = cleaned;
            isValid = true;
        }
    }
    
    // Format for WhatsApp (add @c.us for non-group chats)
    if (isValid && !correctedNumber.includes('@')) {
        correctedNumber = correctedNumber + '@c.us';
    }
    
    return {
        isValid,
        correctedNumber,
        countryCode,
        original: number
    };
}

/**
 * Run a messaging campaign to multiple numbers
 * @param {Array} numbers - Array of phone numbers
 * @param {string} message - Message to send
 * @param {object} options - Campaign options
 */
async function runCampaign(numbers, message, options = {}) {
    const results = {
        total: numbers.length,
        successful: 0,
        failed: 0,
        details: []
    };
    
    const {
        delayBetweenMessages = 2000, // 2 seconds delay
        batchSize = 10,
        maxRetries = 3
    } = options;
    
    // Process in batches to avoid rate limiting
    for (let i = 0; i < numbers.length; i += batchSize) {
        const batch = numbers.slice(i, i + batchSize);
        
        // Process each number in the batch
        for (const number of batch) {
            let retries = 0;
            let sent = false;
            
            while (retries < maxRetries && !sent) {
                try {
                    const result = await sendMessage(number, message);
                    
                    results.details.push({
                        number,
                        success: result.success,
                        timestamp: new Date().toISOString(),
                        retries: retries,
                        error: result.error || null
                    });
                    
                    if (result.success) {
                        results.successful++;
                    } else {
                        results.failed++;
                    }
                    
                    sent = true;
                } catch (error) {
                    retries++;
                    if (retries === maxRetries) {
                        results.details.push({
                            number,
                            success: false,
                            timestamp: new Date().toISOString(),
                            retries: retries,
                            error: `Failed after ${maxRetries} retries: ${error.message}`
                        });
                        results.failed++;
                    }
                }
            }
            
            // Delay between messages in the same batch
            if (delayBetweenMessages > 0) {
                await new Promise(resolve => setTimeout(resolve, delayBetweenMessages));
            }
        }
        
        // Optional delay between batches
        if (options.delayBetweenBatches && i + batchSize < numbers.length) {
            await new Promise(resolve => setTimeout(resolve, options.delayBetweenBatches));
        }
    }
    
    return results;
}
```

```html
        const socket = io();
        let isClientReady = false;

        // Socket event handlers
        socket.on('qr', (qrData) => {
            document.getElementById('status').textContent = 'Scan QR Code with WhatsApp';
            document.getElementById('status').className = 'status authenticating';
            document.getElementById('qrcode').innerHTML = '';
            QRCode.toCanvas(document.getElementById('qrcode'), qrData, { width: 200 });
        });

        socket.on('status', (newStatus) => {
            const statusEl = document.getElementById('status');
            statusEl.textContent = newStatus;
            
            if (newStatus.includes('ready')) {
                statusEl.className = 'status connected';
                document.getElementById('qrcode').innerHTML = '';
                isClientReady = true;
            } else if (newStatus.includes('Authenticated')) {
                statusEl.className = 'status authenticating';
            } else if (newStatus.includes('Disconnected')) {
                statusEl.className = 'status disconnected';
                isClientReady = false;
            }
        });

        socket.on('client-ready', (ready) => {
            isClientReady = ready;
        });

        socket.on('message', (msg) => {
            addMessageToLog(msg, 'incoming');
        });

        socket.on('message_sent', (msg) => {
            addMessageToLog({
                from: 'You → ' + msg.to,
                body: msg.body,
                timestamp: msg.timestamp
            }, 'outgoing');
        });

        // Tab switching
        function switchTab(tabName) {
            // Hide all tabs
            document.querySelectorAll('.tab-content').forEach(tab => {
                tab.classList.remove('active');

            document.querySelectorAll('.tab').forEach(tab => {
                tab.classList.remove('active');
            });
            
            // Show selected tab
            document.getElementById(`tab-${tabName}`).classList.add('active');
            event.target.classList.add('active');

        // Add message to log
        function addMessageToLog(msg, type) {
            const log = document.getElementById('messageLog');
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${type}`;
            
            const time = new Date(msg.timestamp * 1000).toLocaleTimeString();
            messageDiv.innerHTML = `
                <strong>${msg.from}</strong> <small>${time}</small><br>
                ${msg.body}
            `;
            
            log.prepend(messageDiv);
        }

        // API Functions
        async function sendSingleMessage() {
            if (!isClientReady) {
                showResult('single-result', 'Client not ready. Please authenticate first.', 'error');
                return;
            }

            const number = document.getElementById('single-number').value;
            const message = document.getElementById('single-message').value;

            if (!number || !message) {
                showResult('single-result', 'Please enter both number and message', 'error');
                return;
            }

            try {
                const response = await axios.post('/api/send-message', { number, message });
                showResult('single-result', 
                    response.data.success 
                        ? `✓ Message sent to ${response.data.to}`
                        : `✗ Failed: ${response.data.error}`, 
                    response.data.success ? 'success' : 'error'
                );
            } catch (error) {
                showResult('single-result', `Error: ${error.message}`, 'error');
            }
        }

        async function runCampaign() {
            if (!isClientReady) {
                showResult('campaign-result', 'Client not ready. Please authenticate first.', 'error');
                return;
            }

            const numbersText = document.getElementById('campaign-numbers').value;
            const message = document.getElementById('campaign-message').value;
            const delay = parseInt(document.getElementById('campaign-delay').value);

            if (!numbersText || !message) {
                showResult('campaign-result', 'Please enter numbers and message', 'error');
                return;
            }

            const numbers = numbersText.split('\n').filter(n => n.trim());
            
            try {
                const response = await axios.post('/api/run-campaign', {
                    numbers,
                    message,
                    options: { delayBetweenMessages: delay }
                });

                const result = response.data.campaignResults;
                showResult('campaign-result', 
                    `Campaign completed: ${result.successful} successful, ${result.failed} failed`,
                    result.failed === 0 ? 'success' : 'error'
                );

                // Show detailed results
                const progressDiv = document.getElementById('campaign-progress');
                progressDiv.innerHTML = `
                    <h4>Detailed Results:</h4>
                    <ul>
                        ${result.details.map(d => 
                            `<li>${d.number}: ${d.success ? '✓' : '✗'} ${d.error || ''}</li>`
                        ).join('')}
                    </ul>
                `;
            } catch (error) {
                showResult('campaign-result', `Error: ${error.message}`, 'error');
            }
        }

        async function validateNumbers() {
            const numbersText = document.getElementById('validate-numbers').value;
            if (!numbersText.trim()) {
                showResult('validate-result', 'Please enter numbers to validate', 'error');
                return;
            }

            const numbers = numbersText.split('\n').filter(n => n.trim());
            
            try {
                const response = await axios.post('/api/validate-numbers', { numbers });
                const data = response.data;
                
                let html = `<strong>Validation Summary:</strong><br>`;
                html += `Total: ${data.summary.total} | Valid: ${data.summary.valid} | Invalid: ${data.summary.invalid}<br><br>`;
                
                html += `<strong>Details:</strong><br>`;
                data.details.forEach(d => {
                    html += `${d.original}: ${d.isValid ? '✓' : '✗'} `;
                    if (d.isValid) {
                        html += `→ ${d.correctedNumber} (${d.countryCode})`;
                    } else {
                        html += `Invalid format`;
                    }
                    html += `<br>`;
                });
                
                document.getElementById('validate-result').innerHTML = html;
            } catch (error) {
                showResult('validate-result', `Error: ${error.message}`, 'error');
            }
        }

        async function getChats() {
            try {

                            ${chat.participants} participants
                        </div>
                    </div>
                `).join('');
            } catch (error) {
                console.error('Error loading chats:', error);
            }
        }
```

- **Consider using a proxy** if sending large volumes of messages
