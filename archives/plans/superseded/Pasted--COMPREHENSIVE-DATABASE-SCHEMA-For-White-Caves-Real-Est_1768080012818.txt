## **1. OVERALL DATABASE ARCHITECTURE**

### **1.1 Multi-Database Strategy**
```yaml
Database Ecosystem:
├── **PostgreSQL (Primary)**: Core business data, transactions, user management
├── **MongoDB (Property Inventory)**: Flexible property schemas, media metadata
├── **Elasticsearch (Search)**: Property search, full-text indexing
├── **PostGIS (Spatial)**: Geographic queries, mapping, location intelligence
├── **TimescaleDB (Market Data)**: Time-series market analytics
├── **Neo4j (Relationships)**: Property networks, owner relationships
├── **Redis (Cache)**: Session management, real-time data
├── **S3/Glacier (Media)**: Property images, videos, documents
```

### **1.2 Database Cluster Topology**
```yaml
Primary Cluster (UAE North):
  ├── PostgreSQL Master (1): Write operations
  ├── PostgreSQL Replicas (3): Read operations
  ├── MongoDB Sharded Cluster (6 nodes): Property data
  ├── Elasticsearch Cluster (8 nodes): Search operations
  └── Redis Sentinel Cluster (3 nodes): Caching

Disaster Recovery Cluster (UAE Central):
  ├── PostgreSQL Standby (1): Hot standby
  ├── MongoDB Replica Set (3 nodes): Async replication
  └── Cross-region replication for critical data

Analytics Cluster:
  ├── TimescaleDB Hypertables: Market time-series
  ├── Neo4j Cluster: Relationship graphs
  └── Data Lake (S3): Raw property data
```

```sql
-- ================================

-- Users table with tier management
CREATE TABLE users (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    uae_pass_id VARCHAR(100) UNIQUE,
    email VARCHAR(255) UNIQUE NOT NULL,
    phone VARCHAR(20),
    first_name VARCHAR(100),
    last_name VARCHAR(100),

    tier_id INTEGER REFERENCES user_tiers(tier_id),
    company_id UUID REFERENCES companies(company_id),

    role VARCHAR(100),
    status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'inactive', 'suspended', 'pending')),
```

```sql
    time_bucket('1 day', time) AS bucket,
    community_id,
    property_type,
    metric_name,
    avg(metric_value) as avg_value,
```

### **8.2 Redis Lua Scripts**

```lua
-- Script 1: Increment property view with rate limiting
local function incrementPropertyView(property_id, user_id, ip)
  local view_key = "property:views:" .. property_id .. ":" .. os.date("%Y-%m-%d")
  local user_limit_key = "user:views:" .. user_id .. ":" .. os.date("%Y-%m-%d")
  local ip_limit_key = "ip:views:" .. ip .. ":" .. os.date("%Y-%m-%d")
  
  -- Check rate limits
  local user_views = redis.call("GET", user_limit_key) or 0
  local ip_views = redis.call("GET", ip_limit_key) or 0
  
  if tonumber(user_views) > 1000 then
    return {error = "User daily view limit exceeded"}
  end
  
  if tonumber(ip_views) > 5000 then
    return {error = "IP daily view limit exceeded"}
  end
  
  -- Increment counters
  redis.call("INCR", view_key)
  redis.call("INCR", user_limit_key)
  redis.call("INCR", ip_limit_key)
  
  -- Set TTL for daily counters
  redis.call("EXPIRE", view_key, 86400)
  redis.call("EXPIRE", user_limit_key, 86400)
  redis.call("EXPIRE", ip_limit_key, 86400)
  
  -- Update real-time counter
  redis.call("ZINCRBY", "property:popularity", 1, property_id)
  
  return {success = true, views = redis.call("GET", view_key)}
end

-- Script 2: Update property cache with invalidation
local function updatePropertyCache(property_id, property_data)
  local cache_key = "property:details:" .. property_id
  local similar_key = "property:similar:" .. property_id
  
  -- Update main cache
  redis.call("SET", cache_key, cjson.encode(property_data))
  redis.call("EXPIRE", cache_key, 3600)
  
  -- Invalidate similar properties cache
  redis.call("DEL", similar_key)
  
  -- Publish update notification
  redis.call("PUBLISH", "realtime:property:" .. property_id .. ":updates", 
             cjson.encode({
               action = "updated",
               property_id = property_id,
               timestamp = os.time()
             }))
  
  -- Update search index async flag
  redis.call("LPUSH", "queue:search:update", property_id)
  
  return {success = true}
end
```

### **11.1 Daily Maintenance**

```sql
-- Daily maintenance script
CREATE OR REPLACE FUNCTION perform_daily_maintenance()
RETURNS void AS $$
BEGIN
    -- Clean up expired sessions
    DELETE FROM core.user_sessions 
    WHERE expires_at < CURRENT_TIMESTAMP;
    
    -- Archive old activities (older than 90 days)
    INSERT INTO analytics.user_activities_archive
    SELECT * FROM core.user_activities 
    WHERE created_at < CURRENT_TIMESTAMP - INTERVAL '90 days';
    
    DELETE FROM core.user_activities 
    WHERE created_at < CURRENT_TIMESTAMP - INTERVAL '90 days';
    
    -- Update market trends
    PERFORM update_market_trends();
    
    -- Refresh materialized views
    REFRESH MATERIALIZED VIEW CONCURRENTLY analytics.market_daily_trends;
    REFRESH MATERIALIZED VIEW CONCURRENTLY analytics.popular_properties;
    
    -- Vacuum analyze for performance
    VACUUM ANALYZE;
END;
$$ LANGUAGE plpgsql;

-- Schedule daily maintenance
SELECT cron.schedule('daily-maintenance', '0 2 * * *', 
                    'SELECT perform_daily_maintenance();');
```

### **11.2 Data Integrity Checks**

```sql
-- Data integrity validation function
CREATE OR REPLACE FUNCTION validate_data_integrity()
RETURNS TABLE (check_name TEXT, status TEXT, details TEXT) AS $$
BEGIN
    RETURN QUERY
    
    -- Check 1: Orphaned property listings
    SELECT 'Orphaned Listings'::TEXT,
           CASE WHEN COUNT(*) = 0 THEN 'PASS' ELSE 'FAIL' END,
           COUNT(*) || ' orphaned listings found'::TEXT
    FROM property.property_listings pl
    LEFT JOIN property.properties p ON pl.property_id = p.property_id
    WHERE p.property_id IS NULL;
    
    -- Check 2: Invalid property prices
    SELECT 'Invalid Property Prices'::TEXT,
           CASE WHEN COUNT(*) = 0 THEN 'PASS' ELSE 'FAIL' END,
           COUNT(*) || ' properties with invalid prices'::TEXT
    FROM property.properties
    WHERE (pricing->'sale'->>'price')::DECIMAL < 0
       OR (pricing->'rent'->>'monthly_price')::DECIMAL < 0;
    
    -- Check 3: Duplicate property references
    SELECT 'Duplicate References'::TEXT,
           CASE WHEN COUNT(*) = 0 THEN 'PASS' ELSE 'FAIL' END,
           COUNT(*) || ' duplicate reference numbers'::TEXT
    FROM (
        SELECT reference_number, COUNT(*)
        FROM property.properties
        GROUP BY reference_number
        HAVING COUNT(*) > 1
    ) dupes;
    
    -- Check 4: Expired listings not updated
    SELECT 'Expired Listings'::TEXT,
           CASE WHEN COUNT(*) = 0 THEN 'PASS' ELSE 'FAIL' END,
           COUNT(*) || ' expired listings still active'::TEXT
    FROM property.property_listings
    WHERE listing_status = 'active'
      AND expiry_date < CURRENT_DATE;
END;
$$ LANGUAGE plpgsql;
```

## **12. BACKUP AND RECOVERY SCRIPTS**

### **12.1 Database Backup Script**

```bash
#!/bin/bash
# backup_database.sh

# Configuration
BACKUP_DIR="/backups/database"
DATE=$(date +%Y%m%d_%H%M%S)
RETENTION_DAYS=30

# PostgreSQL backup
echo "Starting PostgreSQL backup..."
pg_dump -h localhost -U whitecaves_admin -Fc whitecaves > \
    "${BACKUP_DIR}/postgresql_whitecaves_${DATE}.dump"

# MongoDB backup
echo "Starting MongoDB backup..."
mongodump --host localhost --db whitecaves_properties \
    --out "${BACKUP_DIR}/mongodb_${DATE}"

# Elasticsearch backup
echo "Starting Elasticsearch backup..."
curl -X PUT "localhost:9200/_snapshot/whitecaves_backup/snapshot_${DATE}?wait_for_completion=true"

# Redis backup
echo "Starting Redis backup..."
redis-cli SAVE
cp /var/lib/redis/dump.rdb "${BACKUP_DIR}/redis_${DATE}.rdb"

# Compress backups
echo "Compressing backups..."
tar -czf "${BACKUP_DIR}/full_backup_${DATE}.tar.gz" \
    "${BACKUP_DIR}/postgresql_whitecaves_${DATE}.dump" \
    "${BACKUP_DIR}/mongodb_${DATE}" \
    "${BACKUP_DIR}/redis_${DATE}.rdb"

# Upload to S3
echo "Uploading to S3..."
aws s3 cp "${BACKUP_DIR}/full_backup_${DATE}.tar.gz" \
    s3://whitecaves-backups/database/

# Cleanup old backups
echo "Cleaning up old backups..."
find "${BACKUP_DIR}" -name "*.tar.gz" -mtime +${RETENTION_DAYS} -delete
find "${BACKUP_DIR}" -name "*.dump" -mtime +7 -delete
find "${BACKUP_DIR}" -name "mongodb_*" -mtime +7 -exec rm -rf {} \;
find "${BACKUP_DIR}" -name "redis_*.rdb" -mtime +7 -delete

echo "Backup completed successfully!"
```

### **12.2 Database Recovery Script**

```bash
#!/bin/bash
# restore_database.sh

if [ -z "$1" ]; then
    echo "Usage: $0 <backup_timestamp>"
    exit 1
fi

BACKUP_DATE=$1
BACKUP_FILE="/backups/database/full_backup_${BACKUP_DATE}.tar.gz"

if [ ! -f "$BACKUP_FILE" ]; then
    echo "Backup file not found: $BACKUP_FILE"
    exit 1
fi

echo "Starting database restore from backup: $BACKUP_DATE"

# Extract backup
echo "Extracting backup..."
tar -xzf "$BACKUP_FILE" -C /tmp/

# Stop services
echo "Stopping services..."
systemctl stop whitecaves-app
systemctl stop mongod
systemctl stop elasticsearch
systemctl stop redis

# Restore PostgreSQL
echo "Restoring PostgreSQL..."
pg_restore -h localhost -U whitecaves_admin -d whitecaves \
    --clean --if-exists \
    "/tmp/backups/database/postgresql_whitecaves_${BACKUP_DATE}.dump"

# Restore MongoDB
echo "Restoring MongoDB..."
mongorestore --host localhost \
    "/tmp/backups/database/mongodb_${BACKUP_DATE}"

# Restore Redis
echo "Restoring Redis..."
cp "/tmp/backups/database/redis_${BACKUP_DATE}.rdb" /var/lib/redis/dump.rdb
chown redis:redis /var/lib/redis/dump.rdb

# Restore Elasticsearch
echo "Restoring Elasticsearch..."
curl -X POST "localhost:9200/_snapshot/whitecaves_backup/snapshot_${BACKUP_DATE}/_restore?wait_for_completion=true"

# Start services
echo "Starting services..."
systemctl start redis
systemctl start elasticsearch
systemctl start mongod
systemctl start whitecaves-app

# Cleanup
echo "Cleaning up temporary files..."
rm -rf /tmp/backups

echo "Database restore completed successfully!"
```
