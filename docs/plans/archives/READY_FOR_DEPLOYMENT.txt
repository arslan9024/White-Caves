OPTION 1: VERCEL (RECOMMENDED - Zero-Downtime)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Command:
  make deploy-vercel

Or:
  npm install -g vercel
  vercel --prod

Time: ~5 minutes
Features:
  • Zero-downtime deployment
  • Automatic SSL/TLS
  • Built-in CDN
  • Automatic rollback
  • Analytics included
  • Preview deployments

OPTION 2: DOCKER (Container-Based)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Command:
  make deploy-docker

Or:
  docker build -t white-caves:latest .
  docker run -p 3000:3000 white-caves:latest

Time: ~10 minutes
Features:
  • Consistent environment
  • Easy scaling
  • Version management
  • Rollback capability

OPTION 3: TRADITIONAL (npm start)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Command:
  make deploy-staging

Or:
  npm install --production
  npm run build
  pm2 start server/index.js --name "white-caves"
