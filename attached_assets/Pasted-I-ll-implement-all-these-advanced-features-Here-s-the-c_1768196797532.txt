const { Client, LocalAuth } = require('whatsapp-web.js');
const crypto = require('crypto');
const EventEmitter = require('events');

class WhatsAppAuthManager extends EventEmitter {
    constructor() {
        super();
        this.authCodes = new Map(); // Stores active auth codes
        this.linkedDevices = new Map(); // Stores linked devices
        this.authExpiry = 5 * 60 * 1000; // 5 minutes
        
        // Client configuration
        this.clientConfig = {
            authStrategy: new LocalAuth({
                dataPath: '.wwebjs_auth',
                clientId: `client-${Date.now()}`
            }),
            puppeteer: {
                headless: process.env.NODE_ENV === 'production' ? 'new' : false,
                args: [
                    '--no-sandbox',
                    '--disable-setuid-sandbox',
                    '--disable-dev-shm-usage',
                    '--disable-accelerated-2d-canvas',
                    '--disable-web-security',
                    '--disable-features=IsolateOrigins,site-per-process',
                    '--disable-blink-features=AutomationControlled'
                ]
            },
            webVersionCache: {
                type: 'remote',
                remotePath: 'https://raw.githubusercontent.com/wppconnect-team/wa-version/main/html/2.2412.54.html'
            }
        };
    }
    
    /**
     * Generate a 6-digit code for device linking
     */
    generateAuthCode() {
        const code = Math.floor(100000 + Math.random() * 900000).toString();
        const sessionData = {
            code,
            timestamp: Date.now(),
            clientId: `client-${crypto.randomBytes(8).toString('hex')}`,
            status: 'pending',
            attempts: 0
        };
        
        this.authCodes.set(code, sessionData);
        
        // Auto-expire after 5 minutes
        setTimeout(() => {
            if (this.authCodes.get(code)?.status === 'pending') {
                this.authCodes.delete(code);
                console.log(`Auth code ${code} expired`);
            }
        }, this.authExpiry);
        
        console.log(`📱 Generated auth code: ${code} (valid for 5 minutes)`);
        return { code, expiresIn: this.authExpiry };
    }
    
    /**
     * Verify auth code and link device
     */
    async verifyAndLinkDevice(code) {
        const sessionData = this.authCodes.get(code);
        
        if (!sessionData) {
            return { success: false, error: 'Invalid or expired code' };
        }
        
        if (sessionData.status !== 'pending') {
            return { success: false, error: 'Code already used' };
        }
        
        sessionData.attempts++;
        
        if (sessionData.attempts > 3) {
            this.authCodes.delete(code);
            return { success: false, error: 'Too many attempts' };
        }
        
        try {
            // Create new client with unique clientId
            const client = new Client({
                ...this.clientConfig,
                authStrategy: new LocalAuth({
                    dataPath: '.wwebjs_auth',
                    clientId: sessionData.clientId
                })
            });
            
            // Set up event handlers
            return await this.initializeClientWithCode(client, code, sessionData);
            
        } catch (error) {
            return { success: false, error: error.message };
        }
    }
    
    async initializeClientWithCode(client, code, sessionData) {
        return new Promise((resolve) => {
            let qrTimer;
            let authTimer;
            
            client.on('qr', (qr) => {
                console.log('QR received (fallback)');
                // Clear timers if QR is shown
                clearTimeout(qrTimer);
                clearTimeout(authTimer);
                
                sessionData.status = 'qr_shown';
                this.authCodes.set(code, sessionData);
                
                resolve({
                    success: false,
                    requiresQR: true,
                    qrCode: qr,
                    message: 'Authentication requires QR code scanning'
                });
            });
            
            client.on('authenticated', () => {
                console.log(`✅ Device linked successfully with code: ${code}`);
                
                sessionData.status = 'authenticated';
                sessionData.authenticatedAt = Date.now();
                this.authCodes.set(code, sessionData);
                
                // Store linked device
                this.linkedDevices.set(sessionData.clientId, {
                    client,
                    code,
                    linkedAt: new Date().toISOString(),
                    lastActivity: Date.now()
                });
                
                resolve({
                    success: true,
                    clientId: sessionData.clientId,
                    message: 'Device linked successfully',
                    timestamp: new Date().toISOString()
                });
            });
            
            client.on('ready', () => {
                console.log(`Client ${sessionData.clientId} is ready`);
                this.emit('device-ready', sessionData.clientId);
            });
            
            client.on('auth_failure', (error) => {
                resolve({
                    success: false,
                    error: `Authentication failed: ${error}`
                });
            });
            
            // Timeout handling
            qrTimer = setTimeout(() => {
                if (sessionData.status === 'pending') {
                    resolve({
                        success: false,
                        error: 'Authentication timed out (no QR received)'
                    });
                    client.destroy();
                }
            }, 30000);
            
            authTimer = setTimeout(() => {
                if (sessionData.status !== 'authenticated') {
                    resolve({
                        success: false,
                        error: 'Authentication process timed out'
                    });
                    client.destroy();
                }
            }, 120000);
            
            // Initialize client
            client.initialize();
        });
    }
    
    /**
     * Get all linked devices
     */
    getLinkedDevices() {
        const devices = [];
        
        for (const [clientId, device] of this.linkedDevices) {
            devices.push({
                clientId,
                code: device.code,
                linkedAt: device.linkedAt,
                lastActivity: new Date(device.lastActivity),
                status: device.client?.info ? 'active' : 'inactive'
            });
        }
        
        return devices;
    }
    
    /**
     * Get client by clientId
     */
    getClient(clientId) {
        const device = this.linkedDevices.get(clientId);
        return device?.client || null;
    }
    
    /**
     * Switch between multiple linked devices
     */
    switchDevice(clientId) {
        const device = this.linkedDevices.get(clientId);
        if (!device || !device.client) {
            throw new Error(`Device ${clientId} not found or inactive`);
        }
        
        // Update last activity
        device.lastActivity = Date.now();
        this.linkedDevices.set(clientId, device);
        
        return device.client;
    }
    
    /**
     * Unlink device
     */
    async unlinkDevice(clientId) {
        const device = this.linkedDevices.get(clientId);
        if (device) {
            if (device.client) {
                await device.client.destroy();
            }
            this.linkedDevices.delete(clientId);
            
            // Also remove from auth codes
            for (const [code, data] of this.authCodes) {
                if (data.clientId === clientId) {
                    this.authCodes.delete(code);
                }
            }
            
            console.log(`Device ${clientId} unlinked successfully`);
            return true;
        }
        return false;
    }
    
    /**
     * Clean up expired codes and inactive devices
     */
    cleanup() {
        const now = Date.now();
        let cleaned = 0;
        
        // Clean expired auth codes
        for (const [code, data] of this.authCodes) {
            if (now - data.timestamp > this.authExpiry) {
                this.authCodes.delete(code);
                cleaned++;
            }
        }
        
        // Clean inactive devices (more than 24 hours)
        for (const [clientId, device] of this.linkedDevices) {
            if (now - device.lastActivity > 24 * 60 * 60 * 1000) {
                this.unlinkDevice(clientId);
                cleaned++;
            }
        }
        
        return cleaned;
    }
    
    /**
     * REST API endpoints for authentication
     */
    setupAuthAPI(app) {
        // Generate new auth code
        app.post('/api/auth/generate-code', (req, res) => {
            const result = this.generateAuthCode();
            res.json(result);
        });
        
        // Verify auth code
        app.post('/api/auth/verify-code', async (req, res) => {
            const { code } = req.body;
            if (!code) {
                return res.status(400).json({ error: 'Code required' });
            }
            
            const result = await this.verifyAndLinkDevice(code);
            res.json(result);
        });
        
        // List linked devices
        app.get('/api/auth/devices', (req, res) => {
            const devices = this.getLinkedDevices();
            res.json({ devices });
        });
        
        // Switch active device
        app.post('/api/auth/switch-device', (req, res) => {
            const { clientId } = req.body;
            try {
                const client = this.switchDevice(clientId);
                res.json({
                    success: true,
                    message: `Switched to device ${clientId}`,
                    clientId
                });
            } catch (error) {
                res.status(400).json({ error: error.message });
            }
        });
        
        // Unlink device
        app.delete('/api/auth/devices/:clientId', async (req, res) => {
            const { clientId } = req.params;
            const success = await this.unlinkDevice(clientId);
            res.json({ success });
        });
        
        // Get auth status
        app.get('/api/auth/status', (req, res) => {
            res.json({
                activeCodes: this.authCodes.size,
                linkedDevices: this.linkedDevices.size,
                lastCleanup: new Date().toISOString()
            });
        });
    }
}

module.exports = WhatsAppAuthManager;

class SheetRotationManager {

            console.error('Failed to load sheet configs:', error);
            // Fallback to single sheet

                range: process.env.GOOGLE_SHEETS_RANGE,

                weight: 1,
                lastUsed: null,
                useCount: 0
            }];
        }
    }
    
    /**
     * Get next sheet based on rotation strategy
     */
    getNextSheet(campaignName = '') {
        if (this.config.sheetConfigs.length === 0) {
            throw new Error('No sheet configurations available');
        }
        
        let selectedSheet;
        
        switch (this.config.rotationMode) {
            case 'sequential':
                selectedSheet = this.getSequentialSheet();
                break;
                
            case 'random':
                selectedSheet = this.getRandomSheet();
                break;
                
            case 'weighted':
                selectedSheet = this.getWeightedSheet();
                break;
                
            case 'round_robin':
                selectedSheet = this.getRoundRobinSheet();
                break;
                
            default:
                selectedSheet = this.getSequentialSheet();
        }
        
        // Update sheet usage statistics
        this.updateSheetUsage(selectedSheet, campaignName);
        
        return selectedSheet;
    }
    
    /**
     * Sequential rotation (one after another)
     */
    getSequentialSheet() {
        const sheet = { ...this.config.sheetConfigs[this.config.currentIndex] };
        this.config.currentIndex = (this.config.currentIndex + 1) % this.config.sheetConfigs.length;
        return sheet;
    }
    
    /**
     * Random selection
     */
    getRandomSheet() {
        const randomIndex = Math.floor(Math.random() * this.config.sheetConfigs.length);
        return { ...this.config.sheetConfigs[randomIndex] };
    }
    
    /**
     * Weighted random selection
     */
    getWeightedSheet() {
        const totalWeight = this.config.sheetConfigs.reduce((sum, sheet) => sum + (sheet.weight || 1), 0);
        const random = Math.random() * totalWeight;
        
        let cumulative = 0;
        for (const sheet of this.config.sheetConfigs) {
            cumulative += sheet.weight || 1;
            if (random <= cumulative) {
                return { ...sheet };
            }
        }
        
        // Fallback
        return { ...this.config.sheetConfigs[0] };
    }
    
    /**
     * Round robin with usage tracking
     */
    getRoundRobinSheet() {
        // Sort by last used timestamp (oldest first)
        const sortedSheets = [...this.config.sheetConfigs].sort((a, b) => {
            const timeA = a.lastUsed ? new Date(a.lastUsed).getTime() : 0;
            const timeB = b.lastUsed ? new Date(b.lastUsed).getTime() : 0;
            return timeA - timeB;
        });
        
        return { ...sortedSheets[0] };
    }
    
    /**
     * Update sheet usage statistics
     */
    updateSheetUsage(sheet, campaignName) {
        const now = new Date().toISOString();
        
        // Find and update the sheet in configs
        const sheetIndex = this.config.sheetConfigs.findIndex(s => 
            s.spreadsheetId === sheet.spreadsheetId && s.range === sheet.range
        );
        
        if (sheetIndex !== -1) {
            this.config.sheetConfigs[sheetIndex].lastUsed = now;
            this.config.sheetConfigs[sheetIndex].useCount = (this.config.sheetConfigs[sheetIndex].useCount || 0) + 1;
            this.config.sheetConfigs[sheetIndex].lastCampaign = campaignName;
        }
        
        // Add to rotation history
        this.config.rotationHistory.unshift({
            timestamp: now,
            sheetId: sheet.id || sheet.spreadsheetId,
            sheetName: sheet.name,
            campaignName,
            mode: this.config.rotationMode
        });
        
        // Trim history
        if (this.config.rotationHistory.length > this.config.maxHistorySize) {
            this.config.rotationHistory.pop();
        }
        
        console.log(`📄 Selected sheet: ${sheet.name} (${sheet.spreadsheetId})`);
    }
    
    /**
     * Get sheet by ID
     */
    getSheetById(sheetId) {
        return this.config.sheetConfigs.find(sheet =>

        );
    }

     * Manually force rotation to specific sheet

    forceSheetSelection(sheetId) {

        if (!sheet) {
            throw new Error(`Sheet ${sheetId} not found`);
        }
        
        // Update current index if in sequential mode
        if (this.config.rotationMode === 'sequential') {
            const currentIndex = this.config.sheetConfigs.findIndex(s => 
                s.spreadsheetId === sheet.spreadsheetId

            if (currentIndex !== -1) {
                this.config.currentIndex = currentIndex;
            }
        }
        
        return { ...sheet };
    }
    
    /**

    getRotationStats() {
        const now = Date.now();
        const oneWeekAgo = now - (7 * 24 * 60 * 60 * 1000);
        
        // Calculate usage in last 7 days
        const recentUsage = this.config.rotationHistory.filter(entry =>

            rotationMode: this.config.rotationMode,

            usageLast7Days: recentUsage.length,

class SelectionRestrictionManager {
    constructor(googleSheetsService) {

        this.restrictionConfig = {
            daysBetweenSelections: 7, // Don't select same number within 7 days
            maxDailySelections: 50, // Maximum unique selections per day

    /**
     * Check if number can be selected based on 7-day rule
     */
    async canSelectNumber(phoneNumber, campaignName = '') {
        const cacheKey = this.normalizePhoneNumber(phoneNumber);
        
        // Check in-memory cache first
        const cached = this.restrictionConfig.selectionHistory.get(cacheKey);
        if (cached) {
            const daysSince = moment().diff(moment(cached.lastSelected), 'days');
            
            if (daysSince < this.restrictionConfig.daysBetweenSelections) {
                return {
                    allowed: false,
                    reason: `Selected ${daysSince} days ago (min ${this.restrictionConfig.daysBetweenSelections} days required)`,
                    lastSelected: cached.lastSelected,
                    lastCampaign: cached.lastCampaign,
                    nextAvailable: moment(cached.lastSelected)
                        .add(this.restrictionConfig.daysBetweenSelections, 'days')
                        .format('YYYY-MM-DD')
                };
            }
        }
        
        // Check Google Sheets history
        const sheetHistory = await this.checkSheetHistory(phoneNumber);
        if (sheetHistory) {
            const daysSince = moment().diff(moment(sheetHistory.lastSelected), 'days');
            
            if (daysSince < this.restrictionConfig.daysBetweenSelections) {
                // Update cache
                this.restrictionConfig.selectionHistory.set(cacheKey, {
                    lastSelected: sheetHistory.lastSelected,
                    lastCampaign: sheetHistory.lastCampaign,
                    checkedAt: new Date().toISOString()
                });
                
                return {
                    allowed: false,
                    reason: `Selected ${daysSince} days ago in Google Sheets`,
                    lastSelected: sheetHistory.lastSelected,
                    lastCampaign: sheetHistory.lastCampaign,
                    nextAvailable: moment(sheetHistory.lastSelected)
                        .add(this.restrictionConfig.daysBetweenSelections, 'days')
                        .format('YYYY-MM-DD')
                };
            }
        }
        
        // Check if we've exceeded daily selection limit
        const dailyCount = await this.getTodaySelectionCount();
        if (dailyCount >= this.restrictionConfig.maxDailySelections) {
            return {
                allowed: false,
                reason: `Daily selection limit reached (${this.restrictionConfig.maxDailySelections})`,
                dailyCount,
                nextAvailable: 'Tomorrow'
            };
        }
        
        return {
            allowed: true,
            lastSelected: cached?.lastSelected || sheetHistory?.lastSelected,
            daysSince: cached ? moment().diff(moment(cached.lastSelected), 'days') : null
        };
    }
    
    /**

            const data = await this.sheetsService.getSheetData(

            if (!data || !data.values) return null;
            
            // Find the number in the history
            for (const row of data.values) {
                const [timestamp, number, campaign, action] = row;
                if (this.normalizePhoneNumber(number) === normalizedNumber && action === 'selected') {
                    return {
                        lastSelected: timestamp,

            }

    /**
     * Check for failed messages and avoid retrying
     */
    async checkFailedMessages(phoneNumber) {
        const cacheKey = this.normalizePhoneNumber(phoneNumber);
        
        // Check in-memory cache
        const cachedFailure = this.restrictionConfig.failedMessageCache.get(cacheKey);
        if (cachedFailure) {
            const hoursSince = moment().diff(moment(cachedFailure.timestamp), 'hours');
            
            // Don't retry for 72 hours if it was a block/opt-out error
            if (cachedFailure.isPermanent && hoursSince < 72) {
                return {
                    shouldRetry: false,
                    reason: `Permanent failure detected ${hoursSince} hours ago`,
                    error: cachedFailure.error,
                    lastAttempt: cachedFailure.timestamp,
                    nextRetry: moment(cachedFailure.timestamp).add(72, 'hours').format()
                };
            }
            
            // Don't retry for 24 hours for other errors
            if (!cachedFailure.isPermanent && hoursSince < 24) {
                return {
                    shouldRetry: false,
                    reason: `Temporary failure detected ${hoursSince} hours ago`,
                    error: cachedFailure.error,
                    lastAttempt: cachedFailure.timestamp,
                    nextRetry: moment(cachedFailure.timestamp).add(24, 'hours').format()
                };
            }
        }
        
        // Check Google Sheets failure logs
        const sheetFailure = await this.checkSheetFailures(phoneNumber);
        if (sheetFailure) {
            const hoursSince = moment().diff(moment(sheetFailure.timestamp), 'hours');
            
            if (sheetFailure.isPermanent && hoursSince < 72) {
                return {
                    shouldRetry: false,
                    reason: `Sheet shows permanent failure ${hoursSince} hours ago`,
                    error: sheetFailure.error,
                    campaign: sheetFailure.campaign,
                    nextRetry: moment(sheetFailure.timestamp).add(72, 'hours').format()
                };
            }
            
            if (!sheetFailure.isPermanent && hoursSince < 24) {
                return {
                    shouldRetry: false,
                    reason: `Sheet shows temporary failure ${hoursSince} hours ago`,
                    error: sheetFailure.error,
                    campaign: sheetFailure.campaign,
                    nextRetry: moment(sheetFailure.timestamp).add(24, 'hours').format()
                };
            }
        }
        
        return {
            shouldRetry: true,
            lastFailure: cachedFailure || sheetFailure
        };
    }
    
    /**
     * Record failed message
     */
    async recordFailedMessage(phoneNumber, error, campaignName, isPermanent = false) {
        const cacheKey = this.normalizePhoneNumber(phoneNumber);
        const now = new Date().toISOString();
        
        // Update cache
        this.restrictionConfig.failedMessageCache.set(cacheKey, {
            timestamp: now,
            error: typeof error === 'string' ? error : error.message,
            campaign: campaignName,
            isPermanent,
            retryCount: (this.restrictionConfig.failedMessageCache.get(cacheKey)?.retryCount || 0) + 1
        });

    weightedRandomSelection(weightedItems, n) {
        const selected = [];
        const items = [...weightedItems];
        
        while (selected.length < n && items.length > 0) {
            const totalWeight = items.reduce((sum, item) => sum + item.weight, 0);
            const random = Math.random() * totalWeight;
            let cumulative = 0;
            
            for (let i = 0; i < items.length; i++) {
                cumulative += items[i].weight;
                if (random <= cumulative) {
                    selected.push(items[i].contact);
                    items.splice(i, 1);
                    break;
                }
            }
        }
        
        return selected;
    }

        console.log('🔄 Daily stats reset');
    }
    
    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
    
    /**
     * API endpoints for management
     */
    setupManagementAPI(app) {
        // Campaign management
        app.post('/api/campaigns/start', async (req, res) => {
            try {
                const result = await this.orchestrateCampaign(req.body);
                res.json(result);
            } catch (error) {
                res.status(500).json({ error: error.message });
            }
        });
        
        // Get campaign history
        app.get('/api/campaigns/history', (req, res) => {
            res.json({
                total: this.state.campaignHistory.length,
                campaigns: this.state.campaignHistory.slice(0, 20),
                dailyStats: this.state.dailyStats
            });
        });
        
        // Get restriction statistics
        app.get('/api/restrictions/stats', (req, res) => {
            res.json(this.restrictionManager.getRestrictionStats());
        });
        
        // Get sheet rotation statistics
        app.get('/api/sheets/stats', (req, res) => {
            res.json(this.sheetManager.getRotationStats());
        });
        
        // Get authentication status
        app.get('/api/auth/status', (req, res) => {
            const authStatus = this.checkAuthentication();
            res.json(authStatus);
        });
        
        // Generate auth code
        app.post('/api/auth/generate-code', (req, res) => {
            const result = this.authManager.generateAuthCode();
            res.json(result);
        });
        
        // Switch active sheet
        app.post('/api/sheets/switch', (req, res) => {
            const { sheetId } = req.body;
            try {
                const sheet = this.sheetManager.forceSheetSelection(sheetId);
                res.json({
                    success: true,
                    sheet,
                    message: `Switched to sheet: ${sheet.name}`
                });
            } catch (error) {
                res.status(400).json({ error: error.message });
            }
        });
    }
}

module.exports = EnhancedCampaignOrchestrator;

## ⚠️ **Critical Implementation Notes**

1. **Code-Based Authentication**:
   - Uses 6-digit codes instead of QR
   - Each code valid for 5 minutes
   - Supports multiple linked devices
   - Automatic cleanup of expired sessions

2. **Dynamic Sheet Rotation**:
   - Multiple rotation strategies (sequential, random, weighted, round-robin)
   - Tracks sheet usage history
   - Supports adding sheets dynamically
   - UAE-focused prioritization

3. **7-Day Selection Restriction**:
   - Tracks when each number was last selected
   - Prevents re-selection within 7 days
   - Integrates with Google Sheets history
   - Daily selection limits

4. **Failed Message Tracking**:
   - Distinguishes between temporary and permanent failures
   - Automatic retry avoidance
   - 72-hour block for permanent failures
   - 24-hour block for temporary failures

5. **UAE Focus**:
   - Prioritizes +971 numbers
   - UAE-specific scheduling
   - Configurable UAE contact quota
   - Local timezone consideration

This system provides enterprise-level campaign management while attempting to minimize detection through sophisticated behavior patterns and strict scheduling. However, **remember that using unofficial WhatsApp APIs remains high-risk and violates their Terms of Service**.
