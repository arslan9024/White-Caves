### **2.3 UAE PASS Digital Signature Integration**

#### **Signature Flow Implementation**
```javascript
// UAE PASS Authentication and Signing Flow
class UAEPassSigning {
    async initiateSigning(contractId, partyType) {
        // Step 1: Generate signing request
        const signingRequest = {
            contract_id: contractId,
            party: partyType, // 'landlord' or 'tenant'
            timestamp: new Date().toISOString(),
            signing_token: this.generateToken(),
            callback_url: `${process.env.APP_URL}/signing/callback`
        };
        
        // Step 2: Redirect to UAE PASS
        const uaePassUrl = this.buildUAEPassUrl(signingRequest);
        return { redirect_url: uaePassUrl };
    }
    
    async handleCallback(authCode, signingToken) {
        // Step 3: Exchange auth code for user info
        const userInfo = await this.exchangeCodeForUserInfo(authCode);
        
        // Step 4: Verify identity matches contract party
        const isValid = await this.verifyIdentity(userInfo, signingToken);
        
        if (isValid) {
            // Step 5: Generate digital signature
            const signature = await this.generateDigitalSignature(
                userInfo,
                signingToken
            );
            
            // Step 6: Record signature in blockchain
            await this.recordSignatureInBlockchain(signature);
            
            return {
                success: true,
                signature_id: signature.id,
                signed_at: signature.timestamp,
                user_verified: userInfo
            };
        }
        
        return { success: false, error: "Identity verification failed" };
    }
}
```

### **2.4 Automated Ejari Registration**

#### **Ejari Integration Flow**
```python
class EjariAutomation:
    def __init__(self):
        self.dld_api_base = "https://api.dld.gov.ae/v1"
        self.ejari_api = f"{self.dld_api_base}/ejari"
    
    async def register_contract(self, contract_data):
        """Submit contract to Ejari system"""
        
        # Prepare Ejari payload
        ejari_payload = {
            "contract": {
                "contract_number": contract_data['contract_id'],
                "property": self.format_property_data(contract_data['property']),
                "parties": self.format_parties_data(contract_data['parties']),
                "terms": self.format_terms_data(contract_data['terms']),
                "attachments": self.prepare_attachments(contract_data)
            }
        }
        
        try:
            # Submit to DLD API
            response = await self.call_dld_api(
                "POST", 
                f"{self.ejari_api}/register",
                ejari_payload
            )
            
            if response['success']:
                # Store Ejari registration details
                await self.store_ejari_details(
                    contract_data['contract_id'],
                    response['ejari_data']
                )
                
                # Trigger post-registration workflows
                await self.trigger_post_ejari_workflows(
                    contract_data['contract_id']
                )
                
                return {
                    'success': True,
                    'ejari_number': response['ejari_number'],
                    'registration_date': response['registration_date']
                }
            
        except Exception as e:
            # Fallback: Manual registration with notification
            await self.create_manual_ejari_task(contract_data)
            return {
                'success': False,
                'requires_manual': True,
                'task_created': True
            }
```

```python
class PostSigningWorkflow:
    def __init__(self, contract_id):
        self.contract_id = contract_id
        self.contract = self.load_contract(contract_id)
    
    async def execute_workflow(self):
        """Execute all post-signing automated actions"""
        
        workflows = [
            self.generate_signed_copy(),
            self.notify_all_parties(),
            self.create_ejari_registration(),
            self.setup_payment_schedule(),
            self.create_handover_checklist(),
            self.schedule_property_inspection(),
            self.setup_utilities_transfer(),
            self.create_welcome_package(),
            self.update_crm_status(),
            self.trigger_commission_calculation()
        ]
        
        # Execute in order with dependency tracking
        results = []
        for workflow in workflows:
            try:
                result = await workflow()
                results.append(result)
            except Exception as e:
                logger.error(f"Workflow failed: {workflow.__name__}", e)
                await self.create_manual_task(workflow.__name__, str(e))
        
        return results
    
    async def create_ejari_registration(self):
        """Automated Ejari submission"""
        ejari_service = EjariAutomation()
        return await ejari_service.register_contract(self.contract)
    
    async def setup_payment_schedule(self):
        """Create payment calendar entries"""
        payments = PaymentScheduler.create_schedule(
            self.contract['terms'],
            self.contract['parties']
        )
        
        # Create calendar events
        for payment in payments:
            await CalendarService.create_event(
                title=f"Rent Payment Due - {payment['amount']} AED",
                date=payment['due_date'],
                parties=[self.contract['parties']['tenant']],
                reminders=[7, 3, 1]  # days before
            )
        
        return payments
    
    async def create_handover_checklist(self):
        """Generate digital handover checklist"""
        checklist = HandoverChecklist.generate(
            self.contract['property'],
            self.contract['terms']
        )
        
        # Assign to property manager
        await TaskService.create_task({
            'title': 'Property Handover',
            'assigned_to': self.get_property_manager(),
            'due_date': self.contract['terms']['lease_start'],
            'checklist': checklist
        })
        
        return checklist

class SignatureVerification:
    def __init__(self):

    async def record_signature(self, signature_data):
        """Record signature hash on blockchain"""

        # Store on blockchain
        tx_hash = await self.blockchain.call_contract(
            method="recordSignature",
            args=[signature_hash, signature_data['timestamp']]
        )
        
        # Wait for confirmation
        receipt = await self.blockchain.wait_for_receipt(tx_hash)
        
        return {
            'tx_hash': tx_hash,
            'block_number': receipt.blockNumber,
            'timestamp': receipt.timestamp
        }
    
    async def verify_signature(self, signature_id):
        """Verify signature exists on blockchain"""
        
        # Query blockchain
        result = await self.blockchain.query_contract(
            method="verifySignature",
            args=[signature_id]
        )
        
        return {
            'exists': result['exists'],
            'block_number': result['recorded_at_block'],
            'timestamp': result['recorded_at_time']
        }
```

### **8.1 Agent Performance Dashboard**

```sql
-- Contract Preparation Metrics
CREATE VIEW agent_contract_metrics AS
SELECT 
    u.id as agent_id,
    u.full_name as agent_name,
    COUNT(DISTINCT c.id) as total_contracts,
    AVG(TIME_TO_SEC(c.signed_at - c.created_at) / 3600) as avg_preparation_hours,
    AVG(TIME_TO_SEC(c.fully_signed_at - c.sent_for_signature) / 86400) as avg_signing_days,
    SUM(CASE WHEN c.status = 'fully_signed' THEN 1 ELSE 0 END) as successful_contracts,
    SUM(CASE WHEN c.status = 'expired' THEN 1 ELSE 0 END) as expired_contracts,
    AVG(c.compliance_score) as avg_compliance_score
FROM contracts c
JOIN users u ON c.agent_id = u.id
WHERE c.created_at >= NOW() - INTERVAL '30 days'
GROUP BY u.id, u.full_name;
```
