// Action Scanner Engine
class ActionScanner {
  constructor() {
    this.actionTypes = [
      'onClick', 'onSubmit', 'onChange', 'onKeyDown', 'onKeyUp', 
      'onMouseEnter', 'onMouseLeave', 'onFocus', 'onBlur', 'onLoad',
      'onScroll', 'onResize', 'onDrag', 'onDrop', 'onCopy', 'onPaste',
      'onTouchStart', 'onTouchEnd', 'onDoubleClick', 'onContextMenu'
    ];
  }
  
  async scanProjectForActions() {
    console.log('🔍 Scanning project for actions and event handlers...');
    
    const components = [];
    const allFiles = await this.walkProjectDirectory();
    
    for (const filePath of allFiles) {
      if (this.isComponentFile(filePath)) {
        const component = await this.analyzeComponent(filePath);
        if (component.actions.length > 0) {
          components.push(component);
        }
      }
    }
    
    return {
      timestamp: new Date().toISOString(),
      totalComponents: components.length,
      totalActions: components.reduce((sum, comp) => sum + comp.actions.length, 0),
      components: components
    };
  }
  
  async analyzeComponent(filePath) {
    const sourceCode = await this.readFile(filePath);
    const componentInfo = this.extractComponentInfo(sourceCode, filePath);
    
    // Extract all event handlers
    const actions = this.extractActions(sourceCode, componentInfo);
    
    // Analyze action implementations
    const analyzedActions = await this.analyzeActionImplementations(actions, filePath);
    
    return {
      ...componentInfo,
      filePath,
      actions: analyzedActions,
      metrics: this.calculateActionMetrics(analyzedActions)
    };
  }
  
  extractActions(sourceCode, componentInfo) {
    const actions = [];
    
    // Pattern matching for event handlers
    this.actionTypes.forEach(actionType => {
      const pattern = new RegExp(`${actionType}\\s*=\\s*{([^}]+)}`, 'g');
      const matches = sourceCode.match(pattern) || [];
      
      matches.forEach((match, index) => {
        const handlerName = this.extractHandlerName(match, componentInfo.name, actionType, index);
        const handlerCode = this.extractHandlerCode(match, sourceCode);
        
        actions.push({
          id: `${componentInfo.name}_${actionType}_${index}`,
          name: handlerName,
          type: actionType,
          component: componentInfo.name,
          componentType: componentInfo.type,
          filePath: componentInfo.filePath,
          lineNumber: this.findLineNumber(sourceCode, match),
          code: handlerCode,
          dependencies: this.extractDependencies(handlerCode),
          status: this.determineActionStatus(handlerCode),
          hasMockData: this.hasMockData(handlerCode),
          hasApiCall: this.hasApiCall(handlerCode),
          hasStateChange: this.hasStateChange(handlerCode),
          hasSideEffects: this.hasSideEffects(handlerCode)
        });
      });
    });
    
    // Also look for custom event handlers
    const customHandlers = this.extractCustomHandlers(sourceCode, componentInfo);
    actions.push(...customHandlers);
    
    return actions;
  }
  
  extractHandlerCode(match, sourceCode) {
    // Extract the actual function body
    const handlerPattern = /=\s*{([^}]+(?:\{[^}]*\}[^}]*)*)}/;
    const handlerMatch = match.match(handlerPattern);
    
    if (handlerMatch) {
      return handlerMatch[1].trim();
    }
    
    // Look for arrow functions or function declarations
    const arrowFunctionPattern = /=\s*\([^)]*\)\s*=>\s*{([^}]+(?:\{[^}]*\}[^}]*)*)}/;
    const arrowMatch = match.match(arrowFunctionPattern);
    
    if (arrowMatch) {
      return arrowMatch[1].trim();
    }
    
    return match;
  }
  
  determineActionStatus(handlerCode) {
    const placeholderPatterns = [
      /\/\/\s*TODO/,
      /\/\/\s*FIXME/,
      /console\.log\(["'](TODO|FIXME|Implement|Placeholder)/,
      /\/\*\s*TODO/,
      /throw new Error\(["']Not implemented["']\)/,
      /return\s+null/,
      /return\s+false/,
      /\/\/\s*Mock data/,
      /\/\/\s*Remove this/
    ];
    
    const isPlaceholder = placeholderPatterns.some(pattern => pattern.test(handlerCode));
    
    if (isPlaceholder) return 'placeholder';
    
    const hasRealLogic = this.hasRealLogic(handlerCode);
    
    if (hasRealLogic) return 'implemented';
    
    return 'empty';
  }
  
  hasRealLogic(handlerCode) {
    const realLogicIndicators = [
      /setState\(/,
      /useState\(/,
      /dispatch\(/,
      /await\s+\w+\(/,
      /fetch\(/,
      /axios\./,
      /\.then\(/,
      /\.catch\(/,
      /if\s*\(/,
      /for\s*\(/,
      /while\s*\(/,
      /switch\s*\(/,
      /return\s+\w/,
      /window\.location/,
      /history\./,
      /navigate\(/
    ];
    
    return realLogicIndicators.some(pattern => pattern.test(handlerCode));
  }
  
  calculateActionMetrics(actions) {
    return {
      total: actions.length,
      implemented: actions.filter(a => a.status === 'implemented').length,
      placeholders: actions.filter(a => a.status === 'placeholder').length,
      empty: actions.filter(a => a.status === 'empty').length,
      withApiCalls: actions.filter(a => a.hasApiCall).length,
      withStateChanges: actions.filter(a => a.hasStateChange).length,
      withSideEffects: actions.filter(a => a.hasSideEffects).length
    };
  }
}

// Action Catalog Component
const ActionCatalog = ({ components, onSelectAction, selectedAction }) => {
  const [searchTerm, setSearchTerm] = useState('');
  const [filterType, setFilterType] = useState('all');
  const [filterStatus, setFilterStatus] = useState('all');
  const [sortBy, setSortBy] = useState('component');
  
  const filteredComponents = useMemo(() => {
    return components.map(comp => ({
      ...comp,
      actions: comp.actions.filter(action => {
        // Apply search filter
        if (searchTerm && !action.name.toLowerCase().includes(searchTerm.toLowerCase()) &&
            !action.component.toLowerCase().includes(searchTerm.toLowerCase())) {
          return false;
        }
        
        // Apply type filter
        if (filterType !== 'all' && action.type !== filterType) {
          return false;
        }
        
        // Apply status filter
        if (filterStatus !== 'all' && action.status !== filterStatus) {
          return false;
        }
        
        return true;
      })
    })).filter(comp => comp.actions.length > 0);
  }, [components, searchTerm, filterType, filterStatus]);
  
  const allActions = useMemo(() => {
    const actions = filteredComponents.flatMap(comp => comp.actions);
    
    // Sort actions
    return actions.sort((a, b) => {
      switch(sortBy) {
        case 'component':
          return a.component.localeCompare(b.component);
        case 'type':
          return a.type.localeCompare(b.type);
        case 'status':
          return a.status.localeCompare(b.status);
        case 'name':
          return a.name.localeCompare(b.name);
        default:
          return 0;
      }
    });
  }, [filteredComponents, sortBy]);
  
  return (
    <div className="action-catalog">
      {/* Catalog Header */}
      <div className="catalog-header">
        <div className="search-filter">
          <div className="search-box">
            <i className="icon-search"></i>
            <input
              type="text"
              placeholder="Search actions..."
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
            />
          </div>
          
          <div className="filter-controls">
            <select 
              className="filter-select"
              value={filterType}
              onChange={(e) => setFilterType(e.target.value)}
            >
              <option value="all">All Types</option>
              <option value="onClick">onClick</option>
              <option value="onSubmit">onSubmit</option>
              <option value="onChange">onChange</option>
              <option value="onKeyDown">onKeyDown</option>
              <option value="onMouseEnter">onMouseEnter</option>
              <option value="onFocus">onFocus</option>
              <option value="onLoad">onLoad</option>
            </select>
            
            <select 
              className="filter-select"
              value={filterStatus}
              onChange={(e) => setFilterStatus(e.target.value)}
            >
              <option value="all">All Status</option>
              <option value="implemented">Implemented</option>
              <option value="placeholder">Placeholder</option>
              <option value="empty">Empty</option>
            </select>
            
            <select 
              className="filter-select"
              value={sortBy}
              onChange={(e) => setSortBy(e.target.value)}
            >
              <option value="component">Sort by Component</option>
              <option value="type">Sort by Type</option>
              <option value="status">Sort by Status</option>
              <option value="name">Sort by Name</option>
            </select>
          </div>
        </div>
        
        <div className="catalog-stats">
          <div className="stat-badge">
            <span className="stat-value">{allActions.length}</span>
            <span className="stat-label">Actions</span>
          </div>
          <div className="stat-badge">
            <span className="stat-value">
              {allActions.filter(a => a.status === 'implemented').length}
            </span>
            <span className="stat-label">Implemented</span>
          </div>
          <div className="stat-badge">
            <span className="stat-value">
              {allActions.filter(a => a.hasApiCall).length}
            </span>
            <span className="stat-label">With API Calls</span>
          </div>
        </div>
      </div>
      
      {/* Components Grid */}
      <div className="components-grid">
        {filteredComponents.map(component => (
          <ComponentCard 
            key={component.name}
            component={component}
            onSelectAction={onSelectAction}

        <h3>All Actions ({allActions.length})</h3>
        <div className="actions-table">
          <table>
            <thead>
              <tr>
                <th>Component</th>
                <th>Action</th>
                <th>Type</th>
                <th>Status</th>
                <th>Features</th>
                <th>Actions</th>
              </tr>
            </thead>
            <tbody>

                <tr

// Action Executor with Sandbox
class ActionExecutor {
  constructor() {
    this.sandbox = this.createSandbox();
    this.mockData = this.loadMockData();

  }
  
  createSandbox() {
    // Create isolated execution environment
    const sandbox = {
      // Mock console
      console: {
        log: (...args) => this.logToConsole('log', args),
        error: (...args) => this.logToConsole('error', args),
        warn: (...args) => this.logToConsole('warn', args),
        info: (...args) => this.logToConsole('info', args)
      },
      
      // Mock state functions
      useState: (initialValue) => {
        const state = { current: initialValue };
        const setState = (newValue) => {
          state.current = typeof newValue === 'function' 
            ? newValue(state.current)
            : newValue;
          this.triggerStateChange(state);
          return state.current;
        };
        return [state.current, setState];
      },
      
      // Mock React hooks
      useEffect: (callback, deps) => {
        // Simplified mock - execute immediately
        try {
          callback();
        } catch (error) {
          console.error('useEffect error:', error);
        }
        return () => {}; // Cleanup function
      },
      
      useCallback: (callback, deps) => callback,
      
      useMemo: (factory, deps) => factory(),
      
      // Mock API calls
      api: this.createMockAPI(),
      
      // Mock navigation
      navigate: (path) => this.logNavigation(path),
      
      // Mock storage
      localStorage: this.createMockStorage(),
      sessionStorage: this.createMockStorage(),
      
      // Mock timers
      setTimeout: (fn, delay) => {
        const timerId = Date.now();
        setTimeout(() => {
          try {
            fn();
          } catch (error) {
            console.error('setTimeout error:', error);
          }
        }, delay);
        return timerId;
      },
      
      clearTimeout: (id) => clearTimeout(id),
      
      setInterval: (fn, interval) => {
        const intervalId = setInterval(fn, interval);
        return intervalId;
      },
      
      clearInterval: (id) => clearInterval(id)
    };
    
    // Proxy to catch undefined properties
    return new Proxy(sandbox, {
      get(target, prop) {
        if (prop in target) {
          return target[prop];
        }
        return () => {
          console.warn(`Attempted to access undefined property: ${prop}`);
          return undefined;
        };
      }
    });
  }
  
  createMockAPI() {
    // Mock API responses based on action type
    const mockResponses = {
      // Property-related endpoints
      '/api/v1/properties': {
        GET: { success: true, data: this.mockData.properties },
        POST: { success: true, data: { id: 'prop_' + Date.now() } }
      },
      
      // User-related endpoints
      '/api/v1/users': {
        GET: { success: true, data: this.mockData.users },
        POST: { success: true, data: { id: 'user_' + Date.now() } }
      },
      
      // CRM endpoints
      '/api/v1/crm/leads': {
        GET: { success: true, data: this.mockData.leads },
        POST: { success: true, data: { id: 'lead_' + Date.now() } }
      },
      
      // Default mock
      '*': {
        GET: { success: true, data: { message: 'Mock API response' } },
        POST: { success: true, data: { message: 'Created successfully' } },
        PUT: { success: true, data: { message: 'Updated successfully' } },
        DELETE: { success: true, data: { message: 'Deleted successfully' } }
      }
    };
    
    return {
      get: async (url) => {
        await this.simulateNetworkDelay();
        const endpoint = mockResponses[url] || mockResponses['*'];
        return { data: endpoint.GET };
      },
      
      post: async (url, data) => {
        await this.simulateNetworkDelay();
        const endpoint = mockResponses[url] || mockResponses['*'];
        return { data: { ...endpoint.POST, ...data } };
      },
      
      put: async (url, data) => {
        await this.simulateNetworkDelay();
        const endpoint = mockResponses[url] || mockResponses['*'];
        return { data: { ...endpoint.PUT, ...data } };
      },
      
      delete: async (url) => {
        await this.simulateNetworkDelay();
        const endpoint = mockResponses[url] || mockResponses['*'];
        return { data: endpoint.DELETE };
      },
      
      // Axios-style
      request: async (config) => {
        await this.simulateNetworkDelay();
        const endpoint = mockResponses[config.url] || mockResponses['*'];
        const method = config.method?.toUpperCase() || 'GET';
        return { data: endpoint[method] };
      }
    };
  }
  
  async executeInSandbox(action, inputData = {}) {
    const executionId = `exec_${Date.now()}`;
    const startTime = performance.now();
    
    try {
      // Prepare execution context
      const context = {
        ...this.sandbox,
        event: this.createMockEvent(action.type),
        ...inputData
      };
      
      // Wrap action code in a function
      const actionFunction = new Function(
        'context',
        `with(context) {
          try {
            return (${action.code})(context.event);
          } catch(error) {
            console.error('Action execution error:', error);
            throw error;
          }
        }`
      );
      
      // Execute
      const result = await actionFunction(context);
      const endTime = performance.now();
      const executionTime = endTime - startTime;
      
      // Record execution
      const executionRecord = {
        id: executionId,
        actionId: action.id,
        status: 'success',
        result,
        executionTime,
        timestamp: new Date().toISOString(),
        logs: this.getConsoleLogs()
      };
      
      this.executionHistory.push(executionRecord);
      return executionRecord;
      
    } catch (error) {
      const endTime = performance.now();
      
      const executionRecord = {
        id: executionId,
        actionId: action.id,
        status: 'error',
        error: error.message,
        executionTime: endTime - startTime,
        timestamp: new Date().toISOString(),
        logs: this.getConsoleLogs()
      };
      
      this.executionHistory.push(executionRecord);
      throw executionRecord;
    }
  }
  
  createMockEvent(eventType) {
    const baseEvent = {
      preventDefault: () => console.log('event.preventDefault() called'),
      stopPropagation: () => console.log('event.stopPropagation() called'),
      target: {
        value: 'Mock Value',
        checked: true,
        files: [],
        parentElement: null
      },
      currentTarget: {
        focus: () => console.log('currentTarget.focus() called'),
        blur: () => console.log('currentTarget.blur() called')
      }
    };
    
    switch(eventType) {
      case 'onClick':
        return {
          ...baseEvent,
          type: 'click',
          clientX: 100,
          clientY: 100,
          button: 0
        };
        
      case 'onChange':
        return {
          ...baseEvent,
          type: 'change',
          target: {
            ...baseEvent.target,
            value: 'New Value',
            name: 'inputField'
          }
        };
        
      case 'onSubmit':
        return {
          ...baseEvent,
          type: 'submit',
          target: {
            ...baseEvent.target,
            elements: {
              namedItem: (name) => ({ value: 'Mock Form Value' })
            }
          }
        };
        
      case 'onKeyDown':
        return {
          ...baseEvent,
          type: 'keydown',
          key: 'Enter',
          keyCode: 13,
          code: 'Enter',
          shiftKey: false,
          ctrlKey: false,
          altKey: false,
          metaKey: false
        };
        
      default:
        return baseEvent;
    }
  }
  
  simulateNetworkDelay() {
    return new Promise(resolve => 
      setTimeout(resolve, Math.random() * 500 + 100) // 100-600ms delay
    );
  }
  
  logToConsole(level, args) {
    const logEntry = {
      level,
      message: args.map(arg => 
        typeof arg === 'object' ? JSON.stringify(arg, null, 2) : String(arg)
      ).join(' '),
      timestamp: new Date().toISOString()
    };
    
    // Send to UI console
    window.dispatchEvent(new CustomEvent('console-log', { detail: logEntry }));
  }
}
