### **2.1 Pre-commit Hook Configuration**

```bash
#!/bin/bash
# .git/hooks/pre-commit

echo "🔍 Running White Caves Pre-commit Checks..."

# 1. Check for debug statements
if git diff --cached --name-only | xargs grep -l "console\.log\|debugger\|TODO\|FIXME"; then
    echo "❌ Found debug statements in:"
    git diff --cached --name-only | xargs grep -l "console\.log\|debugger\|TODO\|FIXME"
    echo "Please remove before committing."
    exit 1
fi

# 2. Check for large files
MAX_FILE_SIZE=5242880 # 5MB
for file in $(git diff --cached --name-only); do
    file_size=$(git show :"$file" | wc -c)
    if [ $file_size -gt $MAX_FILE_SIZE ]; then
        echo "❌ File too large: $file ($((file_size/1024))KB)"
        echo "Max allowed: $((MAX_FILE_SIZE/1024))KB"
        exit 1
    fi
done

# 3. Run linters based on file type
for file in $(git diff --cached --name-only); do
    case "$file" in
        *.js|*.jsx|*.ts|*.tsx)
            echo "🔧 Running ESLint on $file..."
            npx eslint "$file" --fix
            git add "$file"
            ;;
        *.py)
            echo "🐍 Running Black on $file..."
            black --check "$file"
            if [ $? -ne 0 ]; then
                black "$file"
                git add "$file"
            fi
            ;;
        *.css|*.scss)
            echo "🎨 Running Stylelint on $file..."
            npx stylelint "$file" --fix
            git add "$file"
            ;;
    esac
done

# 4. Run security checks
echo "🔒 Running security checks..."
npm audit --production
if [ $? -ne 0 ]; then
    echo "⚠️  Security vulnerabilities found. Run 'npm audit fix'"
fi

echo "✅ Pre-commit checks passed!"
```

### **3.2 Module Merge Strategy Script**

```python
#!/usr/bin/env python3
"""
Module Merge Manager for White Caves Platform
Handles merging of module branches with dependency resolution
"""

import subprocess
import json
import sys
from typing import Dict, List, Optional
from dataclasses import dataclass

@dataclass
class Module:
    name: str
    branch: str
    owner: str
    dependencies: List[str]
    last_merged: Optional[str] = None
    merge_status: str = "pending"

class ModuleMergeManager:
    def __init__(self, config_path: str = ".modules.json"):
        self.config_path = config_path
        self.modules = self.load_modules()
        self.merge_queue = []
        
    def load_modules(self) -> Dict[str, Module]:
        """Load module configuration from file"""
        try:
            with open(self.config_path, 'r') as f:
                config = json.load(f)
                
            modules = {}
            for module_config in config.get('modules', []):
                module = Module(**module_config)
                modules[module.name] = module
                
            return modules
        except FileNotFoundError:
            print(f"Config file {self.config_path} not found.")
            sys.exit(1)
    
    def check_dependencies(self, module_name: str) -> bool:
        """Check if all dependencies are merged to develop"""
        module = self.modules[module_name]
        
        for dep_name in module.dependencies:
            if dep_name not in self.modules:
                print(f"❌ Unknown dependency: {dep_name}")
                return False
                
            dep_module = self.modules[dep_name]
            if dep_module.merge_status != "merged":
                print(f"❌ Dependency {dep_name} not merged yet")
                return False
                
        return True
    
    def merge_module(self, module_name: str, dry_run: bool = False) -> bool:
        """Merge a module branch to develop"""
        module = self.modules[module_name]
        
        print(f"\n🔄 Merging {module.name} from {module.branch}")
        
        # Check if we're on develop
        current_branch = self.run_git_command("git branch --show-current")
        if current_branch != "develop":
            print("❌ Must be on develop branch to merge")
            return False
        
        # Check dependencies
        if not self.check_dependencies(module_name):
            print("❌ Dependencies not satisfied")
            return False
        
        # Check for conflicts
        print("🔍 Checking for merge conflicts...")
        self.run_git_command(f"git fetch origin {module.branch}")
        
        merge_base = self.run_git_command(
            f"git merge-base develop origin/{module.branch}"
        )
        
        changes = self.run_git_command(
            f"git log --oneline {merge_base}..origin/{module.branch} | wc -l"
        )
        
        print(f"📊 Changes to merge: {changes.strip()} commits")
        
        if dry_run:
            print("✅ Dry run successful - no conflicts")
            return True
        
        # Perform merge
        try:
            print(f"🤝 Merging {module.branch} into develop...")
            self.run_git_command(f"git merge origin/{module.branch} --no-ff")
            
            # Update module status
            module.merge_status = "merged"
            module.last_merged = self.run_git_command("git log -1 --format=%H")
            
            print(f"✅ Successfully merged {module.name}")
            return True
            
        except subprocess.CalledProcessError as e:
            print(f"❌ Merge failed: {e}")
            print("💡 Resolve conflicts and commit, then run:")
            print(f"   git merge --continue")
            return False
    
    def run_git_command(self, command: str) -> str:
        """Execute a git command and return output"""
        try:
            result = subprocess.run(
                command,
                shell=True,
                check=True,
                capture_output=True,
                text=True
            )
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            print(f"Git command failed: {command}")
            print(f"Error: {e.stderr}")
            raise

def main():
    """CLI interface for module merging"""
    import argparse
    
    parser = argparse.ArgumentParser(
        description="White Caves Module Merge Manager"
    )
    
    parser.add_argument(
        "action",
        choices=["merge", "status", "dependencies", "dry-run"],
        help="Action to perform"
    )
    
    parser.add_argument(
        "module",
        nargs="?",
        help="Module name to merge"
    )
    
    parser.add_argument(
        "--dry-run",
        action="store_true",
        help="Perform a dry run without actual merge"
    )
    
    args = parser.parse_args()
    
    manager = ModuleMergeManager()
    
    if args.action == "status":
        print("\n📋 Module Status:")
        print("-" * 80)
        for name, module in manager.modules.items():
            print(f"{name:25} | {module.branch:35} | {module.merge_status:10}")
        print("-" * 80)
        
    elif args.action == "dependencies" and args.module:
        if args.module not in manager.modules:
            print(f"❌ Unknown module: {args.module}")
            return
            
        module = manager.modules[args.module]
        print(f"\n📦 Dependencies for {args.module}:")
        for dep in module.dependencies:
            status = manager.modules[dep].merge_status if dep in manager.modules else "unknown"
            print(f"  - {dep}: {status}")
            
    elif args.action == "merge" and args.module:
        if args.module not in manager.modules:
            print(f"❌ Unknown module: {args.module}")
            return
            
        success = manager.merge_module(args.module, dry_run=args.dry_run)
        sys.exit(0 if success else 1)
        
    elif args.action == "dry-run" and args.module:
        if args.module not in manager.modules:
            print(f"❌ Unknown module: {args.module}")
            return
            
        success = manager.merge_module(args.module, dry_run=True)
        sys.exit(0 if success else 1)

if __name__ == "__main__":
    main()
```

## **5. AUTOMATED BRANCH MANAGEMENT TOOLS**

### **5.1 Branch Cleanup Script**

```python
#!/usr/bin/env python3
"""
White Caves Branch Cleanup Utility
Automatically cleans up merged and stale branches
"""

import subprocess
import sys
from datetime import datetime, timedelta

class BranchCleanup:
    def __init__(self, days_stale=30):
        self.days_stale = days_stale
        self.protected_branches = [
            'master',
            'develop',
            'main',
            'staging',
            'production'
        ]
        
    def get_merged_branches(self):
        """Get list of branches that have been merged into develop"""
        cmd = "git branch --merged develop"
        result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
        
        branches = []
        for line in result.stdout.split('\n'):
            branch = line.strip().replace('* ', '')
            if branch and branch not in self.protected_branches:
                branches.append(branch)
                
        return branches
    
    def get_stale_branches(self):
        """Get branches with no activity for specified days"""
        cutoff_date = datetime.now() - timedelta(days=self.days_stale)
        
        cmd = "git for-each-ref --format='%(refname:short) %(committerdate:iso)' refs/heads/"
        result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
        
        stale_branches = []
        for line in result.stdout.split('\n'):
            if not line:
                continue
                
            branch, date_str = line.split(' ', 1)
            last_commit_date = datetime.fromisoformat(date_str.split(' ')[0])
            
            if (branch not in self.protected_branches and 
                last_commit_date < cutoff_date):
                stale_branches.append(branch)
                
        return stale_branches
    
    def delete_branch(self, branch_name, remote=False):
        """Delete a branch locally or remotely"""
        try:
            if remote:
                cmd = f"git push origin --delete {branch_name}"
            else:
                cmd = f"git branch -D {branch_name}"
                
            subprocess.run(cmd, shell=True, check=True, capture_output=True)
            return True
            
        except subprocess.CalledProcessError as e:
            print(f"Failed to delete {branch_name}: {e.stderr}")
            return False
    
    def cleanup(self, dry_run=False, remote=False):
        """Main cleanup function"""
        print("🧹 White Caves Branch Cleanup")
        print("=" * 50)
        
        # Get branches to clean
        merged_branches = self.get_merged_branches()
        stale_branches = self.get_stale_branches()
        
        print(f"\n📊 Found {len(merged_branches)} merged branches")
        print(f"📊 Found {len(stale_branches)} stale branches (> {self.days_stale} days)")
        
        if dry_run:
            print("\n🔍 Dry Run - No branches will be deleted")
            print("\nMerged branches to delete:")
            for branch in merged_branches:
                print(f"  - {branch}")
                
            print("\nStale branches to delete:")
            for branch in stale_branches:
                print(f"  - {branch}")
                
            return
        
        # Delete merged branches
        print("\n🗑️  Deleting merged branches...")
        for branch in merged_branches:
            print(f"  Deleting {branch}...", end=' ')
            if self.delete_branch(branch, remote=remote):
                print("✅")
            else:
                print("❌")
        
        # Delete stale branches
        print("\n🗑️  Deleting stale branches...")
        for branch in stale_branches:
            print(f"  Deleting {branch}...", end=' ')
            if self.delete_branch(branch, remote=remote):
                print("✅")
            else:
                print("❌")
        
        print(f"\n✅ Cleanup complete!")

def main():
    import argparse
    
    parser = argparse.ArgumentParser(
        description="White Caves Branch Cleanup Utility"
    )
    
    parser.add_argument(
        "--dry-run",
        action="store_true",
        help="Show what would be deleted without actually deleting"
    )
    
    parser.add_argument(
        "--remote",
        action="store_true",
        help="Also delete remote branches"
    )
    
    parser.add_argument(
        "--days",
        type=int,
        default=30,
        help="Number of days after which a branch is considered stale"
    )
    
    args = parser.parse_args()
    
    cleaner = BranchCleanup(days_stale=args.days)
    cleaner.cleanup(dry_run=args.dry_run, remote=args.remote)

if __name__ == "__main__":
    main()
```

```bash
#!/bin/bash
# scripts/create-feature-branch.sh

# Color codes
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

echo -e "${GREEN}🚀 White Caves Feature Branch Creator${NC}"
echo "========================================"

# Check if we're in a git repo
if ! git rev-parse --git-dir > /dev/null 2>&1; then
    echo -e "${RED}❌ Not in a git repository${NC}"
    exit 1
fi

# Ensure we're on develop
CURRENT_BRANCH=$(git branch --show-current)
if [ "$CURRENT_BRANCH" != "develop" ]; then
    echo -e "${YELLOW}⚠️  You're not on develop branch. Switching...${NC}"
    git checkout develop || { echo -e "${RED}❌ Failed to checkout develop${NC}"; exit 1; }
fi

# Pull latest develop
echo -e "${GREEN}📥 Pulling latest changes from develop...${NC}"
git pull origin develop

# Get feature details
read -p "📝 Enter feature type (pillar1/pillar2/pillar3/pillar4/pillar5/module/hotfix): " FEATURE_TYPE

case $FEATURE_TYPE in

        PREFIX="feature/pillar2"

        PREFIX="feature/pillar3"
        echo "🎯 Pillar 3: Departmental CRM"
        ;;
```

```bash
#!/bin/bash
# scripts/release.sh

set -e

echo "🚀 White Caves Release Process"
echo "=============================="

# Parse arguments
VERSION_TYPE="patch"
DRY_RUN=false

while [[ $# -gt 0 ]]; do
  case $1 in
    --major)
      VERSION_TYPE="major"
      shift
      ;;
    --minor)
      VERSION_TYPE="minor"
      shift
      ;;
    --patch)
      VERSION_TYPE="patch"
      shift
      ;;
    --dry-run)
      DRY_RUN=true
      shift
      ;;
    *)
      echo "Unknown option: $1"
      exit 1
      ;;
  esac
done

# Check if we're on develop
CURRENT_BRANCH=$(git branch --show-current)
if [ "$CURRENT_BRANCH" != "develop" ]; then
  echo "❌ Must be on develop branch to release"
  exit 1
fi

# Pull latest
echo "📥 Pulling latest changes..."
git pull origin develop

# Run tests
echo "🧪 Running tests..."
npm run test
npm run build

# Create release branch
RELEASE_VERSION=$(npm version $VERSION_TYPE --no-git-tag-version | sed 's/v//')
RELEASE_BRANCH="release/v$RELEASE_VERSION"

echo "🌿 Creating release branch: $RELEASE_BRANCH"
git checkout -b "$RELEASE_BRANCH"
git add package*.json
git commit -m "chore(release): bump version to v$RELEASE_VERSION"

if [ "$DRY_RUN" = false ]; then
  # Push release branch
  echo "📤 Pushing release branch..."
  git push -u origin "$RELEASE_BRANCH"
  
  # Create PR to master
  echo "📝 Creating PR to master..."
  gh pr create \
    --base master \
    --head "$RELEASE_BRANCH" \
    --title "Release v$RELEASE_VERSION" \
    --body "## Release v$RELEASE_VERSION
  
  ### Changes
  <!-- List major changes -->
  
  ### Deployment Instructions
  1. Merge this PR to master
  2. CI/CD will deploy to production
  3. Verify deployment at https://whitecaves.ae
  
  ### Rollback Plan
  - Tag: v$RELEASE_VERSION
  - Database backup: automated
  - Rollback script: scripts/rollback.sh"
  
  echo "✅ Release process initiated!"
  echo "📋 Next steps:"
  echo "   1. Wait for PR approval"
  echo "   2. Merge PR to master"
  echo "   3. CI/CD will deploy automatically"
else
  echo "🔍 Dry run complete!"
  echo "Would create release branch: $RELEASE_BRANCH"
  echo "Would bump version to: v$RELEASE_VERSION"
  
  # Clean up
  git checkout develop
  git branch -D "$RELEASE_BRANCH"
  git checkout package*.json
fi
```

### **8.2 Rollback Script**

```bash
#!/bin/bash
# scripts/rollback.sh

set -e

echo "🔄 White Caves Rollback Process"
echo "==============================="

# Get current version
CURRENT_VERSION=$(git describe --tags --abbrev=0)
echo "Current version: $CURRENT_VERSION"

# List recent tags
echo "📋 Recent releases:"
git tag --sort=-v:refname | head -10 | cat -n

# Ask for target version
read -p "🎯 Enter version to rollback to (e.g., v1.2.0): " TARGET_VERSION

# Verify tag exists
if ! git rev-parse "$TARGET_VERSION" >/dev/null 2>&1; then
  echo "❌ Tag $TARGET_VERSION does not exist"
  exit 1
fi

# Confirm rollback
read -p "⚠️  Are you sure you want to rollback to $TARGET_VERSION? (y/n): " CONFIRM
if [[ ! $CONFIRM =~ ^[Yy]$ ]]; then
  echo "Rollback cancelled"
  exit 0
fi

# Create rollback branch
ROLLBACK_BRANCH="rollback/to-$TARGET_VERSION"
echo "🌿 Creating rollback branch: $ROLLBACK_BRANCH"
git checkout -b "$ROLLBACK_BRANCH"

# Revert to target version
echo "⏮️  Reverting to $TARGET_VERSION..."
git reset --hard "$TARGET_VERSION"

# Create revert commit
git commit --allow-empty -m "chore(rollback): revert to $TARGET_VERSION"

if [ "$DRY_RUN" = false ]; then
  # Push rollback branch
  echo "📤 Pushing rollback branch..."
  git push -u origin "$ROLLBACK_BRANCH"
  
  # Create PR
  echo "📝 Creating rollback PR..."
  gh pr create \
    --base master \
    --head "$ROLLBACK_BRANCH" \
    --title "Rollback to $TARGET_VERSION" \
    --body "## Rollback to $TARGET_VERSION
  
  ### Reason for rollback
  <!-- Explain why rollback is needed -->
  
  ### Impact Assessment
  - Database changes: <!-- Yes/No -->
  - Breaking changes: <!-- Yes/No -->
  - User impact: <!-- High/Medium/Low -->
  
  ### Verification Steps
  1. Check application health
  2. Verify critical functionality
  3. Monitor error rates"
  
  echo "✅ Rollback initiated!"
  echo "📋 Next steps:"
  echo "   1. Get PR approved"
  echo "   2. Merge to master"
  echo "   3. CI/CD will deploy rollback"
else
  echo "🔍 Dry run complete!"
  echo "Would rollback to: $TARGET_VERSION"
  
  # Clean up
  git checkout master
  git branch -D "$ROLLBACK_BRANCH"
fi
```

### **10.1 Git Analytics Dashboard**

```python
#!/usr/bin/env python3
"""
White Caves Git Analytics
Track development metrics and team productivity
"""

import subprocess
import json
from datetime import datetime, timedelta
from collections import defaultdict

class GitAnalytics:
    def __init__(self, repo_path="."):
        self.repo_path = repo_path
        
    def get_commit_stats(self, days=30):
        """Get commit statistics for the last N days"""
        since_date = (datetime.now() - timedelta(days=days)).strftime("%Y-%m-%d")
        
        cmd = f"""
        git log --since="{since_date}" \
        --format='%aN|%ad|%s' \
        --date=short \
        --numstat
        """
        
        result = subprocess.run(
            cmd, 
            shell=True, 
            capture_output=True, 
            text=True, 
            cwd=self.repo_path
        )
        
        stats = {
            "total_commits": 0,
            "authors": defaultdict(int),
            "files_changed": 0,
            "lines_added": 0,
            "lines_deleted": 0,
            "by_pillar": defaultdict(int)
        }
        
        lines = result.stdout.strip().split('\n')
        i = 0
        while i < len(lines):
            if '|' in lines[i]:  # Commit line
                author, date, message = lines[i].split('|', 2)
                stats["total_commits"] += 1
                stats["authors"][author] += 1
                
                # Categorize by pillar
                for pillar in ["pillar1", "pillar2", "pillar3", "pillar4", "pillar5"]:
                    if pillar in message.lower():
                        stats["by_pillar"][pillar] += 1
                        break
                        
                i += 1
                
                # Parse file stats
                while i < len(lines) and lines[i] and '\t' in lines[i]:
                    added, deleted, file = lines[i].split('\t')
                    stats["files_changed"] += 1
                    stats["lines_added"] += int(added) if added != '-' else 0
                    stats["lines_deleted"] += int(deleted) if deleted != '-' else 0
                    i += 1
            else:
                i += 1
                
        return stats
    
    def get_branch_health(self):
        """Analyze branch health metrics"""
        cmd = """
        git for-each-ref \
        --format='%(refname:short)|%(committerdate:relative)|%(authorname)' \
        refs/heads/
        """
        
        result = subprocess.run(
            cmd, 
            shell=True, 
            capture_output=True, 
            text=True, 
            cwd=self.repo_path
        )
        
        branches = []
        for line in result.stdout.strip().split('\n'):
            if not line:
                continue
                
            name, age, author = line.split('|', 2)
            branches.append({
                "name": name,
                "age": age,
                "author": author,
                "is_stale": "week" in age and int(age.split()[0]) > 2
            })
            
        return branches
    
    def generate_report(self):
        """Generate comprehensive analytics report"""
        print("📊 White Caves Git Analytics Report")
        print("=" * 50)
        
        # Commit stats
        stats = self.get_commit_stats(30)
        print(f"\n📈 Last 30 Days:")
        print(f"  Total Commits: {stats['total_commits']}")
        print(f"  Files Changed: {stats['files_changed']}")
        print(f"  Lines Added: {stats['lines_added']:,}")
        print(f"  Lines Deleted: {stats['lines_deleted']:,}")
        
        print(f"\n👥 By Author:")
        for author, count in sorted(stats['authors'].items(), key=lambda x: x[1], reverse=True):
            print(f"  {author}: {count} commits")
            
        print(f"\n🎯 By Pillar:")
        for pillar, count in stats['by_pillar'].items():
            print(f"  {pillar}: {count} commits")
            
        # Branch health
        branches = self.get_branch_health()
        stale_branches = [b for b in branches if b['is_stale']]
        
        print(f"\n🌿 Branch Health:")
        print(f"  Total Branches: {len(branches)}")
        print(f"  Stale Branches (>2 weeks): {len(stale_branches)}")
        
        if stale_branches:
            print(f"\n⚠️  Stale Branches:")
            for branch in stale_branches[:5]:  # Show top 5
                print(f"  - {branch['name']} ({branch['age']}) by {branch['author']}")

def main():
    import argparse
    
    parser = argparse.ArgumentParser(
        description="White Caves Git Analytics"
    )
    
    parser.add_argument(
        "--report",
        action="store_true",
        help="Generate analytics report"
    )
    
    parser.add_argument(
        "--json",
        action="store_true",
        help="Output as JSON"
    )
    
    parser.add_argument(
        "--days",
        type=int,
        default=30,
        help="Number of days to analyze"
    )
    
    args = parser.parse_args()
    
    analytics = GitAnalytics()
    
    if args.json:
        stats = analytics.get_commit_stats(args.days)
        print(json.dumps(stats, indent=2))
    else:
        analytics.generate_report()

if __name__ == "__main__":
    main()
```
