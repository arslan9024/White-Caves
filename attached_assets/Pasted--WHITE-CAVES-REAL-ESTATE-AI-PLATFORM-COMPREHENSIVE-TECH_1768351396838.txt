## **PART 3: DLD/RERA/RDC INTEGRATION SPECIFICATIONS**

### **3.1 DLD (DUBAI LAND DEPARTMENT) INTEGRATION**

#### **API Endpoints Integration:**
```javascript
// Complete DLD API Integration Specification
class DLDApiIntegration {
  constructor() {
    this.baseUrl = 'https://api.dubailand.gov.ae/v2';
    this.apiKey = process.env.DLD_API_KEY;
    this.certificatePath = '/ssl/dld_certificate.pem';
  }

  // 1. TITLE DEED VERIFICATION
  async verifyTitleDeed(titleDeedNumber, propertyNumber) {
    const response = await axios.post(
      `${this.baseUrl}/title-deed/verify`,
      {
        title_deed_number: titleDeedNumber,
        property_number: propertyNumber,
        request_source: 'white_caves_platform',
        timestamp: new Date().toISOString()
      },
      {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json',
          'X-DLD-Certificate': this.loadCertificate()
        },
        httpsAgent: new https.Agent({
          cert: this.certificatePath,
          key: this.certificatePath
        })
      }
    );

    // Response Structure
    return {
      isValid: response.data.status === 'VALID',
      ownerDetails: {
        name: response.data.owner_name,
        nationality: response.data.owner_nationality,
        emiratesId: response.data.owner_emirates_id,
        ownershipPercentage: response.data.ownership_percentage
      },
      propertyDetails: {
        type: response.data.property_type,
        area: response.data.property_area,
        location: response.data.property_location,
        community: response.data.property_community,
        plotNumber: response.data.plot_number
      },
      encumbrances: response.data.encumbrances || [],
      restrictions: response.data.restrictions || [],
      lastTransferDate: response.data.last_transfer_date,
      verificationId: response.data.verification_id,
      timestamp: response.data.timestamp
    };
  }

  // 2. PROPERTY TRANSACTION REGISTRATION
  async registerTransaction(transactionData) {
    const dldTransaction = {
      transaction_type: transactionData.type, // SALE, GIFT, INHERITANCE, etc.
      transaction_date: transactionData.date,
      transaction_value: transactionData.amount,
      parties: transactionData.parties.map(party => ({
        type: party.type, // BUYER, SELLER, BROKER
        name: party.name,
        emirates_id: party.emiratesId,
        passport_number: party.passportNumber,
        nationality: party.nationality,
        address: party.address,
        contact_number: party.phone,
        email: party.email
      })),
      property_details: {
        title_deed_number: transactionData.property.titleDeedNumber,
        property_number: transactionData.property.propertyNumber,
        area: transactionData.property.area,
        location: transactionData.property.location
      },
      documents: transactionData.documents.map(doc => ({
        type: doc.type, // MOU, PASSPORT_COPY, TITLE_DEED, etc.
        document_number: doc.documentNumber,
        issue_date: doc.issueDate,
        issuing_authority: doc.issuingAuthority,
        document_url: doc.documentUrl // Secure S3 URL
      })),
      fees: {
        transfer_fee: this.calculateTransferFee(transactionData.amount),
        knowledge_fee: 10, // AED 10
        innovation_fee: 10, // AED 10
        broker_fee: transactionData.brokerCommission || 0,
        total_amount: this.calculateTotalFees(transactionData.amount)
      },
      payment_details: {
        method: transactionData.paymentMethod,
        transaction_reference: transactionData.paymentReference,
        bank_name: transactionData.bankName,
        account_number: transactionData.accountNumber
      },
      broker_details: {
        rera_number: transactionData.brokerReraNumber,
        company_name: transactionData.brokerCompany,
        broker_name: transactionData.brokerName
      }
    };

    const response = await axios.post(
      `${this.baseUrl}/transactions/register`,
      dldTransaction,
      {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json',
          'X-Request-ID': this.generateRequestId(),
          'X-Correlation-ID': transactionData.correlationId
        }
      }
    );

    return {
      dldReference: response.data.transaction_reference,
      status: response.data.status,
      estimatedCompletion: response.data.estimated_completion_date,
      feesPayable: response.data.fees_payable,
      paymentDeadline: response.data.payment_deadline,
      nextSteps: response.data.next_steps
    };
  }

  // 3. PROPERTY VALUATION CERTIFICATE
  async requestValuationCertificate(propertyDetails) {
    const valuationRequest = {
      property_type: propertyDetails.type,
      built_up_area: propertyDetails.builtUpArea,
      plot_area: propertyDetails.plotArea,
      location: propertyDetails.location,
      community: propertyDetails.community,
      age_of_property: propertyDetails.age,
      condition: propertyDetails.condition,
      amenities: propertyDetails.amenities,
      purpose: propertyDetails.purpose // MORTGAGE, TRANSFER, INHERITANCE
    };

    const response = await axios.post(
      `${this.baseUrl}/valuation/request`,
      valuationRequest,
      {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json'
        }
      }
    );

    return {
      valuationId: response.data.valuation_id,
      estimatedValue: response.data.estimated_value,
      validityPeriod: response.data.validity_period, // Usually 6 months
      certificateUrl: response.data.certificate_url,
      issuedDate: response.data.issued_date,
      valuerDetails: response.data.valuer_details
    };
  }

  // 4. OFF-PLAN PROJECT REGISTRATION
  async registerOffPlanProject(projectData) {
    const projectRegistration = {
      developer_rera_number: projectData.developerReraNumber,
      project_name: projectData.projectName,
      location: projectData.location,
      total_units: projectData.totalUnits,
      unit_types: projectData.unitTypes,
      completion_date: projectData.completionDate,
      escrow_account_details: projectData.escrowAccount,
      approved_documents: projectData.approvedDocuments
    };

    const response = await axios.post(
      `${this.baseUrl}/off-plan/register`,
      projectRegistration,
      {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json'
        }
      }
    );

    return {
      projectNumber: response.data.project_number,
      reraPermitNumber: response.data.rera_permit_number,
```
