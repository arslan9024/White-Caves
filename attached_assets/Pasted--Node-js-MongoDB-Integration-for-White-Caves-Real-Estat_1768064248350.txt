## 🗄️ **Database Configuration**

```javascript
// config/database.js
const { MongoClient } = require('mongodb');
require('dotenv').config();

class Database {
  constructor() {
    this.client = null;
    this.db = null;
    this.isConnected = false;
    this.collections = {};
  }

  async connect() {
    try {
      const uri = process.env.MONGODB_URI || 'mongodb://localhost:27017';
      const dbName = process.env.DB_NAME || 'whitecaves_realestate';
      
      this.client = new MongoClient(uri, {
        maxPoolSize: 50,
        minPoolSize: 10,
        maxIdleTimeMS: 10000,
        serverSelectionTimeoutMS: 5000,
        socketTimeoutMS: 45000,
      });

      await this.client.connect();
      this.db = this.client.db(dbName);
      this.isConnected = true;
      
      console.log('✅ MongoDB Connected Successfully');
      console.log(`📊 Database: ${dbName}`);
      
      // Initialize collections
      await this.initializeCollections();
      
      return this.db;
    } catch (error) {
      console.error('❌ MongoDB Connection Error:', error);
      throw error;
    }
  }

  async initializeCollections() {
    const collections = [
      'departments',
      'employees',
      'services',
      'ai_assistants',
      'department_ai_assignments',
      'service_ai_support',
      'employee_ai_access',
      'transactions',
      'audit_logs'
    ];

    for (const collectionName of collections) {
      const collection = this.db.collection(collectionName);
      
      // Create indexes based on collection type
      await this.createCollectionIndexes(collectionName, collection);
      
      this.collections[collectionName] = collection;
    }
  }

  async createCollectionIndexes(collectionName, collection) {
    switch (collectionName) {
      case 'departments':
        await collection.createIndex({ department_id: 1 }, { unique: true });
        await collection.createIndex({ 'metadata.status': 1 });
        await collection.createIndex({ director_employee_id: 1 });
        break;
      
      case 'employees':
        await collection.createIndex({ employee_id: 1 }, { unique: true });
        await collection.createIndex({ email: 1 }, { unique: true });
        await collection.createIndex({ department_id: 1 });
        await collection.createIndex({ 'employment.status': 1 });
        break;
      
      case 'services':
        await collection.createIndex({ service_id: 1 }, { unique: true });
        await collection.createIndex({ category: 1 });
        await collection.createIndex({ 'metadata.status': 1 });
        break;
      
      case 'ai_assistants':
        await collection.createIndex({ assistant_id: 1 }, { unique: true });
        await collection.createIndex({ department_id: 1 });
        await collection.createIndex({ type: 1 });
        break;
      
      case 'transactions':
        await collection.createIndex({ 'year_month': 1, 'department_id': 1 });
        await collection.createIndex({ 'summary.total_value_aed': -1 });
        break;
    }
    
    console.log(`📈 Indexes created for ${collectionName}`);
  }

  getCollection(name) {
    if (!this.collections[name]) {
      throw new Error(`Collection ${name} not initialized`);
    }
    return this.collections[name];
  }

  async disconnect() {
    if (this.client) {
      await this.client.close();
      this.isConnected = false;
      console.log('🔌 MongoDB Disconnected');
    }
  }

  async healthCheck() {
    try {
      await this.db.command({ ping: 1 });
      return {
        status: 'healthy',
        database: this.db.databaseName,
        collections: Object.keys(this.collections).length
      };
    } catch (error) {
      return {
        status: 'unhealthy',
        error: error.message
      };
    }
  }
}

module.exports = new Database();
```

```javascript
// models/Department.js
const { ObjectId } = require('mongodb');
const db = require('../config/database');
const { generateDepartmentId } = require('../utils/idGenerator');

class Department {
  constructor(data) {
    this.department_id = data.department_id || generateDepartmentId();
    this.name = data.name;
    this.mission = data.mission;
    this.color_hex = data.color_hex || '#3B82F6';
    this.director_employee_id = data.director_employee_id;
    this.ai_assistants = data.ai_assistants || [];
    this.stats = data.stats || {
      total_employees: 0,
      total_services: 0,
      monthly_transactions: 0
    };
    this.metadata = {
      created_date: data.metadata?.created_date || new Date(),
      hierarchy_level: data.metadata?.hierarchy_level || 'department',
      status: data.metadata?.status || 'active',
      ...data.metadata
    };
  }

  async save() {
    const collection = db.getCollection('departments');
    return await collection.insertOne(this);
  }

  static async findById(departmentId) {
    const collection = db.getCollection('departments');
    return await collection.findOne({ department_id: departmentId });
  }

  static async findAll(filter = {}) {
    const collection = db.getCollection('departments');
    return await collection.find(filter).toArray();
  }

  static async update(departmentId, updateData) {
    const collection = db.getCollection('departments');
    return await collection.updateOne(
      { department_id: departmentId },
      { 
        $set: {
          ...updateData,
          'metadata.last_updated': new Date()
        }
      }
    );
  }

  static async delete(departmentId) {
    const collection = db.getCollection('departments');
    return await collection.updateOne(
      { department_id: departmentId },
      { 
        $set: {
          'metadata.status': 'inactive',
          'metadata.deleted_at': new Date()
        }
      }
    );
  }

  static async getWithAI(departmentId) {
    const collection = db.getCollection('departments');
    return await collection.aggregate([
      { $match: { department_id: departmentId } },
      {
        $lookup: {
          from: 'ai_assistants',
          localField: 'ai_assistants.assistant_id',
          foreignField: 'assistant_id',
          as: 'ai_assistants_details'
        }
      },
      {
        $lookup: {
          from: 'employees',
          localField: 'director_employee_id',
          foreignField: 'employee_id',
          as: 'director_info'
        }
      },
      {
        $unwind: {
          path: '$director_info',
          preserveNullAndEmptyArrays: true
        }
      }
    ]).toArray();
  }

  static async getDepartmentStatistics() {
    const collection = db.getCollection('departments');
    const departments = await collection.find({ 'metadata.status': 'active' }).toArray();
    
    const stats = {
      total_departments: departments.length,
      total_employees: departments.reduce((sum, dept) => sum + (dept.stats?.total_employees || 0), 0),
      total_services: departments.reduce((sum, dept) => sum + (dept.stats?.total_services || 0), 0),
      total_monthly_transactions: departments.reduce((sum, dept) => sum + (dept.stats?.monthly_transactions || 0), 0),
      departments: departments.map(dept => ({
        name: dept.name,
        employee_count: dept.stats?.total_employees || 0,
        ai_assistants_count: dept.ai_assistants?.length || 0,
        monthly_transactions: dept.stats?.monthly_transactions || 0
      }))
    };
    
    return stats;
  }

  static async assignAIAssistant(departmentId, assistantData) {
    const collection = db.getCollection('departments');
    return await collection.updateOne(
      { department_id: departmentId },
      { 
        $addToSet: { 
          ai_assistants: assistantData 
        },
        $set: {
          'metadata.last_updated': new Date()
        }
      }
    );
  }

  static async removeAIAssistant(departmentId, assistantId) {
    const collection = db.getCollection('departments');
    return await collection.updateOne(
      { department_id: departmentId },
      { 
        $pull: { 
          ai_assistants: { assistant_id: assistantId }
        },
        $set: {
          'metadata.last_updated': new Date()
        }
      }
    );
  }
}

module.exports = Department;
```

```javascript
const db = require('../config/database');
const { generateEmployeeId, validateEmail } = require('../utils/idGenerator');
const { validateUAEPhone, validateEmiratesID } = require('../utils/validators');

class Employee {
  constructor(data) {
    this.employee_id = data.employee_id || generateEmployeeId();
    this.internal_id = data.internal_id || '';
    this.name = data.name;
    this.email = data.email;
    this.department_id = data.department_id;
    this.job_title = data.job_title;
    this.reports_to = data.reports_to || null;
    
    this.responsibilities = data.responsibilities || [];
    
    this.contact = {
      work_phone: validateUAEPhone(data.contact?.work_phone),
      mobile: validateUAEPhone(data.contact?.mobile),
      uae_pass_id: validateEmiratesID(data.contact?.uae_pass_id),
      ...data.contact
    };
    
    this.employment = {
      type: data.employment?.type || 'full-time',
      hire_date: data.employment?.hire_date || new Date(),
      employment_status: data.employment?.employment_status || 'active',
      cost_center: data.employment?.cost_center || '',
      salary_band: data.employment?.salary_band || '',
      ...data.employment
    };
    
    this.ai_access_permissions = data.ai_access_permissions || [];
    
    this.metadata = {
      created_at: data.metadata?.created_at || new Date(),
      last_updated: data.metadata?.last_updated || new Date(),
      version: data.metadata?.version || 1,
      created_by: data.metadata?.created_by || 'system',
      ...data.metadata
    };
  }

  validate() {
    const errors = [];
    
    if (!validateEmail(this.email)) {
      errors.push('Invalid email format');
    }
    
    if (!this.email.endsWith('@whitecaves.ae')) {
      errors.push('Email must be from whitecaves.ae domain');
    }
    
    if (!this.department_id) {
      errors.push('Department ID is required');
    }
    
    if (!this.job_title) {
      errors.push('Job title is required');
    }
    
    return errors;
  }

  async save() {
    const errors = this.validate();
    if (errors.length > 0) {
      throw new Error(`Validation failed: ${errors.join(', ')}`);
    }
    
    const collection = db.getCollection('employees');
    return await collection.insertOne(this);
  }

    const collection = db.getCollection('employees');
    return await collection.findOne({ employee_id: employeeId });

      {
        $graphLookup: {
          from: 'employees',
          startWith: '$reports_to',
          connectFromField: 'reports_to',
          connectToField: 'employee_id',
          as: 'management_chain',
          depthField: 'depth',
          maxDepth: maxDepth
        }
      },
      {
        $project: {
          employee_id: 1,
          name: 1,
          job_title: 1,
          department_id: 1,
          reports_to: 1,
          management_chain: {
            $map: {
              input: '$management_chain',
              as: 'manager',
              in: {
                employee_id: '$$manager.employee_id',
                name: '$$manager.name',
                job_title: '$$manager.job_title',
                depth: '$$manager.depth'
              }
            }
          }
        }
      }
    ];
    
    return await collection.aggregate(pipeline).toArray();
  }
```

```javascript
  static async addTransactionToBucket(yearMonth, departmentId, transaction) {
    const collection = db.getCollection('transactions');
    
    const transactionData = {
      transaction_id: generateTransactionId(),
      ...transaction,
      timestamp: new Date()
    };
    
    // Find or create bucket
    const bucketId = `${yearMonth}-${departmentId}`;
    let bucket = await collection.findOne({ 
      year_month: yearMonth, 
      department_id: departmentId 
    });
    
    if (!bucket) {
      bucket = {
        bucket_type: 'monthly',
        year_month: yearMonth,
        department_id: departmentId,
        summary: {
          total_transactions: 1,
          total_value_aed: transaction.value_aed || 0,
          avg_transaction_value: transaction.value_aed || 0
        },
        sample_transaction: transactionData,
        created_at: new Date()
      };
      return await collection.insertOne(bucket);
    } else {
      // Update existing bucket
      const updateOperations = {
        $inc: {
          'summary.total_transactions': 1,
          'summary.total_value_aed': transaction.value_aed || 0
        },
        $set: {
          'summary.avg_transaction_value': {
            $divide: [
              { $add: [
                '$summary.total_value_aed', 
                transaction.value_aed || 0
              ]},
              { $add: ['$summary.total_transactions', 1] }
            ]
          },
          updated_at: new Date()
        }
      };
      
      return await collection.updateOne(
        { year_month: yearMonth, department_id: departmentId },
        updateOperations
      );
    }
  }
```

```javascript
function validateUAEPhone(phone) {
  if (!phone) return null;
  
  const cleaned = phone.replace(/\D/g, '');
  const uaeRegex = /^(9715|05)\d{8}$/;
  
  if (uaeRegex.test(cleaned)) {
    return `+971${cleaned.slice(-9)}`;
  }
  
  // If already in international format
  if (phone.startsWith('+9715') && phone.length === 13) {
    return phone;
  }
  
  throw new Error('Invalid UAE phone number format');
}
```

```javascript
// Calculate commission based on property value
function calculateCommission(propertyValue, serviceType = 'standard') {
  const rates = {
    'standard': 0.02, // 2%
    'luxury': 0.015,  // 1.5% for luxury (>10M)
    'off-plan': 0.025, // 2.5%
    'commercial': 0.0175 // 1.75%
  };
  
  const rate = rates[serviceType] || rates.standard;
  return propertyValue * rate;
}

// Generate year-month string for transaction buckets
function getYearMonth(date = new Date()) {
  return moment(date).format('YYYY-MM');
}

// Calculate service duration in business days
function calculateBusinessDays(startDate, endDate) {
  let count = 0;
  const current = new Date(startDate);
  const end = new Date(endDate);
  
  while (current <= end) {
    const dayOfWeek = current.getDay();
    if (dayOfWeek !== 5 && dayOfWeek !== 6) { // Not Friday or Saturday
      count++;
    }
    current.setDate(current.getDate() + 1);
  }
  
  return count;
}

function validateRERALicense(licenseNumber) {
  // RERA license format: RERA-XXXXX

  return regex.test(licenseNumber);
}

function calculateROI(purchasePrice, annualRent, expenses = 0) {

  if (!phone) return '';
  
  if (phone.startsWith('+9715')) {
    return phone.replace('+9715', '05');
  }
  
  return phone;
}

// Get next business day (skip weekends)
function getNextBusinessDay(date = new Date()) {
  const nextDay = new Date(date);
  nextDay.setDate(nextDay.getDate() + 1);
  
  const dayOfWeek = nextDay.getDay();
  if (dayOfWeek === 5) { // Friday
    nextDay.setDate(nextDay.getDate() + 2);
  } else if (dayOfWeek === 6) { // Saturday
    nextDay.setDate(nextDay.getDate() + 1);
  }
  
  return nextDay;
}
```

```javascript
      summary.by_employment_type[empType] = (summary.by_employment_type[empType] || 0) + 1;
      
      // Count by status

      summary.by_status[status] = (summary.by_status[status] || 0) + 1;
```

```javascript
      }
      
      grouped[category].count++;
      grouped[category].total_monthly_volume += service.metrics?.monthly_volume || 0;
      grouped[category].services.push({
        name: service.name,
        monthly_volume: service.metrics?.monthly_volume || 0,
        avg_revenue: service.metrics?.avg_revenue_per_service || 0
      });
    });
    
    return grouped;
  }

  async getAIPerformanceSummary(aiAssistants) {
    const performance = {
      total_queries: 0,
      avg_uptime: 0,
      avg_response_time: 0,
      top_performers: [],
      needs_attention: []
    };
    
    let totalUptime = 0;
    let totalResponseTime = 0;
    
    aiAssistants.forEach(ai => {
      const perf = ai.performance || {};
      
      performance.total_queries += perf.monthly_queries || 0;
      totalUptime += perf.uptime_percentage || 0;
      totalResponseTime += perf.avg_response_time_ms || 0;
      
      // Check for top performers (uptime > 99.5% and error rate < 0.5%)
      if ((perf.uptime_percentage || 0) > 99.5 && (perf.error_rate || 0) < 0.005) {
        performance.top_performers.push({
          name: ai.name,
          uptime: perf.uptime_percentage,
          error_rate: perf.error_rate,
          monthly_queries: perf.monthly_queries
        });
      }
      
      // Check for AI needing attention (uptime < 95% or error rate > 2%)
      if ((perf.uptime_percentage || 0) < 95 || (perf.error_rate || 0) > 0.02) {
        performance.needs_attention.push({
          name: ai.name,
          uptime: perf.uptime_percentage,
          error_rate: perf.error_rate,
          issues: [
            ...(perf.uptime_percentage < 95 ? ['Low uptime'] : []),
            ...(perf.error_rate > 0.02 ? ['High error rate'] : [])
          ]
        });
      }
    });
    
    performance.avg_uptime = aiAssistants.length > 0 
      ? totalUptime / aiAssistants.length 
      : 0;
    performance.avg_response_time = aiAssistants.length > 0 
      ? totalResponseTime / aiAssistants.length 
      : 0;
    
    // Sort top performers by monthly queries
    performance.top_performers.sort((a, b) => b.monthly_queries - a.monthly_queries);
    
    return performance;
  }

  generateInsights(departments, employees, services, aiAssistants) {
    const insights = [];
    
    // Employee distribution insight
    const avgEmployeesPerDept = employees.length / Math.max(departments.length, 1);

    );
    const understaffed = departments.filter(dept =>

        level: 'warning',

      });
```

```javascript
// middleware/auth.js
const jwt = require('jsonwebtoken');
const config = require('../config/environment');

// Simple authentication middleware
const authenticate = (req, res, next) => {
  const token = req.headers['authorization']?.split(' ')[1];
  
  if (!token) {
    return res.status(401).json({
      success: false,
      error: 'Access token required'
    });
  }
  
  try {
    const decoded = jwt.verify(token, config.JWT_SECRET);
    req.user = decoded;
    next();
  } catch (error) {
    return res.status(401).json({
      success: false,
      error: 'Invalid token'
    });
  }
};

// Authorization middleware
const authorize = (roles) => {
  return (req, res, next) => {
    if (!req.user) {
      return res.status(401).json({
        success: false,
        error: 'User not authenticated'
      });
    }
    
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({
        success: false,
        error: 'Insufficient permissions'
      });
    }
    
    next();
  };
};

module.exports = {
  authenticate,
  authorize
};
```

```javascript
// server.js
const express = require('express');
const cors = require('cors');
const helmet = require('helmet');
const rateLimit = require('express-rate-limit');
require('dotenv').config();

// Database connection
const database = require('./config/database');
const config = require('./config/environment');

// Import routes
const apiRoutes = require('./routes/api');

// Initialize Express app
const app = express();

// Security middleware
app.use(helmet());
app.use(cors({
  origin: process.env.CORS_ORIGIN || 'http://localhost:3000',
  credentials: true
}));

// Rate limiting
const limiter = rateLimit({
  windowMs: config.RATE_LIMIT_WINDOW,
  max: config.RATE_LIMIT_MAX,
  message: {
    success: false,
    error: 'Too many requests from this IP, please try again later'
  }
});
app.use('/api', limiter);

// Body parsing middleware
app.use(express.json({ limit: '10mb' }));
app.use(express.urlencoded({ extended: true, limit: '10mb' }));

// Request logging middleware
app.use((req, res, next) => {
  console.log(`${new Date().toISOString()} - ${req.method} ${req.url}`);
  next();
});

// API Routes
app.use('/api/v1', apiRoutes);

// 404 Handler
app.use((req, res) => {
  res.status(404).json({
    success: false,
    error: 'Endpoint not found'
  });
});

// Error handling middleware
app.use((err, req, res, next) => {
  console.error('Server Error:', err);
  
  res.status(err.status || 500).json({
    success: false,
    error: process.env.NODE_ENV === 'development' 
      ? err.message 
      : 'Internal server error'
  });
});

// Start server function
async function startServer() {
  try {
    // Connect to database
    await database.connect();
    
    // Start server
    const server = app.listen(config.PORT, () => {
      console.log(`🚀 Server running on port ${config.PORT}`);
      console.log(`📊 Environment: ${config.NODE_ENV}`);
      console.log(`🔗 API Base: http://localhost:${config.PORT}/api/v1`);
    });
    
    // Graceful shutdown
    process.on('SIGTERM', () => {
      console.log('SIGTERM received, shutting down gracefully');
      server.close(() => {
        database.disconnect();
        console.log('Server closed');
        process.exit(0);
      });
    });
    
    process.on('SIGINT', () => {
      console.log('SIGINT received, shutting down gracefully');
      server.close(() => {
        database.disconnect();
        console.log('Server closed');
        process.exit(0);
      });
    });
    
  } catch (error) {
    console.error('Failed to start server:', error);
    process.exit(1);
  }
}

// Start the server
startServer();
```
