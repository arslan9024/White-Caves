    /**
     * Analyze conversation for property leads with ML
     */
    async analyzeConversationForLeads(contactId, forceAnalysis = false) {
        const cacheKey = contactId;
        const cached = this.analyzedChats.get(cacheKey);
        
        // Return cached result if available and not forced
        if (cached && !forceAnalysis && 
            moment().diff(moment(cached.timestamp), 'hours') < 24) {
            return cached.result;
        }
        
        console.log(`🔍 AI analyzing conversation with ${contactId} for property leads`);
        
        try {
            const chat = await this.client.getChatById(contactId);
            const messages = await chat.fetchMessages({ limit: 200 }); // Get last 200 messages
            
            if (!messages || messages.length === 0) {
                return this.createEmptyAnalysis(contactId, 'No messages found');
            }
            
            // Filter messages older than 1 year with multiple exchanges
            const oldMessages = this.filterOldConversations(messages);
            
            if (oldMessages.length < 5) { // Need at least 5 messages for meaningful analysis
                return this.createEmptyAnalysis(contactId, 'Insufficient conversation history');
            }
            
            // Analyze conversation using multiple ML approaches
            const analysis = await this.performMultiAnalysis(oldMessages, contactId);
            
            // Determine if this is a verified property lead
            const isVerifiedLead = this.evaluateAsVerifiedLead(analysis);
            
            const result = {
                contactId,
                isVerifiedLead,
                analysis,
                leadScore: analysis.leadScore,
                confidence: analysis.confidence,
                verificationDate: isVerifiedLead ? new Date().toISOString() : null,
                lastAnalyzed: new Date().toISOString()
            };
            
            // Cache result
            this.analyzedChats.set(cacheKey, {
                result,
                timestamp: new Date().toISOString()
            });
            
            return result;
            
        } catch (error) {
            console.error(`AI analysis failed for ${contactId}:`, error);
            return this.createEmptyAnalysis(contactId, `Analysis error: ${error.message}`);
        }
    }
    
    /**
     * Filter conversations older than 1 year with multiple exchanges
     */
    filterOldConversations(messages) {
        const oneYearAgo = moment().subtract(1, 'year');
        
        return messages.filter(msg => {
            const msgDate = moment(msg.timestamp * 1000);
            
            // Check if message is older than 1 year
            if (msgDate.isAfter(oneYearAgo)) {
                return false;
            }
            
            // Check if message has at least 20 characters (meaningful content)
            if (!msg.body || msg.body.length < 20) {
                return false;
            }
            
            return true;
        });
    }
    
    /**
     * Perform multi-layered ML analysis
     */
    async performMultiAnalysis(messages, contactId) {
        // 1. Basic keyword analysis
        const keywordAnalysis = this.performKeywordAnalysis(messages);
        
        // 2. Sentiment analysis
        const sentimentAnalysis = this.performSentimentAnalysis(messages);
        
        // 3. Intent detection using NLP
        const intentAnalysis = this.detectPropertyIntent(messages);
        
        // 4. Conversation structure analysis
        const structureAnalysis = this.analyzeConversationStructure(messages);
        
        // 5. Advanced AI analysis (if Hugging Face is available)
        let aiAnalysis = {};
        if (this.hf) {
            aiAnalysis = await this.performAdvancedAIAnalysis(messages);
        }
        
        // 6. Calculate lead score
        const leadScore = this.calculateLeadScore(
            keywordAnalysis, 
            sentimentAnalysis, 
            intentAnalysis, 
            structureAnalysis,
            aiAnalysis
        );
        
        // 7. Extract property details
        const extractedDetails = this.extractPropertyDetails(messages);
        
        return {
            keywordAnalysis,
            sentimentAnalysis,
            intentAnalysis,
            structureAnalysis,
            aiAnalysis,
            extractedDetails,
            leadScore,
            confidence: this.calculateConfidence(
                keywordAnalysis,
                sentimentAnalysis,
                structureAnalysis
            ),
            messagesAnalyzed: messages.length,
            lastMessageDate: messages[0] ? 
                moment(messages[0].timestamp * 1000).format('YYYY-MM-DD') : null
        };
    }
    
    /**
     * Keyword-based analysis
     */
    performKeywordAnalysis(messages) {
        const results = {
            propertyTypes: new Set(),
            actions: new Set(),
            details: new Set(),
            prices: [],
            confirmations: 0,
            keywordMatches: 0
        };
        
        const allText = messages.map(m => m.body.toLowerCase()).join(' ');
        
        // Analyze in both languages
        ['english', 'arabic'].forEach(lang => {
            const keywords = this.propertyKeywords[lang];
            
            // Check property types
            keywords.propertyTypes.forEach(type => {
                if (allText.includes(type.toLowerCase())) {
                    results.propertyTypes.add(`${type} (${lang})`);
                    results.keywordMatches++;
                }
            });
            
            // Check actions
            keywords.actions.forEach(action => {
                if (allText.includes(action.toLowerCase())) {
                    results.actions.add(`${action} (${lang})`);
                    results.keywordMatches++;
                }
            });
            
            // Check details
            keywords.details.forEach(detail => {
                if (allText.includes(detail.toLowerCase())) {
                    results.details.add(`${detail} (${lang})`);
                    results.keywordMatches++;
                }
            });
            
            // Extract prices
            const priceRegex = /(\d+[,\d]*\.?\d*)\s*(aed|dhs|درهم|دولار|dollar|usd)/gi;
            const priceMatches = allText.match(priceRegex);
            if (priceMatches) {
                results.prices.push(...priceMatches);
                results.keywordMatches += priceMatches.length;
            }
            
            // Check confirmations
            keywords.confirmation.forEach(conf => {
                const regex = new RegExp(`\\b${conf}\\b`, 'gi');
                if (regex.test(allText)) {
                    results.confirmations++;
                    results.keywordMatches++;
                }
            });
        });
        
        return results;
    }
    
    /**
     * Sentiment analysis
     */
    performSentimentAnalysis(messages) {
        const sentiments = messages.map(msg => {
            try {
                // Basic sentiment analysis
                const score = this.sentiment.getSentiment(
                    this.tokenizer.tokenize(msg.body || '')
                );
                
                return {
                    text: msg.body.substring(0, 100),
                    score,
                    sentiment: score > 0 ? 'positive' : score < 0 ? 'negative' : 'neutral',
                    fromMe: msg.fromMe
                };
            } catch (e) {
                return { score: 0, sentiment: 'neutral', fromMe: msg.fromMe };
            }
        });
        
        const averageScore = sentiments.reduce((sum, s) => sum + s.score, 0) / sentiments.length;
        
        return {
            averageScore,
            positiveCount: sentiments.filter(s => s.score > 0).length,
            negativeCount: sentiments.filter(s => s.score < 0).length,
            neutralCount: sentiments.filter(s => s.score === 0).length,
            interactions: sentiments.filter(s => !s.fromMe).length,
            details: sentiments.slice(0, 5) // Top 5 for details
        };
    }
    
    /**
     * Detect property-related intent
     */
    detectPropertyIntent(messages) {
        const intents = [];
        const intentKeywords = ['looking for', 'need', 'want', 'require', 'searching', 
                               'أبحث عن', 'بحاجة', 'أريد', 'أحتاج', 'أبحث'];
        
        messages.forEach(msg => {
            if (msg.fromMe) return; // Only analyze user messages
            
            const text = msg.body.toLowerCase();
            let detectedIntent = null;
            
            // Check for property intent
            if (this.containsPropertyKeywords(text)) {
                detectedIntent = {
                    type: 'property_inquiry',
                    confidence: 0.8,
                    text: msg.body.substring(0, 150),
                    timestamp: moment(msg.timestamp * 1000).format('YYYY-MM-DD')
                };
            }
            
            // Check for urgency
            if (this.containsUrgencyKeywords(text)) {
                detectedIntent = {
                    type: 'urgent_inquiry',
                    confidence: 0.9,
                    text: msg.body.substring(0, 150),
                    timestamp: moment(msg.timestamp * 1000).format('YYYY-MM-DD')
                };
            }
            
            if (detectedIntent) {
                intents.push(detectedIntent);
            }
        });
        
        return {
            intents,
            hasPropertyIntent: intents.some(i => i.type === 'property_inquiry'),
            hasUrgentIntent: intents.some(i => i.type === 'urgent_inquiry'),
            totalIntents: intents.length
        };
    }
    
    /**
     * Analyze conversation structure
     */
    analyzeConversationStructure(messages) {
        if (messages.length < 3) {
            return { isValidStructure: false, reason: 'Too few messages' };
        }
        
        // Group messages by day
        const messagesByDay = {};
        messages.forEach(msg => {
            const day = moment(msg.timestamp * 1000).format('YYYY-MM-DD');
            if (!messagesByDay[day]) {
                messagesByDay[day] = [];
            }
            messagesByDay[day].push({
                fromMe: msg.fromMe,
                timestamp: msg.timestamp,
                bodyLength: msg.body ? msg.body.length : 0
            });
        });
        
        // Calculate metrics
        const daysWithMessages = Object.keys(messagesByDay).length;
        const averageMessagesPerDay = messages.length / daysWithMessages;
        const ourMessages = messages.filter(m => m.fromMe).length;
        const theirMessages = messages.filter(m => !m.fromMe).length;
        const responseRatio = theirMessages > 0 ? ourMessages / theirMessages : 0;
        
        // Check for meaningful exchange pattern
        let hasQuestionAnswerPattern = false;
        for (let i = 0; i < messages.length - 1; i++) {
            if (messages[i].fromMe && !messages[i + 1].fromMe) {
                // We asked, they responded
                hasQuestionAnswerPattern = true;
                break;
            }
        }
        
        return {
            isValidStructure: daysWithMessages > 1 && hasQuestionAnswerPattern,
            daysWithMessages,
            averageMessagesPerDay: averageMessagesPerDay.toFixed(2),
            ourMessages,
            theirMessages,
            responseRatio: responseRatio.toFixed(2),
            hasQuestionAnswerPattern,
            conversationDurationDays: this.calculateConversationDuration(messages)
        };
    }
    
    /**
     * Advanced AI analysis using Hugging Face
     */
    async performAdvancedAIAnalysis(messages) {
        if (!this.hf) return {};
        
        try {
            // Combine recent messages for analysis
            const recentText = messages
                .slice(0, 10)
                .map(m => m.body)
                .filter(Boolean)
                .join(' ')
                .substring(0, 1000);
            
            if (!recentText || recentText.length < 50) {
                return {};
            }
            
            // Use zero-shot classification for property intent
            const classification = await this.hf.zeroShotClassification({
                model: 'facebook/bart-large-mnli',
                inputs: [recentText],
                parameters: {
                    candidate_labels: [
                        'property inquiry',
                        'rental request',
                        'property sale',
                        'general conversation',
                        'customer support',
                        'business inquiry'
                    ]
                }
            });
            
            // Extract entities (if we have a NER model)
            let entities = [];
            try {
                const nerResult = await this.hf.tokenClassification({
                    model: 'dslim/bert-base-NER',
                    inputs: recentText.substring(0, 512)
                });
                entities = nerResult;
            } catch (nerError) {
                console.log('NER analysis skipped:', nerError.message);
            }
            
            return {
                classification: classification[0],
                entities,
                textAnalyzed: recentText.length,
                modelUsed: 'facebook/bart-large-mnli'
            };
            
        } catch (error) {
            console.warn('Advanced AI analysis failed:', error.message);
            return {};
        }
    }
    
    /**
     * Calculate lead score
     */
    calculateLeadScore(keywordAnalysis, sentimentAnalysis, intentAnalysis, structureAnalysis, aiAnalysis) {
        let score = 0;
        
        // Score based on keyword matches
        if (keywordAnalysis.keywordMatches > 10) score += 30;
        else if (keywordAnalysis.keywordMatches > 5) score += 20;
        else if (keywordAnalysis.keywordMatches > 2) score += 10;
        
        // Score based on property types found
        if (keywordAnalysis.propertyTypes.size > 0) score += 15;
        
        // Score based on prices found
        if (keywordAnalysis.prices.length > 0) score += 20;
        
        // Score based on confirmations
        if (keywordAnalysis.confirmations > 0) score += 25;
        
        // Score based on sentiment
        if (sentimentAnalysis.averageScore > 0.5) score += 10;
        
        // Score based on intent
        if (intentAnalysis.hasPropertyIntent) score += 20;
        if (intentAnalysis.hasUrgentIntent) score += 15;
        
        // Score based on conversation structure
        if (structureAnalysis.isValidStructure) {
            score += 15;
            if (structureAnalysis.daysWithMessages > 3) score += 5;
            if (structureAnalysis.conversationDurationDays > 30) score += 10;
        }
        
        // Score based on AI analysis
        if (aiAnalysis.classification) {
            const propertyScore = aiAnalysis.classification.scores[0];
            if (propertyScore > 0.7) score += 30;
            else if (propertyScore > 0.5) score += 20;
            else if (propertyScore > 0.3) score += 10;
        }
        
        return Math.min(100, score);
    }
    
    /**
     * Calculate confidence level
     */
    calculateConfidence(keywordAnalysis, sentimentAnalysis, structureAnalysis) {
        let confidence = 0;
        
        // High confidence if many keyword matches
        if (keywordAnalysis.keywordMatches > 8) confidence += 40;
        else if (keywordAnalysis.keywordMatches > 4) confidence += 25;
        
        // High confidence if positive sentiment
        if (sentimentAnalysis.averageScore > 0.3) confidence += 20;
        
        // High confidence if structured conversation
        if (structureAnalysis.isValidStructure && structureAnalysis.daysWithMessages > 2) {
            confidence += 30;
        }
        
        // Additional confidence for price mentions
        if (keywordAnalysis.prices.length > 0) confidence += 10;
        
        return Math.min(100, confidence);
    }
    
    /**
     * Extract property details from conversation
     */
    extractPropertyDetails(messages) {
        const details = {
            propertyType: null,
            budget: null,
            location: null,
            bedrooms: null,
            timeframe: null,
            requirements: []
        };
        
        const allText = messages.map(m => m.body).join(' ').toLowerCase();
        
        // Extract property type
        for (const [lang, keywords] of Object.entries(this.propertyKeywords)) {
            for (const type of keywords.propertyTypes) {
                if (allText.includes(type.toLowerCase())) {
                    details.propertyType = `${type} (${lang})`;
                    break;
                }
            }
            if (details.propertyType) break;
        }
        
        // Extract budget
        const budgetRegex = /(\d+[,\d]*\.?\d*)\s*(aed|dhs|درهم|دولار|dollar|usd)/gi;
        const budgetMatches = allText.match(budgetRegex);
        if (budgetMatches && budgetMatches.length > 0) {
            details.budget = budgetMatches[0];
        }
        
        // Extract bedrooms
        const bedroomRegex = /(\d+)\s*(bedroom|bed|br|غرفة|غرف)/gi;
        const bedroomMatch = bedroomRegex.exec(allText);
        if (bedroomMatch) {
            details.bedrooms = bedroomMatch[1];
        }
        
        // Extract timeframe
        const timeframeKeywords = ['immediately', 'urgent', 'asap', 'next week', 'this month',
                                  'فورا', 'عاجل', 'الاسبوع القادم', 'هذا الشهر'];
        for (const keyword of timeframeKeywords) {
            if (allText.includes(keyword)) {
                details.timeframe = keyword;
                break;
            }
        }
        
        // Extract requirements
        const requirementKeywords = ['furnished', 'parking', 'balcony', 'pool', 'gym',
                                    'مفروش', 'موقف سيارات', 'بلكونة', 'مسبح', 'جيم'];
        details.requirements = requirementKeywords.filter(keyword => 
            allText.includes(keyword)
        );
        
        return details;
    }
    
    /**
     * Evaluate if contact is a verified lead
     */
    evaluateAsVerifiedLead(analysis) {
        const { leadScore, confidence, keywordAnalysis, structureAnalysis } = analysis;
        
        // Must meet minimum threshold
        if (leadScore < this.leadThreshold) {
            return false;
        }
        
        // Must have property keywords
        if (keywordAnalysis.propertyTypes.size === 0 && keywordAnalysis.actions.size === 0) {
            return false;
        }
        
        // Must have meaningful conversation structure
        if (!structureAnalysis.isValidStructure) {
            return false;
        }
        
        // Must have at least one confirmation
        if (keywordAnalysis.confirmations === 0) {
            return false;
        }
        
        return true;
    }
    
    /**
     * Helper methods
     */
    containsPropertyKeywords(text) {
        const lowerText = text.toLowerCase();
        return Object.values(this.propertyKeywords).some(langKeywords => 
            langKeywords.propertyTypes.some(type => lowerText.includes(type)) ||
            langKeywords.actions.some(action => lowerText.includes(action))
        );
    }
    
    containsUrgencyKeywords(text) {
        const urgencyWords = ['urgent', 'immediate', 'asap', 'quick', 'soon', 'عاجل', 'فوري', 'سريع'];
        return urgencyWords.some(word => text.toLowerCase().includes(word));
    }
    
    calculateConversationDuration(messages) {
        if (messages.length < 2) return 0;
        
        const firstMsg = messages[messages.length - 1];
        const lastMsg = messages[0];
        
        const firstDate = moment(firstMsg.timestamp * 1000);
        const lastDate = moment(lastMsg.timestamp * 1000);
        
        return lastDate.diff(firstDate, 'days');
    }
    
    createEmptyAnalysis(contactId, reason) {
        return {
            contactId,
            isVerifiedLead: false,
            analysis: {
                keywordAnalysis: { keywordMatches: 0 },
                leadScore: 0,
                confidence: 0,
                reason
            },
            lastAnalyzed: new Date().toISOString()
        };
    }
    
    /**
     * Batch analyze multiple contacts
     */
    async batchAnalyzeContacts(contactIds, batchSize = 10) {
        const results = {
            verifiedLeads: [],
            potentialLeads: [],
            notLeads: [],
            errors: []
        };
        
        console.log(`🤖 Batch analyzing ${contactIds.length} contacts for property leads`);
        
        for (let i = 0; i < contactIds.length; i += batchSize) {
            const batch = contactIds.slice(i, i + batchSize);
            
            console.log(`Processing batch ${Math.floor(i/batchSize) + 1}: ${batch.length} contacts`);
            
            const batchPromises = batch.map(async (contactId, index) => {
                try {
                    await this.delay(index * 1000); // Stagger requests
                    
                    const analysis = await this.analyzeConversationForLeads(contactId);
                    
                    return {
                        contactId,
                        analysis,
                        batchIndex: i + index
                    };
                } catch (error) {
                    console.error(`Batch analysis failed for ${contactId}:`, error.message);
                    return {
                        contactId,
                        error: error.message,
                        batchIndex: i + index
                    };
                }
            });
            
            const batchResults = await Promise.all(batchPromises);
            
            // Categorize results
            batchResults.forEach(result => {
                if (result.error) {
                    results.errors.push({
                        contactId: result.contactId,
                        error: result.error
                    });
                } else if (result.analysis.isVerifiedLead) {
                    results.verifiedLeads.push({
                        contactId: result.contactId,
                        score: result.analysis.leadScore,
                        confidence: result.analysis.confidence,
                        details: result.analysis.analysis.extractedDetails
                    });
                } else if (result.analysis.leadScore > 30) {
                    results.potentialLeads.push({
                        contactId: result.contactId,
                        score: result.analysis.leadScore,
                        reason: result.analysis.analysis.reason
                    });
                } else {
                    results.notLeads.push(result.contactId);
                }
            });
            
            // Delay between batches
            if (i + batchSize < contactIds.length) {
                await this.delay(5000);
            }
        }
        
        console.log(`Batch analysis complete:`);
        console.log(`✅ Verified Leads: ${results.verifiedLeads.length}`);
        console.log(`⚠️  Potential Leads: ${results.potentialLeads.length}`);
        console.log(`❌ Not Leads: ${results.notLeads.length}`);
        console.log(`🚨 Errors: ${results.errors.length}`);
        
        return results;
    }
    
    /**
     * Get analysis statistics
     */
    getAnalysisStats() {
        const stats = {
            totalAnalyzed: this.analyzedChats.size,
            verifiedLeads: 0,
            averageScore: 0,
            byScoreRange: {
                '80-100': 0,
                '60-79': 0,
                '40-59': 0,
                '20-39': 0,
                '0-19': 0
            }
        };
        
        let totalScore = 0;
        
        for (const [_, data] of this.analyzedChats) {
            const score = data.result.leadScore || 0;
            totalScore += score;
            
            if (data.result.isVerifiedLead) {
                stats.verifiedLeads++;
            }
            
            if (score >= 80) stats.byScoreRange['80-100']++;
            else if (score >= 60) stats.byScoreRange['60-79']++;
            else if (score >= 40) stats.byScoreRange['40-59']++;
            else if (score >= 20) stats.byScoreRange['20-39']++;
            else stats.byScoreRange['0-19']++;
        }
        
        stats.averageScore = this.analyzedChats.size > 0 ? 
            (totalScore / this.analyzedChats.size).toFixed(2) : 0;
        
        return stats;
    }
    
    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

module.exports = AIConversationAnalyzer;

const { GoogleSpreadsheet } = require('google-spreadsheet');

class VerifiedLeadManager {

    calculateCompositeAIScore(contact) {
        const weights = {

            leadPotential: 0.3,
            patternSuitability: 0.2,

        // Engagement score
        if (contact.aiScores?.engagement) {
            score += contact.aiScores.engagement * weights.engagement;

        // Lead potential
        if (contact.aiScores?.leadPotential) {
            score += (contact.aiScores.leadPotential / 100) * weights.leadPotential;
        }
        
        // Pattern suitability
        if (contact.aiScores?.patternSuitability) {
            score += contact.aiScores.patternSuitability * weights.patternSuitability;

        // Recency bonus (if recently contacted)
        if (contact.lastContacted) {
            const daysSince = moment().diff(moment(contact.lastContacted), 'days');
            if (daysSince > 60) {
                score += weights.recency; // Higher score for not recently contacted

            score += weights.recency; // Never contacted

                contact,
                campaignConfig,
                results
            );
            
            try {
                // Send message
                const sendResult = await this.client.sendMessage(
                    contact.whatsappId || contact.validatedNumber,
                    personalizedMessage
                );
                
                results.sent++;
                successStreak++;
                failureStreak = 0;
                
                // Record successful send
                results.details.push({
                    contact: contact.validatedNumber,
                    success: true,
                    timestamp: new Date().toISOString(),
                    aiScore: contact.aiScores?.composite || 0,
                    delayUsed: adaptiveDelay,
                    messagePreview: personalizedMessage.substring(0, 100)
                });
                
                console.log(`✅ AI-sent to ${contact.name || contact.validatedNumber}`);
                
                // Check for immediate response (within 60 seconds)
                setTimeout(async () => {
                    const immediateResponse = await this.checkImmediateResponse(
                        contact.validatedNumber
                    );
                    
                    if (immediateResponse) {
                        results.responses.push({
                            contact: contact.validatedNumber,
                            response: immediateResponse.body.substring(0, 200),
                            timestamp: new Date().toISOString(),
                            responseTime: (Date.now() - sendResult.timestamp) / 1000
                        });
                        
                        // Analyze response for lead potential
                        const responseAnalysis = await this.analyzeResponseForLead(
                            immediateResponse.body,
                            contact
                        );
                        
                        if (responseAnalysis.isPotentialLead) {
                            results.leadsIdentified++;
                            
                            // Mark as potential lead
                            await this.markAsPotentialLead(
                                contact,
                                responseAnalysis,
                                'Immediate campaign response'
                            );
                        }
                    }
                }, 60000); // Check after 60 seconds
                
                // Adaptive success streak handling
                if (successStreak >= 5) {
                    results.aiAdjustments.push({
                        action: 'slow_down',
                        reason: 'Success streak detected',
                        streak: successStreak,
                        timestamp: new Date().toISOString()
                    });
                    
                    adaptiveDelay += 2000; // Add 2 seconds
                    successStreak = 0;
                }
                
            } catch (error) {
                results.failed++;
                failureStreak++;
                successStreak = 0;
                
                results.details.push({
                    contact: contact.validatedNumber,
                    success: false,
                    error: error.message,
                    timestamp: new Date().toISOString(),
                    aiScore: contact.aiScores?.composite || 0
                });
                
                console.error(`❌ AI-send failed for ${contact.validatedNumber}:`, error.message);
                
                // Adaptive failure handling
                if (failureStreak >= 3) {
                    results.aiAdjustments.push({
                        action: 'pause',
                        reason: 'Failure streak detected',
                        streak: failureStreak,
                        timestamp: new Date().toISOString()
                    });
                    
                    await this.delay(30000); // 30-second pause
                    failureStreak = 0;
                }
                
                // Learn from failure
                this.learnFromFailure(contact, error, campaignConfig);
            }
        }
        
        results.endTime = new Date().toISOString();
        results.duration = moment(results.endTime).diff(moment(results.startTime), 'minutes');
        
        return results;
    }

AICampaignOrchestrator.prototype.containsPropertyKeywords = function(text) {
    const propertyWords = ['apartment', 'villa', 'house', 'rent', 'sale', 'property',
                          'شقة', 'فيلا', 'منزل', 'إيجار', 'بيع', 'عقار'];
    return propertyWords.some(word => text.includes(word));
};

AICampaignOrchestrator.prototype.containsPositiveResponse = function(text) {
    const positiveWords = ['yes', 'interested', 'available', 'please', 'ok', 'good',
                          'نعم', 'مهتم', 'متوفر', 'من فضلك', 'حسنا', 'جيد'];
    return positiveWords.some(word => text.includes(word));
};

AICampaignOrchestrator.prototype.containsPropertyDetails = function(text) {
    const detailWords = ['bedroom', 'bathroom', 'price', 'location', 'area',
                        'غرفة', 'حمام', 'سعر', 'موقع', 'منطقة'];
    return detailWords.some(word => text.includes(word));
};

AICampaignOrchestrator.prototype.extractPropertyType = function(text) {
    const types = [
        { keywords: ['apartment', 'flat', 'studio', 'شقة', 'استوديو'], type: 'Apartment' },
        { keywords: ['villa', 'house', 'منزل', 'فيلا'], type: 'Villa' },
        { keywords: ['commercial', 'office', 'shop', 'تجاري', 'مكتب'], type: 'Commercial' },
        { keywords: ['land', 'plot', 'أرض', 'قطعة'], type: 'Land' }
    ];
    
    for (const { keywords, type } of types) {
        if (keywords.some(keyword => text.includes(keyword))) {
            return type;
        }
    }
    
    return 'Unknown';
};

AICampaignOrchestrator.prototype.extractUrgency = function(text) {
    const urgencyWords = ['urgent', 'immediate', 'asap', 'soon', 'quick', 'عاجل', 'فوري', 'سريع'];
    
    if (urgencyWords.some(word => text.includes(word))) {
        return 'High';
    }
    
    const moderateWords = ['next week', 'this month', 'الاسبوع القادم', 'هذا الشهر'];
    if (moderateWords.some(word => text.includes(word))) {
        return 'Medium';
    }
    
    return 'Low';
};

        this.setupWebSocket();
        
        // Initialize
        this.initialize();
    }
    
    async initialize() {
        try {
            // Initialize WhatsApp client
            await this.whatsappClient.initialize();
            console.log('✅ WhatsApp client initialized');
            
            // Load existing leads
            await this.leadManager.loadVerifiedLeads();
            console.log('✅ Verified leads loaded');
            
            // Generate initial dashboard
            await this.dashboard.generateDashboard();
            console.log('✅ AI dashboard initialized');
            
            // Start server
            this.startServer();
            
        } catch (error) {
            console.error('Failed to initialize AI system:', error);
            process.exit(1);
        }
    }
    
    setupMiddleware() {
        this.app.use(helmet());
        this.app.use(compression());
        this.app.use(express.json());
        
        // Rate limiting
        const limiter = rateLimit({
            windowMs: 15 * 60 * 1000,
            max: 100
        });
        this.app.use('/api/', limiter);
    }
    
    setupRoutes() {
        // Health check
        this.app.get('/health', (req, res) => {
            res.json({
                status: 'ok',
                timestamp: new Date().toISOString(),
                ai: {
                    version: '2.1.0',
                    status: 'active',
                    leadsAnalyzed: this.leadManager.getAllLeads().length
                }
            });
        });
        
        // Dashboard
        this.app.get('/api/dashboard', async (req, res) => {
            try {
                const dashboardData = await this.dashboard.generateDashboard();
                res.json(dashboardData);
            } catch (error) {
                res.status(500).json({ error: error.message });
            }
        });
        
        // Start AI campaign
        this.app.post('/api/campaigns/ai', async (req, res) => {
            try {
                const result = await this.aiOrchestrator.executeIntelligentCampaign(
                    req.body
                );
                res.json(result);
            } catch (error) {
                res.status(500).json({ error: error.message });
            }
        });
        
        // Analyze conversations for leads
        this.app.post('/api/ai/analyze', async (req, res) => {
            try {
                const { contactIds } = req.body;
                
                if (!contactIds || !Array.isArray(contactIds)) {
                    return res.status(400).json({ error: 'contactIds array required' });
                }
                
                const results = await this.aiOrchestrator.aiAnalyzer.batchAnalyzeContacts(
                    contactIds
                );
                
                res.json(results);
            } catch (error) {
                res.status(500).json({ error: error.message });
            }
        });
        
        // Get verified leads
        this.app.get('/api/leads', (req, res) => {
            const filter = req.query;
            const leads = this.leadManager.getAllLeads(filter);
            
            res.json({
                total: leads.length,
                leads,
                stats: this.leadManager.getLeadStats()
            });
        });
        
        // Get follow-ups
        this.app.get('/api/followups', (req, res) => {
            const today = this.leadManager.getTodaysFollowUps();
            const upcoming = this.leadManager.getUpcomingFollowUps(7);
            
            res.json({
                today,
                upcoming,
                priority: today.filter(fu => fu.category === 'HOT')
            });
        });
        
        // Update follow-up status
        this.app.post('/api/followups/:leadId', async (req, res) => {
            try {
                const { leadId } = req.params;
                const { followUpIndex, status, notes } = req.body;
                
                const result = await this.leadManager.updateFollowUpStatus(
                    leadId,
                    followUpIndex,
                    status,
                    notes
                );
                
                res.json(result);
            } catch (error) {
                res.status(500).json({ error: error.message });
            }
        });
        
        // AI insights endpoint
        this.app.get('/api/ai/insights', async (req, res) => {
            try {
                const dashboardData = await this.dashboard.generateDashboard();
                res.json({
                    insights: dashboardData.insights,
                    predictions: dashboardData.predictions,
                    recommendations: dashboardData.recommendations
                });
            } catch (error) {
                res.status(500).json({ error: error.message });
            }
        });
    }
    
    setupWebSocket() {
        this.dashboard.setupWebSocket(this.io);
        
        // Additional WebSocket events
        this.io.on('connection', (socket) => {
            console.log('Client connected to AI system');
            
            // Send real-time updates
            const updateInterval = setInterval(() => {
                socket.emit('ai-update', {
                    timestamp: new Date().toISOString(),
                    leads: this.leadManager.getLeadStats(),
                    followUps: this.leadManager.getTodaysFollowUps().length
                });
            }, 10000);
            
            socket.on('disconnect', () => {
                clearInterval(updateInterval);
                console.log('Client disconnected');
            });
        });
    }
    
    startServer() {
        const PORT = process.env.PORT || 3000;
        
        this.server.listen(PORT, '127.0.0.1', () => {

    }
