export function sanitizePhoneNumber(phoneNumber) {
  if (!phoneNumber) return '';
  
  // Remove all non-digits
  const digitsOnly = String(phoneNumber).replace(/\D/g, '');
  
  // Handle empty or too short numbers
  if (digitsOnly.length < 5) return '';
  
  return digitsOnly;
}

export function formatToInternational(number, defaultCountry = 'AE') {
  try {
    const phoneNumber = parsePhoneNumberFromString(number, defaultCountry);
    
    if (phoneNumber && phoneNumber.isValid()) {
      return phoneNumber.format('E.164').replace('+', '');
    }
    
    // Fallback logic for UAE numbers
    const sanitized = sanitizePhoneNumber(number);
    
    // If number starts with 971 and is correct length
    if (sanitized.startsWith('971') && sanitized.length === 12) {
      return sanitized;
    }
    
    // If number is 9 digits (UAE without country code)
    if (sanitized.length === 9 && !sanitized.startsWith('0')) {
      return `971${sanitized}`;
    }
    
    // If number is 10 digits with leading 0
    if (sanitized.length === 10 && sanitized.startsWith('0')) {
      return `971${sanitized.slice(1)}`;
    }
    
    return ''; // Invalid number
    
  } catch (error) {
    console.error('Phone number parsing failed:', error);
    return '';
  }
}

export function rectifyContactNumbers(data) {
  const validNumbers = new Set();
  
  for (const row of data) {
    // Check multiple possible column names
    const possibleColumns = ['MOBILE', 'MOBILE 2', 'SECONDARY MOBILE', 'Phone', 'Contact'];
    
    for (const column of possibleColumns) {
      if (row[column]) {
        const formatted = formatToInternational(row[column]);
        if (formatted) {
          validNumbers.add(`${formatted}@c.us`);
          break; // Use first valid number
        }
      }
    }

}

// Refactored with async processing:
import { EventEmitter } from 'events';
import { logger } from '../utils/logger.js';

export class BroadcastManager extends EventEmitter {
  constructor(client, options = {}) {
    super();
    this.client = client;
    this.concurrentLimit = options.concurrentLimit || 5;
    this.delayBetweenMessages = options.delayBetweenMessages || 3000;
    this.workingHours = options.workingHours || { start: 8, end: 19 };
  }

  async sendBroadcast(numbers, message, options = {}) {
    const {
      newChatsOnly = true,
      onProgress = () => {},
      onComplete = () => {}
    } = options;

    const results = {
      sent: 0,
      failed: 0,
      skipped: 0,
      details: []
    };

    // Process in batches
    const batches = this.chunkArray(numbers, this.concurrentLimit);

    for (const [batchIndex, batch] of batches.entries()) {
      const batchPromises = batch.map(async (number) => {
        try {
          // Check if within working hours
          if (!this.isWithinWorkingHours()) {
            logger.info('Outside working hours, pausing...');
            await this.sleepUntilWorkingHours();
          }

          const chat = await this.client.getChatById(number);
          
          if (newChatsOnly && chat.timestamp) {
            results.skipped++;
            return { number, status: 'skipped', reason: 'existing_chat' };
          }

          const sent = await this.client.sendMessage(number, message);
          results.sent++;
          
          logger.info(`Message sent to ${number}`);
          return { number, status: 'sent', messageId: sent.id.id };

        } catch (error) {
          results.failed++;
          logger.error(`Failed to send to ${number}:`, error);
          return { number, status: 'failed', error: error.message };
        }
      });

      const batchResults = await Promise.allSettled(batchPromises);
      results.details.push(...batchResults);

      // Progress callback
      onProgress({
        completed: (batchIndex + 1) * batch.length,
        total: numbers.length,
        results
      });

      // Delay between batches (except last)
      if (batchIndex < batches.length - 1) {
        await this.delay(this.delayBetweenMessages);
      }
    }

    onComplete(results);
    return results;
  }

  chunkArray(array, size) {
    const chunks = [];
    for (let i = 0; i < array.length; i += size) {
      chunks.push(array.slice(i, i + size));
    }
    return chunks;
  }

  isWithinWorkingHours() {
    const hour = new Date().getHours();
    return hour >= this.workingHours.start && hour <= this.workingHours.end;
  }

  async sleepUntilWorkingHours() {
    const now = new Date();
    const tomorrow = new Date(now);
    tomorrow.setDate(tomorrow.getDate() + 1);
    tomorrow.setHours(this.workingHours.start, 0, 0, 0);

    const delay = tomorrow - now;
    logger.info(`Sleeping for ${this.formatMsToTime(delay)} until working hours`);
    
    await this.delay(delay);
  }

  delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

      ? this.delayMs + Math.random() * this.delayMs * 0.5

    const now = new Date();

    await this.delay(waitTime);
  }
