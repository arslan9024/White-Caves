2. Advanced Dubai Property Search (Core Query)ts

async function searchDubaiProperties(filters: {
  minPrice?: number;
  maxPrice?: number;
  bedrooms?: number;
  community?: string[];
  propertyType?: string[];
  minSize?: number;
}): Promise<DubaiProperty[]> {
  return prisma.dubaiProperty.findMany({
    where: {
      priceAED: {
        gte: filters.minPrice,
        lte: filters.maxPrice,
      },
      bedrooms: filters.bedrooms,
      community: { in: filters.community },
      type: { in: filters.propertyType },
      sizeSqFt: { gte: filters.minSize },
      status: 'available',
    },
    orderBy: { priceAED: 'asc' },
    take: 24,
  });
}

3. AML Quick Risk Score (Very simplified version)ts

function quickAMLRiskScore(tx: {
  amountAED: number;
  paymentMethod: string;
  clientNationality: string;
  isPEP: boolean;
  propertyPriceDeviation: number; // % from market value
}): number {
  let score = 0;

  if (tx.amountAED > 2_000_000) score += 30;
  if (tx.paymentMethod === 'cash') score += 40;
  if (['Russia', 'Iran', 'Syria'].includes(tx.clientNationality)) score += 35;
  if (tx.isPEP) score += 25;
  if (Math.abs(tx.propertyPriceDeviation) > 30) score += 20;

  return Math.min(100, score); // 0–100 risk score
}

4. UAE PASS + Role Check (Express middleware)ts

const requireVerifiedClient = async (req: Request, res: Response, next: NextFunction) => {
  const token = req.headers['x-uaepass-token'] as string;

  if (!token) return res.status(401).json({ error: 'UAE PASS token required' });

  try {
    const payload = await verifyUaePassToken(token); // your verification logic
    if (payload.accessTier !== 'verified_client') {
      return res.status(403).json({ error: 'Verified client access required' });
    }
    req.user = payload;
    next();
  } catch (err) {
    res.status(401).json({ error: 'Invalid UAE PASS token' });
  }
};

7. Very Compact Property Status Updatets

async function updatePropertyStatus(id: string, newStatus: string) {
  await prisma.dubaiProperty.update({
    where: { id },
    data: {
      status: newStatus,
      updatedAt: new Date(),
      ...(newStatus === 'sold' && { soldDate: new Date() }),
      ...(newStatus === 'reserved' && { reservedUntil: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000) })
    }
  });
}
