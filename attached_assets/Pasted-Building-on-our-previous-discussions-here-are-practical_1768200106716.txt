```javascript
const moment = require('moment-timezone'); // Use moment-timezone for robust handling

class LocalTimeScheduler {
    constructor() {
        // Map country codes to primary timezones (this is a simplified example)
        this.countryToTimezone = {
            '1': 'America/New_York', // US
            '44': 'Europe/London',    // UK
            '91': 'Asia/Kolkata',     // India
            '86': 'Asia/Shanghai',    // China
            '33': 'Europe/Paris',     // France
            '49': 'Europe/Berlin',    // Germany
            '81': 'Asia/Tokyo',       // Japan
            // ... expand as needed
        };
    }

    // Determine the best send window for a contact's local time
    getOptimalSendWindow(countryCode) {
        const timezone = this.countryToTimezone[countryCode] || 'UTC';
        
        // Define "business hours" in the local timezone (e.g., 9 AM to 5 PM)
        return {
            timezone: timezone,
            startHourLocal: 9,
            endHourLocal: 17,
            // Avoid lunch hours, etc.
            excludedHours: [{ start: 12, end: 13 }] // Local lunch break
        };
    }

    // Check if "now" is a good time to send in the contact's locale
    isGoodTimeToSend(countryCode) {
        const window = this.getOptimalSendWindow(countryCode);
        const nowLocal = moment().tz(window.timezone);
        const currentHour = nowLocal.hour();
        const currentDay = nowLocal.day(); // 0 = Sunday, 1 = Monday, etc.

        // Don't send on weekends
        if (currentDay === 0 || currentDay === 6) {
            return { allowed: false, reason: 'Weekend in target locale' };
        }

        // Check if within business hours
        if (currentHour < window.startHourLocal || currentHour >= window.endHourLocal) {
            return { allowed: false, reason: 'Outside business hours in target locale' };
        }

        // Check excluded hours (e.g., lunch)
        for (const block of window.excludedHours) {
            if (currentHour >= block.start && currentHour < block.end) {
                return { allowed: false, reason: 'Local lunch break' };
            }
        }

        return { allowed: true, nextAvailable: null };
    }

    // Calculate delay until the next good send window
    calculateDelayUntilWindow(countryCode) {
        const window = this.getOptimalSendWindow(countryCode);
        const nowLocal = moment().tz(window.timezone);
        
        // If currently in a bad period, calculate when the next good period starts
        // (Simplified logic - you would expand this)
        if (nowLocal.hour() < window.startHourLocal) {
            const nextStart = moment().tz(window.timezone).startOf('day').hour(window.startHourLocal);
            return nextStart.diff(nowLocal, 'milliseconds');
        }
        // ... handle other cases (after hours, weekend)
        
        return 0; // Good to send now
    }
}
```

```javascript
// enhanced-optout-detector.js
class EnhancedOptOutDetector {
    constructor(client) {
        this.client = client;
        this.optOutKeywords = [
            'stop', 'unsubscribe', 'remove', 'do not send', 'opt-out',
            'opt out', 'don\'t message', 'no messages', 'block',
            'خذني من القائمة', // Arabic: remove me from list
            'أوقف الرسائل', // Arabic: stop messages
            'إلغاء الاشتراك' // Arabic: unsubscribe
        ];
        
        this.blockDetectionKeywords = [
            'blocked', 'block you', 'blocked you', 'حظرتك', // Arabic: I blocked you
            'مش حابب', // Arabic: don't like
            'مزعج' // Arabic: annoying
        ];
        
        // WhatsApp specific error codes for blocks
        this.whatsappBlockErrors = {
            '131021': 'Recipient blocked sender',
            '131026': 'Recipient not on WhatsApp',
            '131045': 'Recipient deactivated account',
            '131050': 'User opted out of marketing messages'
        };
        
        // Rate limiting for chat history checks
        this.chatCheckHistory = new Map();
        this.maxChecksPerMonth = 5;
        
        // Cache for opt-out status
        this.optOutCache = new Map();
        this.cacheDuration = 7 * 24 * 60 * 60 * 1000; // 7 days
    }
    
    async checkContactStatus(contactId, forceCheck = false) {
        const cacheKey = contactId;
        const cached = this.optOutCache.get(cacheKey);
        
        // Return cached result if still valid
        if (cached && (Date.now() - cached.timestamp) < this.cacheDuration && !forceCheck) {
            return cached.status;
        }
        
        // Check rate limiting
        if (!this.canCheckChatHistory(contactId)) {
            console.log(`Rate limited: Skipping chat check for ${contactId}`);
            return {
                canMessage: true, // Default to allowing when rate limited
                reason: 'Rate limited - assuming safe',
                lastChecked: new Date().toISOString(),
                requiresManualReview: true
            };
        }
        
        const status = {
            canMessage: true,
            reason: 'No issues detected',
            lastChecked: new Date().toISOString(),
            details: {},
            requiresManualReview: false
        };
        
        try {
            // 1. Check if we can get chat (indicates not blocked)
            const chat = await this.client.getChatById(contactId);
            
            if (!chat) {
                status.canMessage = false;
                status.reason = 'Cannot access chat - may be blocked';
                status.details.chatAccess = 'failed';
                this.recordChatCheck(contactId, 'failed_access');
            } else {
                status.details.chatAccess = 'success';
                this.recordChatCheck(contactId, 'success');
                
                // 2. Check last 50 messages for opt-out keywords
                const messages = await chat.fetchMessages({ limit: 50 });
                status.details.totalMessages = messages.length;
                
                let foundOptOut = false;
                let foundBlock = false;
                
                for (const msg of messages) {
                    if (msg.fromMe) {
                        // Check our own message delivery status
                        if (msg.ack === -1 || msg.ack === 0) {
                            // Message not delivered or sent
                            status.details.lastDeliveryStatus = 'failed';
                        }
                        continue;
                    }
                    
                    const bodyLower = msg.body.toLowerCase();
                    
                    // Check for opt-out requests
                    for (const keyword of this.optOutKeywords) {
                        if (bodyLower.includes(keyword)) {
                            foundOptOut = true;
                            status.details.optOutFound = {
                                keyword: keyword,
                                message: msg.body.substring(0, 100),
                                timestamp: msg.timestamp
                            };
                            break;
                        }
                    }
                    
                    // Check for block threats
                    for (const keyword of this.blockDetectionKeywords) {
                        if (bodyLower.includes(keyword)) {
                            foundBlock = true;
                            status.details.blockThreat = {
                                keyword: keyword,
                                message: msg.body.substring(0, 100),
                                timestamp: msg.timestamp
                            };
                            break;
                        }
                    }
                    
                    if (foundOptOut || foundBlock) break;
                }
                
                if (foundOptOut) {
                    status.canMessage = false;
                    status.reason = 'User requested opt-out in chat history';
                }
                
                if (foundBlock) {
                    status.canMessage = false;
                    status.reason = 'User indicated they may block us';
                    status.requiresManualReview = true;
                }
                
                // 3. Check if we've sent to this contact recently and failed
                const ourMessages = messages.filter(m => m.fromMe);
                if (ourMessages.length > 0) {
                    const lastOurMessage = ourMessages[0];
                    if (lastOurMessage.ack === -1) {
                        // Message not delivered
                        status.details.lastMessageStatus = 'undelivered';
                        status.details.lastMessageTime = lastOurMessage.timestamp;
                        
                        // If multiple undelivered, likely blocked
                        const undeliveredCount = ourMessages.filter(m => m.ack === -1).length;
                        if (undeliveredCount > 2) {
                            status.canMessage = false;
                            status.reason = `Multiple messages undelivered (${undeliveredCount}) - likely blocked`;
                        }
                    }
                }
            }
            
            // 4. Try to send a test "ping" message (with user permission simulation)
            if (status.canMessage && process.env.ENABLE_PING_TEST === 'true') {
                const pingResult = await this.pingTest(contactId);
                if (!pingResult.success) {
                    status.canMessage = false;
                    status.reason = `Ping test failed: ${pingResult.error}`;
                    status.details.pingTest = pingResult;
                    
                    // Check if it's a WhatsApp block error
                    if (this.whatsappBlockErrors[pingResult.errorCode]) {
                        status.details.blockError = this.whatsappBlockErrors[pingResult.errorCode];
                    }
                }
            }
            
        } catch (error) {
            console.error(`Error checking status for ${contactId}:`, error.message);
            
            // Analyze error for block indications
            const errorMsg = error.message.toLowerCase();
            if (errorMsg.includes('not found') || 
                errorMsg.includes('blocked') || 
                errorMsg.includes('401') ||
                errorMsg.includes('failed to fetch')) {
                status.canMessage = false;
                status.reason = `Error suggests blocked: ${error.message}`;
                status.details.errorAnalysis = 'likely_blocked';
            } else {
                // Unknown error - be conservative
                status.canMessage = false;
                status.reason = `Check failed: ${error.message}`;
                status.requiresManualReview = true;
            }
            
            this.recordChatCheck(contactId, 'error');
        }
        
        // Cache the result
        this.optOutCache.set(cacheKey, {
            status: status,
            timestamp: Date.now()
        });
        
        return status;
    }
    
    async pingTest(contactId) {
        // Send a minimal, non-intrusive test message
        // This should only be done with extreme caution
        try {
            const testMessage = " "; // Empty space message
            const result = await this.client.sendMessage(contactId, testMessage);
            
            return {
                success: true,
                messageId: result.id?.id,
                timestamp: result.timestamp
            };
        } catch (error) {
            // Extract error code if present
            const errorCode = error.message.match(/\d+/)?.[0];
            
            return {
                success: false,
                error: error.message,
                errorCode: errorCode,
                timestamp: new Date().toISOString()
            };
        }
    }
    
    canCheckChatHistory(contactId) {
        const now = new Date();
        const monthKey = `${now.getFullYear()}-${now.getMonth()}`;
        const contactKey = `${contactId}-${monthKey}`;
        
        if (!this.chatCheckHistory.has(contactKey)) {
            this.chatCheckHistory.set(contactKey, {
                count: 0,
                firstCheck: Date.now(),
                lastCheck: Date.now()
            });
        }
        
        const record = this.chatCheckHistory.get(contactKey);
        
        if (record.count >= this.maxChecksPerMonth) {
            // Check if we're in a new month
            const recordMonth = new Date(record.firstCheck).getMonth();
            if (now.getMonth() !== recordMonth) {
                // Reset for new month
                record.count = 0;
                record.firstCheck = Date.now();
            } else {
                return false;
            }
        }
        
        record.count++;
        record.lastCheck = Date.now();
        
        return true;
    }
    
    recordChatCheck(contactId, result) {
        const now = new Date();
        const monthKey = `${now.getFullYear()}-${now.getMonth()}`;
        const contactKey = `${contactId}-${monthKey}`;
        
        if (!this.chatCheckHistory.has(contactKey)) {
            this.chatCheckHistory.set(contactKey, {
                count: 0,
                checks: []
            });
        }
        
        const record = this.chatCheckHistory.get(contactKey);
        record.checks.push({
            timestamp: now.toISOString(),
            result: result
        });
        
        // Keep only last 10 checks
        if (record.checks.length > 10) {
            record.checks.shift();
        }
    }
    
    getChatCheckStats(contactId) {
        const now = new Date();
        const monthKey = `${now.getFullYear()}-${now.getMonth()}`;
        const contactKey = `${contactId}-${monthKey}`;
        
        return this.chatCheckHistory.get(contactKey) || {
            count: 0,
            checks: []
        };
    }
    
    // Manual override for opt-out status
    markAsOptedOut(contactId, reason = 'Manual override') {
        const status = {
            canMessage: false,
            reason: reason,
            lastChecked: new Date().toISOString(),
            details: { manualOverride: true },
            requiresManualReview: false
        };
        
        this.optOutCache.set(contactId, {
            status: status,
            timestamp: Date.now()
        });
        
        console.log(`Manually marked ${contactId} as opted out: ${reason}`);
    }
    
    // Clear cache for a contact
    clearCache(contactId) {
        this.optOutCache.delete(contactId);
        
        // Also clear from check history
        for (const [key] of this.chatCheckHistory) {
            if (key.startsWith(contactId)) {
                this.chatCheckHistory.delete(key);
            }
        }
    }
}
```

```javascript
        }
        
        return {
            filteredContacts,
            skippedContacts,
            stats: {
                total: contacts.length,
                passedFilter: filteredContacts.length,
                skipped: skippedContacts.length,
                filterRate: (filteredContacts.length / contacts.length * 100).toFixed(1) + '%'
            }
        };
    }
    
    async canContactNow(contact, spreadsheetId, range) {
        const now = moment();
        
        // Check cache first
        const cacheKey = contact.validatedNumber;
        if (this.contactHistory.has(cacheKey)) {
            const cached = this.contactHistory.get(cacheKey);
            if (now.diff(moment(cached.timestamp), 'hours') < 1) {
                return cached.result;
            }
        }
        
        // Check if we have last contacted date in contact data
        let lastContacted = contact.lastContacted;
        
        if (!lastContacted && contact.sheetRow) {
            // Try to fetch from Google Sheets
            try {
                const sheetData = await this.sheetsService.getCellValue(
                    spreadsheetId,
                    `${range.split('!')[0]}!D${contact.sheetRow}` // Assuming column D has last contacted
                );
                lastContacted = sheetData?.value || null;
            } catch (error) {
                console.log(`Could not fetch sheet data for ${contact.validatedNumber}`);
            }
        }
        
        if (!lastContacted) {
            // Never contacted before
            const result = {
                allowed: true,
                reason: 'Never contacted before',
                lastContacted: null,
                daysSince: null
            };
            
            this.contactHistory.set(cacheKey, {
                result: result,
                timestamp: now.toISOString()
            });
            
            return result;
        }
        
        // Parse last contacted date
        const lastContactMoment = moment(lastContacted);
        const daysSince = now.diff(lastContactMoment, 'days');
        
        if (daysSince < this.minDaysBetweenContacts) {
            const result = {
                allowed: false,
                reason: `Contacted ${daysSince} days ago (minimum ${this.minDaysBetweenContacts} days required)`,
                lastContacted: lastContacted,
                daysSince: daysSince,
                nextAvailable: lastContactMoment.add(this.minDaysBetweenContacts, 'days').format('YYYY-MM-DD')
            };
            
            this.contactHistory.set(cacheKey, {
                result: result,
                timestamp: now.toISOString()
            });
            
            return result;
        }
        
        // Can contact
        const result = {
            allowed: true,
            reason: `Last contacted ${daysSince} days ago (ok)`,
            lastContacted: lastContacted,
            daysSince: daysSince
        };
        
        this.contactHistory.set(cacheKey, {
            result: result,
            timestamp: now.toISOString()
        });
        
        return result;
    }
    
    // Batch update last contacted dates
    async updateLastContacted(contacts, spreadsheetId, range) {
        const today = moment().format('YYYY-MM-DD');
        const updates = [];
        
        for (const contact of contacts) {
            if (contact.sheetRow) {
                updates.push({
                    range: `${range.split('!')[0]}!D${contact.sheetRow}`,
                    values: [[today]]
                });
                
                // Update cache
                this.contactHistory.set(contact.validatedNumber, {
                    result: {
                        allowed: false,
                        reason: `Just contacted today`,
                        lastContacted: today,
                        daysSince: 0,
                        nextAvailable: moment().add(this.minDaysBetweenContacts, 'days').format('YYYY-MM-DD')
                    },
                    timestamp: moment().toISOString()
                });
            }
        }
        
        // Update in batches
        if (updates.length > 0) {
            try {
                await this.sheetsService.batchUpdateValues(spreadsheetId, updates);
                console.log(`✅ Updated last contacted for ${updates.length} contacts`);
            } catch (error) {
                console.error('Failed to update last contacted dates:', error);
            }
        }
    }
    
    // Get contact history summary
    getContactHistorySummary() {
        const summary = {
            totalContacts: this.contactHistory.size,
            recentlyContacted: 0,
            availableForContact: 0,
            next30Days: []
        };
        
        const now = moment();
        
        for (const [number, data] of this.contactHistory) {
            const lastContacted = data.result.lastContacted;
            
            if (!lastContacted) {
                summary.availableForContact++;
                continue;
            }
            
            const lastContactMoment = moment(lastContacted);
            const daysSince = now.diff(lastContactMoment, 'days');
            
            if (daysSince < 30) {
                summary.recentlyContacted++;
            }
            
            if (daysSince < this.minDaysBetweenContacts) {
                const nextAvailable = lastContactMoment.add(this.minDaysBetweenContacts, 'days');
                if (nextAvailable.diff(now, 'days') <= 30) {
                    summary.next30Days.push({
                        number: number,
                        lastContacted: lastContacted,
                        nextAvailable: nextAvailable.format('YYYY-MM-DD'),
                        daysUntil: nextAvailable.diff(now, 'days')
                    });
                }
            } else {
                summary.availableForContact++;
            }
        }
        
        // Sort next30Days by soonest available
        summary.next30Days.sort((a, b) => a.daysUntil - b.daysUntil);
        
        return summary;
    }
}
```

```javascript
const moment = require('moment-timezone');
const ISOCountryMapper = require('./iso-country-mapper');

class TimezoneScheduler {
    constructor() {
        this.countryMapper = new ISOCountryMapper();
        
        // Default schedule for unknown countries
        this.defaultSchedule = {
            businessHours: { start: 9, end: 17, lunch: { start: 12, end: 13 } },

                { start: 14, end: 16, weight: 0.4 }, // Afternoon

        this.countrySpecificSchedules = {
            'AE': { // UAE - Friday/Saturday weekend
                businessHours: { start: 8, end: 18, lunch: { start: 13, end: 14 } },
                weekend: [5, 6], // Friday, Saturday
                optimalWindows: [
                    { start: 9, end: 12, weight: 0.5 },
                    { start: 16, end: 19, weight: 0.5 }
                ],
                ramadanAdjustment: true // Special consideration for Ramadan
            },
            'SA': { // Saudi Arabia
                businessHours: { start: 8, end: 17, lunch: { start: 12, end: 13 } },
                weekend: [5, 6],
                optimalWindows: [
                    { start: 10, end: 13, weight: 0.6 },
                    { start: 16, end: 19, weight: 0.4 }
                ]
            },
            'IN': { // India
                businessHours: { start: 10, end: 19, lunch: { start: 13, end: 14 } },
                weekend: [0], // Sunday only
                optimalWindows: [
                    { start: 10, end: 13, weight: 0.4 },
                    { start: 16, end: 19, weight: 0.6 }
                ]
            }
        };
    }
    
    analyzeContactSchedule(contactNumber) {
        const countryInfo = this.countryMapper.analyzeNumber(contactNumber);
        const schedule = this.countrySpecificSchedules[countryInfo.alpha2] || this.defaultSchedule;
        
        return {
            ...countryInfo,
            schedule: schedule,
            currentLocalTime: this.getLocalTime(countryInfo.primaryTimezone),
            optimalSendTimes: this.calculateOptimalTimes(countryInfo, schedule)
        };
    }
    
    getLocalTime(timezone) {
        return {
            time: moment().tz(timezone).format('HH:mm'),
            day: moment().tz(timezone).day(),
            hour: moment().tz(timezone).hour(),
            timezone: timezone,
            isDST: moment().tz(timezone).isDST()
        };
    }
    
    canSendNow(contactNumber, options = {}) {
        const analysis = this.analyzeContactSchedule(contactNumber);
        const localTime = analysis.currentLocalTime;
        const schedule = analysis.schedule;
        
        // Check if it's weekend
        if (schedule.weekend.includes(localTime.day)) {
            return {
                allowed: false,
                reason: `Weekend in ${analysis.countryName} (${this.getDayName(localTime.day)})`,
                nextAvailable: this.getNextWeekday(analysis.primaryTimezone, schedule.weekend),
                analysis: analysis
            };
        }
        
        // Check business hours
        const hour = localTime.hour;
        if (hour < schedule.businessHours.start || hour >= schedule.businessHours.end) {
            return {
                allowed: false,
                reason: `Outside business hours (${schedule.businessHours.start}:00-${schedule.businessHours.end}:00 local time)`,
                nextAvailable: this.getNextBusinessHour(analysis.primaryTimezone, schedule.businessHours),
                analysis: analysis
            };
        }
        
        // Check lunch break
        if (schedule.businessHours.lunch && 
            hour >= schedule.businessHours.lunch.start && 
            hour < schedule.businessHours.lunch.end) {
            return {
                allowed: false,
                reason: `Lunch break in ${analysis.countryName}`,
                nextAvailable: moment().tz(analysis.primaryTimezone)
                    .hour(schedule.businessHours.lunch.end)
                    .minute(0)
                    .second(0)
                    .toISOString(),
                analysis: analysis
            };
        }
        
        // Check optimal windows (optional)
        if (options.requireOptimalWindow) {
            const inOptimalWindow = schedule.optimalWindows.some(window => 
                hour >= window.start && hour < window.end
            );
            
            if (!inOptimalWindow) {
                const nextWindow = this.getNextOptimalWindow(analysis, schedule);
                return {
                    allowed: false,
                    reason: 'Not in optimal sending window',
                    nextAvailable: nextWindow,
                    analysis: analysis
                };
            }
        }
        
        return {
            allowed: true,
            reason: 'Within acceptable sending window',
            analysis: analysis,
            optimality: this.calculateOptimalityScore(hour, schedule)
        };
    }
    
    calculateOptimalTimes(countryInfo, schedule) {
        const now = moment().tz(countryInfo.primaryTimezone);
        const today = now.clone().startOf('day');
        const optimalTimes = [];
        
        for (const window of schedule.optimalWindows) {
            const startTime = today.clone().hour(window.start).minute(0);
            const endTime = today.clone().hour(window.end).minute(0);
            
            // If window has already passed today, schedule for tomorrow
            if (now.isAfter(endTime)) {
                startTime.add(1, 'day');
                endTime.add(1, 'day');
            }
            
            optimalTimes.push({
                start: startTime.toISOString(),
                end: endTime.toISOString(),
                localStart: startTime.format('HH:mm'),
                localEnd: endTime.format('HH:mm'),
                weight: window.weight,
                isToday: startTime.isSame(now, 'day')
            });
        }
        
        // Sort by soonest
        optimalTimes.sort((a, b) => new Date(a.start) - new Date(b.start));
        
        return optimalTimes;
    }
    
    getNextBusinessHour(timezone, businessHours) {
        const now = moment().tz(timezone);
        const today = now.clone().startOf('day');
        
        if (now.hour() < businessHours.start) {
            // Business hours haven't started today
            return today.clone().hour(businessHours.start).minute(0).toISOString();
        } else if (now.hour() >= businessHours.end) {
            // Business hours ended, schedule for tomorrow
            return today.clone().add(1, 'day').hour(businessHours.start).minute(0).toISOString();
        }
        
        // Shouldn't reach here if checking properly
        return now.clone().add(1, 'hour').toISOString();
    }
    
    getNextWeekday(timezone, weekendDays) {
        let date = moment().tz(timezone);
        
        // Keep adding days until we find a non-weekend day
        while (weekendDays.includes(date.day())) {
            date = date.add(1, 'day');
        }
        
        return date.startOf('day').hour(9).minute(0).toISOString();
    }
    
    calculateOptimalityScore(currentHour, schedule) {
        let maxWeight = 0;
        let currentWeight = 0;
        
        for (const window of schedule.optimalWindows) {
            maxWeight += window.weight;
            if (currentHour >= window.start && currentHour < window.end) {
                currentWeight = window.weight;
            }
        }
        
        return maxWeight > 0 ? (currentWeight / maxWeight) * 100 : 50;
    }
    
    getDayName(dayIndex) {
        const days = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
        return days[dayIndex];
    }
    
    // Batch schedule analysis
    analyzeContactList(contactNumbers) {
        const results = {
            canSendNow: [],
            cannotSendNow: [],
            byCountry: {},
            summary: {
                total: contactNumbers.length,
                availableNow: 0,
                byTimeWindow: {
                    morning: 0,
                    afternoon: 0,
                    evening: 0,
                    outsideHours: 0,
                    weekend: 0
                }
            }
        };
        
        for (const number of contactNumbers) {
            const check = this.canSendNow(number);
            const analysis = check.analysis;
            
            // Categorize by country
            if (!results.byCountry[analysis.countryName]) {
                results.byCountry[analysis.countryName] = {
                    count: 0,
                    availableNow: 0,
                    timezone: analysis.primaryTimezone
                };
            }
            results.byCountry[analysis.countryName].count++;
            
            if (check.allowed) {
                results.canSendNow.push({
                    number: number,
                    country: analysis.countryName,
                    localTime: analysis.currentLocalTime.time,
                    optimality: check.optimality
                });
                results.summary.availableNow++;
                results.byCountry[analysis.countryName].availableNow++;
                
                // Categorize by time window
                const hour = analysis.currentLocalTime.hour;
                if (hour >= 6 && hour < 12) results.summary.byTimeWindow.morning++;
                else if (hour >= 12 && hour < 17) results.summary.byTimeWindow.afternoon++;
                else results.summary.byTimeWindow.evening++;
            } else {
                results.cannotSendNow.push({
                    number: number,
                    country: analysis.countryName,
                    reason: check.reason,
                    nextAvailable: check.nextAvailable,
                    localTime: analysis.currentLocalTime.time
                });
                
                if (check.reason.includes('Weekend')) {
                    results.summary.byTimeWindow.weekend++;
                } else {
                    results.summary.byTimeWindow.outsideHours++;
                }
            }
        }
        
        return results;
    }
    
    // Generate sending schedule for next 24 hours
    generate24HourSchedule(contactNumbers) {
        const schedule = {};
        const now = moment();
        
        // Create hourly slots for next 24 hours
        for (let i = 0; i < 24; i++) {
            const hourTime = now.clone().add(i, 'hours');
            const hourKey = hourTime.format('YYYY-MM-DD HH:00');
            schedule[hourKey] = {
                contacts: [],
                countries: new Set(),
                optimalCount: 0
            };
        }
        
        // Assign contacts to optimal slots
        for (const number of contactNumbers) {
            const analysis = this.analyzeContactSchedule(number);
            const optimalTimes = analysis.optimalSendTimes;
            
            for (const slot of optimalTimes) {
                const slotTime = moment(slot.start);
                const hourKey = slotTime.format('YYYY-MM-DD HH:00');
                
                if (schedule[hourKey]) {
                    schedule[hourKey].contacts.push({
                        number: number,
                        country: analysis.countryName,
                        optimality: slot.weight * 100
                    });
                    schedule[hourKey].countries.add(analysis.countryName);
                    if (slot.weight > 0.5) schedule[hourKey].optimalCount++;
                }
            }
        }
        
        // Filter out empty slots and sort
        const filteredSchedule = Object.entries(schedule)
            .filter(([_, data]) => data.contacts.length > 0)
            .sort((a, b) => a[0].localeCompare(b[0]))
            .slice(0, 12); // Next 12 hours
        
        return filteredSchedule;
    }
}
```

```javascript
    applyTimezoneScheduling(contacts) {
        if (!this.settings.respectTimezones) {
            return {
                availableNow: contacts,
                unavailableNow: [],

        }

    selectFinalContacts(contacts, randomize = true) {
        const availableQuota = this.settings.maxDailyMessages - this.state.todaySent;
        
        if (availableQuota <= 0) {
            return [];
        }
        
        let selectedContacts = [...contacts];
        
        if (randomize) {
            // Weighted random selection based on optimality
            selectedContacts = this.weightedRandomSelection(contacts, availableQuota);
        } else {
            // Take first N contacts
            selectedContacts = contacts.slice(0, Math.min(availableQuota, contacts.length));
        }
        
        console.log(`🎯 Selected ${selectedContacts.length} contacts (quota: ${availableQuota})`);
        
        return selectedContacts;

    weightedRandomSelection(contacts, n) {
        if (contacts.length <= n) {
            return this.shuffleArray([...contacts]);
        }
        
        // Calculate weights based on optimality
        const weightedContacts = contacts.map(contact => {
            let weight = 1;
            
            // Higher weight for better optimality
            if (contact.optimality > 80) weight = 3;
            else if (contact.optimality > 60) weight = 2;
            
            // Higher weight for never contacted
            if (!contact.lastContacted) weight *= 2;
            
            return { contact, weight };
        });
        
        const selected = [];
        const totalWeight = weightedContacts.reduce((sum, wc) => sum + wc.weight, 0);
        
        while (selected.length < n && weightedContacts.length > 0) {
            const random = Math.random() * totalWeight;
            let cumulative = 0;
            
            for (let i = 0; i < weightedContacts.length; i++) {
                cumulative += weightedContacts[i].weight;
                if (random <= cumulative) {
                    selected.push(weightedContacts[i].contact);
                    totalWeight -= weightedContacts[i].weight;
                    weightedContacts.splice(i, 1);
                    break;
                }
            }
        }
        
        return selected;
    }
    
    shuffleArray(array) {
        for (let i = array.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [array[i], array[j]] = [array[j], array[i]];
        }
        return array;
    }
    
    async executeCampaign(contacts, messageTemplate, campaignName, spreadsheetId, range) {
        console.log(`📤 Executing campaign: ${campaignName}`);
        
        const results = {
            sent: 0,
            failed: 0,
            details: [],
            startTime: new Date().toISOString()
        };
        
        for (const contact of contacts) {
            // Check daily limit
            if (this.state.todaySent >= this.settings.maxDailyMessages) {
                console.log('⏹️ Daily limit reached. Stopping campaign.');
                break;
            }
            
            // Personalize message
            const personalizedMessage = this.personalizeMessage(messageTemplate, contact);
            
            // Calculate dynamic delay
            const delay = this.calculateDynamicDelay(results.sent);
            await this.delay(delay);
            
            try {
                // Send message
                const sendResult = await this.client.sendMessage(
                    contact.whatsappId || contact.validatedNumber,
                    personalizedMessage
                );
                
                results.sent++;
                this.state.todaySent++;
                this.state.contactsProcessed.add(contact.validatedNumber);
                
                // Update last contacted
                await this.recencyFilter.updateLastContacted(
                    [contact],
                    spreadsheetId,
                    range
                );
                
                results.details.push({
                    contact: contact.validatedNumber,
                    name: contact.name,
                    country: contact.countryName,
                    nationality: contact.nationality,
                    success: true,
                    messageId: sendResult.id?.id,
                    timestamp: new Date().toISOString(),
                    localTime: contact.fullAnalysis?.currentLocalTime?.time,
                    optimality: contact.optimality,
                    safetyStatus: contact.safetyStatus?.reason
                });
                
                console.log(`✅ Sent to ${contact.name || contact.validatedNumber} (${contact.countryName})`);
                
            } catch (error) {
                results.failed++;
                this.state.todayFailed++;
                
                results.details.push({
                    contact: contact.validatedNumber,
                    name: contact.name,
                    country: contact.countryName,
                    nationality: contact.nationality,
                    success: false,
                    error: error.message,
                    timestamp: new Date().toISOString(),
                    localTime: contact.fullAnalysis?.currentLocalTime?.time
                });
                
                console.error(`❌ Failed to send to ${contact.validatedNumber}:`, error.message);
                
                // Check if it's a block error
                const errorCode = error.message.match(/\d+/)?.[0];
                if (errorCode && ['131021', '131026', '131045', '131050'].includes(errorCode)) {
                    // Mark as blocked/opted out
                    this.optOutDetector.markAsOptedOut(
                        contact.validatedNumber,
                        `WhatsApp error ${errorCode}: ${error.message}`
                    );
                }
            }
        }
        
        results.endTime = new Date().toISOString();
        results.duration = moment(results.endTime).diff(moment(results.startTime), 'minutes');
        
        return results;
    }
    
    personalizeMessage(template, contact) {
        const variables = {
            name: contact.name || 'there',
            firstName: contact.name?.split(' ')[0] || 'there',
            country: contact.countryName,
            day: moment().format('dddd'),

            time: moment().format('h:mm a'),
            localTime: contact.fullAnalysis?.currentLocalTime?.time || moment().format('HH:mm'),
            nationality: contact.nationality
        };
        
        let personalized = template;
        Object.entries(variables).forEach(([key, value]) => {
            personalized = personalized.replace(new RegExp(`{${key}}`, 'gi'), value);
        });
        
        return personalized;
    }
    
    calculateDynamicDelay(messageCount) {
        // Base delay with fatigue factor
        const baseDelay = 5000 + (Math.log(messageCount + 1) * 1000);
        
        // Random variation
        const randomFactor = 0.5 + Math.random() * 0.5;
        const randomDelay = baseDelay * randomFactor;
        
        // Occasional longer pauses
        const pauseChance = 0.03;
        const pause = Math.random() < pauseChance ? 
                      30000 + Math.random() * 60000 : 0;
        
        return Math.min(30000, baseDelay + randomDelay + pause);
    }

    groupByCountry(items) {
        const byCountry = {};
        
        for (const item of items) {
            const country = item.countryName || item.country || 'Unknown';
            if (!byCountry[country]) {
                byCountry[country] = [];
            }
            byCountry[country].push(item);
        }
        
        // Convert to array with counts
        return Object.entries(byCountry).map(([country, items]) => ({
            country,
            count: items.length,
            sample: items.slice(0, 3)
        })).sort((a, b) => b.count - a.count);
    }
    
    groupSchedulingByCountry(available, unavailable) {
        const byCountry = {};
        
        // Add available
        for (const contact of available) {
            const country = contact.countryName;
            if (!byCountry[country]) {
                byCountry[country] = { available: 0, unavailable: 0 };
            }
            byCountry[country].available++;
        }
        
        // Add unavailable
        for (const item of unavailable) {
            const country = item.contact.countryName;
            if (!byCountry[country]) {
                byCountry[country] = { available: 0, unavailable: 0 };
            }
            byCountry[country].unavailable++;
        }
        
        return byCountry;
    }
    
    generateRecommendations(contactsData, recencyResult, safetyResult, schedulingResult, campaignResult) {
        const recommendations = [];
        
        // Check UAE focus
        const uaeContacts = contactsData.validContacts.filter(c => c.countryCode === '971').length;
        if (uaeContacts === 0) {
            recommendations.push('⚠️ No UAE (+971) contacts found in your list. Consider adding UAE contacts.');
        }
        
        // Check safety failure rate
        const safetyFailureRate = safetyResult.stats.failed / safetyResult.stats.totalChecked;
        if (safetyFailureRate > 0.3) {
            recommendations.push('🚨 High safety failure rate (>30%). Review your contact list for opt-outs/blocks.');
        }
        
        // Check scheduling efficiency
        const schedulingEfficiency = schedulingResult.availableNow.length / safetyResult.safeContacts.length;
        if (schedulingEfficiency < 0.5) {
            recommendations.push('⏰ Low scheduling efficiency (<50%). Consider expanding sending windows or targeting different timezones.');
        }
        
        // Check chat check rate
        const checkStats = this.optOutDetector.getChatCheckStats('sample');
        if (checkStats.count > 3) {
            recommendations.push('🔍 High chat check frequency detected. Maintain <5 checks per month per contact to avoid detection.');
        }
        
        return recommendations;
    }
    
    createResult(message, campaignName, details = {}) {

            }
        };
    }
    
    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
    
    // Reset daily counters
    resetDailyCounters() {
        this.state.todaySent = 0;
        this.state.todayFailed = 0;
        this.state.contactsProcessed.clear();
        this.state.campaignStart = new Date().toISOString();

            },
            limits: {
                dailyRemaining: this.settings.maxDailyMessages - this.state.todaySent,
                canSendMore: this.state.todaySent < this.settings.maxDailyMessages
            }
        };
    }
```

## ⚠️ **Critical Implementation Notes**

1. **ISO 3166-1 Data**: You need to complete the `countryData` object with all 200+ country codes. Consider using a library like `country-codes` or `libphonenumber-js`.

2. **Chat Check Rate Limiting**: The system enforces ≤5 chat checks per month per contact. Monitor this closely.

3. **UAE Focus**: The system prioritizes UAE numbers and uses UAE-specific scheduling.

4. **Detection Avoidance**:
   - Random delays between 5-30 seconds
   - Human-like pauses (3% probability)
   - Limited chat history checks
   - Timezone-aware scheduling
   - Weighted random contact selection

5. **Safety First**: When in doubt, the system errs on the side of not sending to avoid blocks.

This integrated system provides maximum protection while maintaining functionality. However, remember that **no unofficial WhatsApp solution is completely safe from detection**. Use with extreme caution and consider migrating to the official WhatsApp Business API for production use.

```javascript
// whatsapp-auth-manager.js
const { Client, LocalAuth } = require('whatsapp-web.js');
const crypto = require('crypto');
const EventEmitter = require('events');

class WhatsAppAuthManager extends EventEmitter {
    constructor() {
        super();
        this.authCodes = new Map(); // Stores active auth codes
        this.linkedDevices = new Map(); // Stores linked devices
        this.authExpiry = 5 * 60 * 1000; // 5 minutes
        
        // Client configuration
        this.clientConfig = {
            authStrategy: new LocalAuth({
                dataPath: '.wwebjs_auth',
                clientId: `client-${Date.now()}`
            }),
            puppeteer: {
                headless: process.env.NODE_ENV === 'production' ? 'new' : false,
                args: [
                    '--no-sandbox',
                    '--disable-setuid-sandbox',
                    '--disable-dev-shm-usage',
                    '--disable-accelerated-2d-canvas',
                    '--disable-web-security',
                    '--disable-features=IsolateOrigins,site-per-process',
                    '--disable-blink-features=AutomationControlled'
                ]
            },
            webVersionCache: {
                type: 'remote',
                remotePath: 'https://raw.githubusercontent.com/wppconnect-team/wa-version/main/html/2.2412.54.html'
            }
        };
    }
    
    /**
     * Generate a 6-digit code for device linking
     */
    generateAuthCode() {
        const code = Math.floor(100000 + Math.random() * 900000).toString();
        const sessionData = {
            code,
            timestamp: Date.now(),
            clientId: `client-${crypto.randomBytes(8).toString('hex')}`,
            status: 'pending',
            attempts: 0
        };
        
        this.authCodes.set(code, sessionData);
        
        // Auto-expire after 5 minutes
        setTimeout(() => {
            if (this.authCodes.get(code)?.status === 'pending') {
                this.authCodes.delete(code);
                console.log(`Auth code ${code} expired`);
            }
        }, this.authExpiry);
        
        console.log(`📱 Generated auth code: ${code} (valid for 5 minutes)`);
        return { code, expiresIn: this.authExpiry };
    }
    
    /**
     * Verify auth code and link device
     */
    async verifyAndLinkDevice(code) {
        const sessionData = this.authCodes.get(code);
        
        if (!sessionData) {
            return { success: false, error: 'Invalid or expired code' };
        }
        
        if (sessionData.status !== 'pending') {
            return { success: false, error: 'Code already used' };
        }
        
        sessionData.attempts++;
        
        if (sessionData.attempts > 3) {
            this.authCodes.delete(code);
            return { success: false, error: 'Too many attempts' };
        }
        
        try {
            // Create new client with unique clientId
            const client = new Client({
                ...this.clientConfig,
                authStrategy: new LocalAuth({
                    dataPath: '.wwebjs_auth',
                    clientId: sessionData.clientId
                })
            });
            
            // Set up event handlers
            return await this.initializeClientWithCode(client, code, sessionData);
            
        } catch (error) {
            return { success: false, error: error.message };
        }
    }
    
    async initializeClientWithCode(client, code, sessionData) {
        return new Promise((resolve) => {
            let qrTimer;
            let authTimer;
            
            client.on('qr', (qr) => {
                console.log('QR received (fallback)');
                // Clear timers if QR is shown
                clearTimeout(qrTimer);
                clearTimeout(authTimer);
                
                sessionData.status = 'qr_shown';
                this.authCodes.set(code, sessionData);
                
                resolve({
                    success: false,
                    requiresQR: true,
                    qrCode: qr,
                    message: 'Authentication requires QR code scanning'
                });
            });
            
            client.on('authenticated', () => {
                console.log(`✅ Device linked successfully with code: ${code}`);
                
                sessionData.status = 'authenticated';
                sessionData.authenticatedAt = Date.now();
                this.authCodes.set(code, sessionData);
                
                // Store linked device
                this.linkedDevices.set(sessionData.clientId, {
                    client,
                    code,
                    linkedAt: new Date().toISOString(),
                    lastActivity: Date.now()
                });
                
                resolve({
                    success: true,
                    clientId: sessionData.clientId,
                    message: 'Device linked successfully',
                    timestamp: new Date().toISOString()
                });
            });
            
            client.on('ready', () => {
                console.log(`Client ${sessionData.clientId} is ready`);
                this.emit('device-ready', sessionData.clientId);
            });
            
            client.on('auth_failure', (error) => {
                resolve({
                    success: false,
                    error: `Authentication failed: ${error}`
                });
            });
            
            // Timeout handling
            qrTimer = setTimeout(() => {
                if (sessionData.status === 'pending') {
                    resolve({
                        success: false,
                        error: 'Authentication timed out (no QR received)'
                    });
                    client.destroy();
                }
            }, 30000);
            
            authTimer = setTimeout(() => {
                if (sessionData.status !== 'authenticated') {
                    resolve({
                        success: false,
                        error: 'Authentication process timed out'
                    });
                    client.destroy();
                }
            }, 120000);
            
            // Initialize client
            client.initialize();
        });
    }
    
    /**
     * Get all linked devices
     */
    getLinkedDevices() {
        const devices = [];
        
        for (const [clientId, device] of this.linkedDevices) {
            devices.push({
                clientId,
                code: device.code,
                linkedAt: device.linkedAt,
                lastActivity: new Date(device.lastActivity),
                status: device.client?.info ? 'active' : 'inactive'
            });
        }
        
        return devices;
    }
    
    /**
     * Get client by clientId
     */
    getClient(clientId) {
        const device = this.linkedDevices.get(clientId);
        return device?.client || null;
    }
    
    /**
     * Switch between multiple linked devices
     */
    switchDevice(clientId) {
        const device = this.linkedDevices.get(clientId);
        if (!device || !device.client) {
            throw new Error(`Device ${clientId} not found or inactive`);
        }
        
        // Update last activity
        device.lastActivity = Date.now();
        this.linkedDevices.set(clientId, device);
        
        return device.client;
    }
    
    /**
     * Unlink device
     */
    async unlinkDevice(clientId) {
        const device = this.linkedDevices.get(clientId);
        if (device) {
            if (device.client) {
                await device.client.destroy();
            }
            this.linkedDevices.delete(clientId);
            
            // Also remove from auth codes
            for (const [code, data] of this.authCodes) {
                if (data.clientId === clientId) {
                    this.authCodes.delete(code);
                }
            }
            
            console.log(`Device ${clientId} unlinked successfully`);
            return true;
        }
        return false;
    }
    
    /**
     * Clean up expired codes and inactive devices
     */
    cleanup() {
        const now = Date.now();
        let cleaned = 0;
        
        // Clean expired auth codes
        for (const [code, data] of this.authCodes) {
            if (now - data.timestamp > this.authExpiry) {
                this.authCodes.delete(code);
                cleaned++;
            }
        }
        
        // Clean inactive devices (more than 24 hours)
        for (const [clientId, device] of this.linkedDevices) {
            if (now - device.lastActivity > 24 * 60 * 60 * 1000) {
                this.unlinkDevice(clientId);
                cleaned++;
            }
        }
        
        return cleaned;
    }
    
    /**
     * REST API endpoints for authentication
     */
    setupAuthAPI(app) {
        // Generate new auth code
        app.post('/api/auth/generate-code', (req, res) => {
            const result = this.generateAuthCode();
            res.json(result);
        });
        
        // Verify auth code
        app.post('/api/auth/verify-code', async (req, res) => {
            const { code } = req.body;
            if (!code) {
                return res.status(400).json({ error: 'Code required' });
            }
            
            const result = await this.verifyAndLinkDevice(code);
            res.json(result);
        });
        
        // List linked devices
        app.get('/api/auth/devices', (req, res) => {
            const devices = this.getLinkedDevices();
            res.json({ devices });
        });
        
        // Switch active device
        app.post('/api/auth/switch-device', (req, res) => {
            const { clientId } = req.body;
            try {
                const client = this.switchDevice(clientId);
                res.json({
                    success: true,
                    message: `Switched to device ${clientId}`,
                    clientId
                });
            } catch (error) {
                res.status(400).json({ error: error.message });
            }
        });
        
        // Unlink device
        app.delete('/api/auth/devices/:clientId', async (req, res) => {
            const { clientId } = req.params;
            const success = await this.unlinkDevice(clientId);
            res.json({ success });
        });
        
        // Get auth status
        app.get('/api/auth/status', (req, res) => {
            res.json({
                activeCodes: this.authCodes.size,
                linkedDevices: this.linkedDevices.size,
                lastCleanup: new Date().toISOString()
            });
        });
    }
}
```

```javascript
// sheet-rotation-manager.js
const moment = require('moment');
const crypto = require('crypto');

class SheetRotationManager {
    constructor(googleSheetsService) {
        this.sheetsService = googleSheetsService;
        
        // Sheet rotation configuration
        this.config = {
            rotationMode: process.env.SHEET_ROTATION_MODE || 'sequential', // sequential, random, weighted
            sheetConfigs: [], // Will be loaded from env or config file
            currentIndex: 0,
            rotationHistory: [],
            maxHistorySize: 100
        };
        
        // Load sheet configurations
        this.loadSheetConfigs();
    }
    
    /**
     * Load sheet configurations from environment or config file
     */
    loadSheetConfigs() {
        try {
            // Option 1: From environment variable (JSON string)
            if (process.env.SHEET_CONFIGS) {
                this.config.sheetConfigs = JSON.parse(process.env.SHEET_CONFIGS);
                console.log(`Loaded ${this.config.sheetConfigs.length} sheet configurations from env`);
            }
            
            // Option 2: From config file
            else {
                const configFile = require('./sheet-configs.json');
                this.config.sheetConfigs = configFile.sheets || [];
                console.log(`Loaded ${this.config.sheetConfigs.length} sheet configurations from file`);
            }
            
            // Validate configurations
            this.validateSheetConfigs();
            
        } catch (error) {
            console.error('Failed to load sheet configs:', error);
            // Fallback to single sheet
            this.config.sheetConfigs = [{
                id: 'default',
                spreadsheetId: process.env.GOOGLE_SHEETS_SPREADSHEET_ID,
                range: process.env.GOOGLE_SHEETS_RANGE,
                name: 'Default Sheet',
                weight: 1,
                lastUsed: null,
                useCount: 0
            }];
        }
    }
    
    /**
     * Get next sheet based on rotation strategy
     */
    getNextSheet(campaignName = '') {
        if (this.config.sheetConfigs.length === 0) {
            throw new Error('No sheet configurations available');
        }
        
        let selectedSheet;
        
        switch (this.config.rotationMode) {
            case 'sequential':
                selectedSheet = this.getSequentialSheet();
                break;
                
            case 'random':
                selectedSheet = this.getRandomSheet();
                break;
                
            case 'weighted':
                selectedSheet = this.getWeightedSheet();
                break;
                
            case 'round_robin':
                selectedSheet = this.getRoundRobinSheet();
                break;
                
            default:
                selectedSheet = this.getSequentialSheet();
        }
        
        // Update sheet usage statistics
        this.updateSheetUsage(selectedSheet, campaignName);
        
        return selectedSheet;
    }
    
    /**
     * Sequential rotation (one after another)
     */
    getSequentialSheet() {
        const sheet = { ...this.config.sheetConfigs[this.config.currentIndex] };
        this.config.currentIndex = (this.config.currentIndex + 1) % this.config.sheetConfigs.length;
        return sheet;
    }
    
    /**
     * Random selection
     */
    getRandomSheet() {
        const randomIndex = Math.floor(Math.random() * this.config.sheetConfigs.length);
        return { ...this.config.sheetConfigs[randomIndex] };
    }
    
    /**
     * Weighted random selection
     */
    getWeightedSheet() {
        const totalWeight = this.config.sheetConfigs.reduce((sum, sheet) => sum + (sheet.weight || 1), 0);
        const random = Math.random() * totalWeight;
        
        let cumulative = 0;
        for (const sheet of this.config.sheetConfigs) {

        };
```

```javascript
    /**
     * Final contact selection
     */
    selectFinalContacts(contacts, campaignConfig) {
        const maxContacts = campaignConfig.maxContacts || 50;
        
        if (contacts.length <= maxContacts) {
            return this.shuffleArray([...contacts]);
        }
        
        // Weighted selection based on country priority and optimality
        const weightedContacts = contacts.map(contact => {
            let weight = 1;
            
            // Higher weight for UAE contacts
            if (contact.countryCode === '971') weight *= 3;
            
            // Higher weight for optimal send times
            if (contact.optimality > 80) weight *= 2;
            else if (contact.optimality > 60) weight *= 1.5;
            
            // Lower weight for recently contacted
            if (contact.lastContacted) {
                const daysSince = moment().diff(moment(contact.lastContacted), 'days');
                if (daysSince < 30) weight *= 0.5;
            }
            
            return { contact, weight };
        });
        
        return this.weightedRandomSelection(weightedContacts, maxContacts);
    }
    
    /**
     * Execute campaign with all protections
     */
    async executeCampaign(client, contacts, campaignConfig, campaignId, sheetConfig) {
        console.log(`🚀 Executing campaign ${campaignId} with ${contacts.length} contacts`);
        
        const results = {
            campaignId,
            startTime: new Date().toISOString(),
            sheetUsed: sheetConfig.name,
            sent: 0,
            failed: 0,
            permanentFailures: 0,
            details: []
        };
        
        for (let i = 0; i < contacts.length; i++) {
            const contact = contacts[i];
            
            try {
                // 1. Pre-send validation
                const preSendCheck = await this.preSendValidation(contact, campaignId);
                if (!preSendCheck.valid) {
                    results.failed++;
                    results.details.push({
                        contact: contact.validatedNumber,
                        success: false,
                        reason: preSendCheck.reason,
                        timestamp: new Date().toISOString(),
                        stage: 'pre_send_validation'
                    });
                    continue;
                }
                
                // 2. Calculate dynamic delay
                const delay = this.calculateSmartDelay(i, results.sent);
                await this.delay(delay);
                
                // 3. Personalize message
                const personalizedMessage = this.personalizeMessage(
                    campaignConfig.messageTemplate,
                    contact,
                    campaignConfig
                );
                
                // 4. Send message
                const sendResult = await client.sendMessage(
                    contact.whatsappId || contact.validatedNumber,
                    personalizedMessage
                );
                
                // 5. Record success
                results.sent++;
                this.state.dailyStats.messagesSent++;
                this.state.dailyStats.uniqueContacts.add(contact.validatedNumber);
                
                await this.restrictionManager.recordSelection(
                    contact.validatedNumber,
                    campaignId,
                    'sent_successfully'
                );
                
                results.details.push({
                    contact: contact.validatedNumber,
                    name: contact.name,
                    country: contact.countryName,
                    nationality: contact.nationality,
                    success: true,
                    messageId: sendResult.id?.id,
                    timestamp: new Date().toISOString(),
                    localTime: contact.currentLocalTime?.time,
                    delayUsed: delay,
                    campaignId
                });
                
                console.log(`✅ Sent to ${contact.name || contact.validatedNumber} (${contact.countryName})`);
                
                // 6. Update contact history in sheet
                await this.updateContactHistory(contact, sheetConfig, 'sent');
                
            } catch (error) {
                // 7. Handle failure
                results.failed++;
                
                const errorMsg = error.message.toLowerCase();
                const isPermanent = this.isPermanentError(errorMsg);
                
                if (isPermanent) {
                    results.permanentFailures++;
                }
                
                await this.restrictionManager.recordFailedMessage(
                    contact.validatedNumber,
                    error,
                    campaignId,
                    isPermanent
                );
                
                results.details.push({
                    contact: contact.validatedNumber,
                    success: false,
                    error: error.message,
                    isPermanent,
                    timestamp: new Date().toISOString(),
                    retryRecommendation: isPermanent ? 'Do not retry' : 'Retry after 24h'
                });
                
                console.error(`❌ Failed to send to ${contact.validatedNumber}:`, error.message);
                
                // 8. Adaptive response to failures
                if (isPermanent || results.failed > 5) {
                    console.warn('⚠️ Multiple failures detected, considering pause');
                    await this.delay(10000); // Extra pause
                }
            }
            
            // 9. Check daily limits
            if (this.state.dailyStats.messagesSent >= 100) { // Hard daily limit
                console.log('⏹️ Daily message limit reached, stopping campaign');
                break;
            }
        }
        
        results.endTime = new Date().toISOString();
        results.duration = moment(results.endTime).diff(moment(results.startTime), 'minutes');
        
        return results;
    }
    
    /**
     * Pre-send validation
     */
    async preSendValidation(contact, campaignId) {
        const checks = [
            // Check if number was selected in last 7 days
            async () => {
                const selectionCheck = await this.restrictionManager.canSelectNumber(
                    contact.validatedNumber,
                    campaignId
                );
                if (!selectionCheck.allowed) {
                    return { valid: false, reason: selectionCheck.reason };
                }
                return { valid: true };
            },
            
            // Check for recent failures
            async () => {
                const failureCheck = await this.restrictionManager.checkFailedMessages(
                    contact.validatedNumber
                );
                if (!failureCheck.shouldRetry) {
                    return { valid: false, reason: failureCheck.reason };
                }
                return { valid: true };
            },
            
            // Check if still in optimal time window
            () => {
                const currentHour = moment().tz(contact.primaryTimezone).hour();
                const schedule = this.timezoneScheduler.getCountrySchedule(contact.countryCode);
                
                if (schedule && schedule.businessHours) {
                    if (currentHour < schedule.businessHours.start || 
                        currentHour >= schedule.businessHours.end) {
                        return {
                            valid: false,
                            reason: `Outside business hours in ${contact.countryName}`
                        };
                    }
                }
                return { valid: true };
            }
        ];
        
        for (const check of checks) {
            const result = await check();
            if (!result.valid) {
                return result;
            }
        }
        
        return { valid: true };
    }
    
    /**
     * Post-campaign processing
     */
    async postCampaignProcessing(campaignResult, sheetConfig, campaignId) {
        // 1. Log campaign results
        await this.logCampaignResults(campaignResult, sheetConfig);
        
        // 2. Update sheet rotation statistics
        this.sheetManager.updateSheetUsage(sheetConfig, campaignId);
        
        // 3. Update campaign history
        this.state.campaignHistory.push({
            campaignId,
            timestamp: new Date().toISOString(),
            sheet: sheetConfig.name,
            stats: {
                sent: campaignResult.sent,
                failed: campaignResult.failed,
                duration: campaignResult.duration
            }
        });
        
        // 4. Clean old history
        if (this.state.campaignHistory.length > 100) {
            this.state.campaignHistory = this.state.campaignHistory.slice(-100);
        }
        
        // 5. Run cleanup
        this.restrictionManager.cleanupOldEntries();
        this.authManager.cleanup();
        
        console.log(`📊 Campaign ${campaignId} completed processing`);
    }

    compileResults(campaignId, campaignConfig, sheetConfig, contactsData,

    /**
     * Utility methods
     */
    calculateSmartDelay(messageIndex, successCount) {
        // Base delay increases with message count
        let baseDelay = 5000 + (Math.log(messageIndex + 1) * 1000);
        
        // Adjust based on success rate
        if (successCount > 0) {
            const successRate = successCount / (messageIndex + 1);
            if (successRate > 0.9) baseDelay *= 0.8; // Speed up if high success
            if (successRate < 0.5) baseDelay *= 1.5; // Slow down if low success
        }
        
        // Random variation
        baseDelay *= (0.8 + Math.random() * 0.4);
        
        // Hard limits
        return Math.max(3000, Math.min(30000, baseDelay));
    }
    
    isPermanentError(errorMessage) {
        const permanentIndicators = [
            'blocked',
            'banned',
            'not authorized',
            '131021',
            '131026',
            '131045',
            '131050',
            'user not found',
            'not on whatsapp'
        ];
        
        return permanentIndicators.some(indicator => 
            errorMessage.toLowerCase().includes(indicator.toLowerCase())
        );
    }
    
    weightedRandomSelection(weightedItems, n) {
        const selected = [];
        const items = [...weightedItems];
        
        while (selected.length < n && items.length > 0) {
            const totalWeight = items.reduce((sum, item) => sum + item.weight, 0);
            const random = Math.random() * totalWeight;
            let cumulative = 0;
            
            for (let i = 0; i < items.length; i++) {
                cumulative += items[i].weight;
                if (random <= cumulative) {
                    selected.push(items[i].contact);
                    items.splice(i, 1);
                    break;
                }
            }
        }
        
        return selected;
    }
    
    shuffleArray(array) {
        for (let i = array.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [array[i], array[j]] = [array[j], array[i]];
        }
        return array;
    }
```

```javascript
/**
 * Validate and correct phone number format
 * @param {string} number - Phone number to validate
 * @returns {object} {isValid: boolean, correctedNumber: string, countryCode: string}
 */
function validateAndCorrectNumber(number) {
    // Remove all non-digit characters
    let cleaned = number.replace(/\D/g, '');
    
    // Common country codes for validation
    const countryCodes = {
        '1': 'US/Canada',
        '44': 'UK',
        '91': 'India',
        '86': 'China',
        '33': 'France',
        '49': 'Germany',
        '81': 'Japan',
        '55': 'Brazil',
        '7': 'Russia',
        '61': 'Australia'
    };
    
    // Check if number already has country code
    let countryCode = '';
    let correctedNumber = '';
    let isValid = false;
    
    // Check for known country codes
    for (const [code, country] of Object.entries(countryCodes)) {
        if (cleaned.startsWith(code)) {
            countryCode = code;
            correctedNumber = cleaned;
            isValid = true;
            break;
        }
    }
    
    // If no country code found, try to detect
    if (!isValid) {
        // For demonstration, assuming Indian numbers if 10 digits
        if (cleaned.length === 10) {
            countryCode = '91';
            correctedNumber = '91' + cleaned;
            isValid = true;
        } else if (cleaned.length >= 10 && cleaned.length <= 15) {
            // Generic formatting
            correctedNumber = cleaned;
            isValid = true;
        }
    }
    
    // Format for WhatsApp (add @c.us for non-group chats)
    if (isValid && !correctedNumber.includes('@')) {
        correctedNumber = correctedNumber + '@c.us';
    }
    
    return {
        isValid,
        correctedNumber,
        countryCode,
        original: number
    };
}

/**
 * Send message to a single number
 * @param {string} number - WhatsApp number
 * @param {string} message - Message content
 * @param {object} options - Additional options
 */
async function sendMessage(number, message, options = {}) {
    try {
        const validation = validateAndCorrectNumber(number);
        
        if (!validation.isValid) {
            throw new Error(`Invalid number format: ${number}`);
        }

    } catch (error) {

        return {
            success: false,

        };
    }
}

/**
 * Run a messaging campaign to multiple numbers
 * @param {Array} numbers - Array of phone numbers
 * @param {string} message - Message to send
 * @param {object} options - Campaign options
 */
async function runCampaign(numbers, message, options = {}) {
    const results = {
        total: numbers.length,
        successful: 0,
        failed: 0,
        details: []
    };
    
    const {
        delayBetweenMessages = 2000, // 2 seconds delay
        batchSize = 10,
        maxRetries = 3
    } = options;
    
    // Process in batches to avoid rate limiting
    for (let i = 0; i < numbers.length; i += batchSize) {
        const batch = numbers.slice(i, i + batchSize);
        
        // Process each number in the batch
        for (const number of batch) {
            let retries = 0;
            let sent = false;
            
            while (retries < maxRetries && !sent) {
                try {
                    const result = await sendMessage(number, message);
                    
                    results.details.push({
                        number,
                        success: result.success,
                        timestamp: new Date().toISOString(),
                        retries: retries,
                        error: result.error || null
                    });
                    
                    if (result.success) {
                        results.successful++;
                    } else {
                        results.failed++;
                    }
                    
                    sent = true;
                } catch (error) {
                    retries++;
                    if (retries === maxRetries) {
                        results.details.push({
                            number,
                            success: false,
                            timestamp: new Date().toISOString(),
                            retries: retries,
                            error: `Failed after ${maxRetries} retries: ${error.message}`
                        });
                        results.failed++;
                    }
                }
            }
            
            // Delay between messages in the same batch
            if (delayBetweenMessages > 0) {
                await new Promise(resolve => setTimeout(resolve, delayBetweenMessages));
            }
        }
        
        // Optional delay between batches
        if (options.delayBetweenBatches && i + batchSize < numbers.length) {
            await new Promise(resolve => setTimeout(resolve, options.delayBetweenBatches));
        }
    }
    
    return results;
}
```

```html
                const response = await axios.post('/api/validate-numbers', { numbers });
                const data = response.data;
                
                let html = `<strong>Validation Summary:</strong><br>`;
                html += `Total: ${data.summary.total} | Valid: ${data.summary.valid} | Invalid: ${data.summary.invalid}<br><br>`;
                
                html += `<strong>Details:</strong><br>`;
                data.details.forEach(d => {
                    html += `${d.original}: ${d.isValid ? '✓' : '✗'} `;
                    if (d.isValid) {
                        html += `→ ${d.correctedNumber} (${d.countryCode})`;
                    } else {
                        html += `Invalid format`;
                    }
                    html += `<br>`;
                });
                
                document.getElementById('validate-result').innerHTML = html;
            } catch (error) {
                showResult('validate-result', `Error: ${error.message}`, 'error');
            }
        }

        async function getChats() {

                const response = await axios.get('/api/chats');
                const chats = response.data.chats.slice(0, 10); // Show first 10 chats
```
