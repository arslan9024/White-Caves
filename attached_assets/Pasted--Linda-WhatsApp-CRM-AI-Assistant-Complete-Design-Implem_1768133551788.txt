```python
class ConversationRouter:
    """
    Routes incoming WhatsApp messages to appropriate agents or AI
    """
    
    def route_message(self, message: WhatsAppMessage) -> RouteDecision:
        # Step 1: Identify sender
        sender = self.identify_sender(message.from_number)
        
        # Step 2: Check for active conversation
        active_convo = self.find_active_conversation(sender)
        
        if active_convo:
            # Route to same agent if available
            return RouteDecision(
                type="AGENT",
                agent_id=active_convo.assigned_agent,
                conversation_id=active_convo.id
            )
        
        # Step 3: Classify message intent
        intent = self.classify_intent(message.text)
        
        # Step 4: Apply routing rules
        if intent in ["greeting", "faq", "hours"]:
            # Handle with AI
            return RouteDecision(
                type="AI",
                ai_handler="auto_reply",
                conversation_id=None
            )
        else:
            # Assign to human agent
            agent = self.assign_to_agent(
                sender=sender,
                intent=intent,
                priority=self.calculate_priority(message)
            )
            
            return RouteDecision(
                type="AGENT",
                agent_id=agent.id,
                conversation_id=self.create_conversation(sender, agent)
            )
    
    def assign_to_agent(self, sender, intent, priority) -> Agent:
        """
        Smart agent assignment algorithm
        """
        available_agents = self.get_available_agents()
        
        # Score each agent
        scores = {}
        for agent in available_agents:
            score = 0
            
            # Skill match (30%)
            if intent in agent.skills:
                score += 30
            
            # Workload (25%)
            workload = agent.current_conversations / agent.max_conversations
            score += 25 * (1 - workload)
            
            # Language match (20%)
            if sender.language in agent.languages:
                score += 20
            
            # Previous interaction (15%)
            if sender.id in agent.previous_customers:
                score += 15
            
            # Response time performance (10%)
            score += 10 * (agent.avg_response_time_score)
            
            scores[agent.id] = score
        
        # Return best agent
        return max(scores.items(), key=lambda x: x[1])[0]
```

```python
class AIServices:
    """
    AI capabilities for WhatsApp conversations
    """
    
    def __init__(self):
        self.nlp = WhatsAppNLP()
        self.sentiment = SentimentAnalyzer()
        self.translator = TranslationService()
        self.auto_reply = AutoReplyEngine()
    
    async def process_incoming(self, message: WhatsAppMessage) -> AIResponse:
        """
        Process incoming message with AI
        """
        # Language detection
        language = await self.detect_language(message.text)
        
        # Translation if needed
        translated = await self.translator.translate_if_needed(
            message.text, 
            target_language="en"
        )
        
        # Intent classification
        intent = await self.nlp.classify_intent(translated.text)
        
        # Sentiment analysis
        sentiment = await self.sentiment.analyze(translated.text)
        
        # Entity extraction
        entities = await self.nlp.extract_entities(translated.text)
        
        # Auto-reply generation (if applicable)
        auto_reply = None
        if self.should_auto_reply(intent, sentiment):
            auto_reply = await self.auto_reply.generate(
                intent=intent,
                context=message.context,
                language=language
            )
        
        return AIResponse(
            language=language,
            intent=intent,
            sentiment=sentiment,
            entities=entities,
            auto_reply=auto_reply,
            priority=self.calculate_priority(intent, sentiment),
            suggested_tags=self.suggest_tags(intent, entities)
        )
    
    def should_auto_reply(self, intent: str, sentiment: dict) -> bool:
        """
        Determine if AI should auto-reply
        """
        auto_reply_intents = [
            "greeting", "thank_you", "hours", 
            "location", "website", "faq_general"
        ]
        
        # Don't auto-reply to negative sentiment
        if sentiment.get("score", 0) < -0.3:
            return False
        
        return intent in auto_reply_intents
```

```python
class SmartReplyEngine:
    """
    Generates context-aware reply suggestions
    """
    
    async def generate_suggestions(self, conversation: Conversation) -> List[Suggestion]:
        suggestions = []
        
        # 1. Template-based suggestions
        templates = await self.get_relevant_templates(conversation)
        suggestions.extend(templates)
        
        # 2. AI-generated suggestions
        ai_suggestions = await self.generate_ai_replies(conversation)
        suggestions.extend(ai_suggestions)
        
        # 3. Previous successful replies
        similar_conversations = await self.find_similar_conversations(conversation)
        historical_suggestions = await self.extract_successful_replies(similar_conversations)
        suggestions.extend(historical_suggestions)
        
        # 4. Rank and filter
        ranked = await self.rank_suggestions(
            suggestions,
            conversation.context,
            conversation.customer_profile
        )
        
        return ranked[:5]  # Top 5 suggestions
    
    async def generate_ai_replies(self, conversation: Conversation) -> List[Suggestion]:
        """
        Generate AI-powered reply suggestions using GPT
        """
        prompt = f"""
        Conversation Context:
        Customer: {conversation.last_customer_message}
        
        Previous messages:
        {self.format_history(conversation.history)}
        
        Customer Profile:
        {conversation.customer_profile}
        
        Generate 3 appropriate responses for a customer service agent:
        1. A empathetic response addressing their concern
        2. A solution-oriented response
        3. A question to gather more information
        
        Tone: Professional but friendly
        Length: 1-2 sentences each
        """
        
        responses = await self.gpt_api.generate(prompt)
        
        return [
            Suggestion(
                text=response,
                type="ai_generated",
                confidence=0.85,
                metadata={"generation_method": "gpt4"}
            )
            for response in responses
        ]
```

```python
class SentimentPriorityManager:
    """
    Monitors conversation sentiment and adjusts priority
    """
    
    async def analyze_conversation(self, conversation_id: str) -> PriorityUpdate:
        conversation = await self.get_conversation(conversation_id)
        
        # Analyze recent messages
        recent_messages = conversation.messages[-10:]  # Last 10 messages
        
        sentiment_scores = []
        urgency_indicators = []
        
        for message in recent_messages:
            if message.direction == "incoming":
                # Sentiment analysis
                sentiment = await self.sentiment_analyzer.analyze(message.text)
                sentiment_scores.append(sentiment.score)
                
                # Urgency detection
                urgency = self.detect_urgency_indicators(message.text)
                urgency_indicators.extend(urgency)
        
        # Calculate metrics
        avg_sentiment = sum(sentiment_scores) / len(sentiment_scores) if sentiment_scores else 0
        urgency_score = len(set(urgency_indicators))
        
        # Determine priority
        priority = self.calculate_priority(
            avg_sentiment=avg_sentiment,
            urgency_score=urgency_score,
            wait_time=conversation.wait_time,
            customer_tier=conversation.customer.tier
        )
        
        # Check for escalation
        needs_escalation = (
            avg_sentiment < -0.6 or  # Very negative
            urgency_score >= 3 or     # Multiple urgency indicators
            conversation.wait_time > 300  # Waiting > 5 minutes
        )
        
        return PriorityUpdate(
            conversation_id=conversation_id,
            new_priority=priority,
            needs_escalation=needs_escalation,
            sentiment_trend="improving" if avg_sentiment > -0.3 else "deteriorating",
            suggested_actions=self.get_suggested_actions(priority, needs_escalation)
        )
```

```python
class WhatsAppLeadScorer:
    """
    Scores leads based on WhatsApp interactions
    """
    
    async def score_lead(self, contact_id: str) -> LeadScore:
        contact = await self.get_contact(contact_id)
        conversations = await self.get_conversations(contact_id)
        
        score = 0
        factors = []
        
        # 1. Engagement frequency (30 points max)
        messages_last_7_days = self.count_messages_last_n_days(conversations, 7)
        engagement_score = min(30, messages_last_7_days * 3)
        score += engagement_score
        factors.append(f"Engagement: +{engagement_score}")
        
        # 2. Message intent (25 points max)
        intents = self.extract_intents(conversations)
        intent_score = self.calculate_intent_score(intents)
        score += intent_score
        factors.append(f"Intent quality: +{intent_score}")
        
        # 3. Response time (20 points max)
        avg_response_time = self.calculate_avg_response_time(conversations)
        response_score = max(0, 20 - (avg_response_time / 60))  # 20 points for instant response
        score += response_score
        factors.append(f"Responsiveness: +{response_score}")
        
        # 4. Conversation completion (15 points)
        completed_conversations = self.count_completed_conversations(conversations)
        completion_score = min(15, completed_conversations * 5)
        score += completion_score
        factors.append(f"Completion rate: +{completion_score}")
        
        # 5. Positive sentiment (10 points)
        sentiment_score = self.calculate_sentiment_score(conversations)
        score += sentiment_score
        factors.append(f"Sentiment: +{sentiment_score}")
        
        # Determine lead status
        status = self.determine_lead_status(score)
        
        return LeadScore(
            contact_id=contact_id,
            score=score,
            status=status,
            factors=factors,
            last_updated=datetime.utcnow(),
            recommended_action=self.get_recommended_action(status, score)
        )
```

```python
class OptInManager:
    """
    Manages WhatsApp opt-ins and compliance
    """
    
    async def process_opt_in(self, phone_number: str, source: str) -> OptInResult:
        """
        Process a new opt-in request
        """
        
        # 1. Check if already opted in
        existing = await self.get_opt_in_status(phone_number)
        if existing and existing.status == "active":
            return OptInResult(
                success=False,
                error="ALREADY_OPTED_IN",
                message="This number is already opted in"
            )
        
        # 2. Validate phone number
        if not self.validate_phone_number(phone_number):
            return OptInResult(
                success=False,
                error="INVALID_PHONE",
                message="Invalid phone number format"
            )
        
        # 3. Check compliance rules
        compliance_check = await self.check_compliance(phone_number, source)
        if not compliance_check.passed:
            return OptInResult(
                success=False,
                error="COMPLIANCE_FAILED",
                message=compliance_check.reason
            )
        
        # 4. Record opt-in
        opt_in = await self.create_opt_in_record(
            phone_number=phone_number,
            source=source,
            method="whatsapp",
            consent_text=compliance_check.consent_text,
            ip_address=self.get_request_ip()
        )
        
        # 5. Send confirmation message
        await self.send_confirmation_message(phone_number, opt_in.id)
        
        # 6. Update CRM
        await self.update_crm_opt_in_status(phone_number, "opted_in")
        
        return OptInResult(
            success=True,
            opt_in_id=opt_in.id,
            message="Opt-in successful. Confirmation sent."
        )
    
    async def process_opt_out(self, phone_number: str, keyword: str = "STOP") -> OptOutResult:
        """
        Process opt-out request
        """
        # 1. Update opt-in status
        await self.update_opt_in_status(phone_number, "opted_out", keyword)
        
        # 2. Add to suppression list
        await self.add_to_suppression_list(phone_number)
        
        # 3. Update all active campaigns
        await self.update_campaigns_for_opt_out(phone_number)
        
        # 4. Update CRM
        await self.update_crm_opt_in_status(phone_number, "opted_out")
        
        # 5. Send confirmation
        await self.send_opt_out_confirmation(phone_number)
        
        return OptOutResult(
            success=True,
            message="You have been unsubscribed from all messages."
        )
```

```python
class WhatsAppBusinessAPI:
    """
    Wrapper for WhatsApp Business Cloud API
    """
    
    async def send_message(self, to: str, message: dict) -> SendResult:
        """
        Send message via WhatsApp Business API
        """
        url = f"{self.base_url}/{self.phone_number_id}/messages"
        
        headers = {
            "Authorization": f"Bearer {self.access_token}",
            "Content-Type": "application/json"
        }
        
        try:
            response = await self.session.post(
                url,
                json=message,
                headers=headers
            )
            
            if response.status == 200:
                data = await response.json()
                return SendResult(
                    success=True,
                    message_id=data.get("messages", [{}])[0].get("id"),
                    whatsapp_response=data
                )
            else:
                error_data = await response.json()
                return SendResult(
                    success=False,
                    error_code=error_data.get("error", {}).get("code"),
                    error_message=error_data.get("error", {}).get("message")
                )
                
        except Exception as e:
            return SendResult(
                success=False,
                error_code="NETWORK_ERROR",
                error_message=str(e)
            )
    
    async def handle_webhook(self, payload: dict) -> WebhookResult:
        """
        Process incoming webhook from WhatsApp
        """
        entry = payload.get("entry", [{}])[0]
        changes = entry.get("changes", [{}])[0]
        value = changes.get("value", {})
        
        # Handle different webhook types
        if "messages" in value:
            return await self.process_message_webhook(value)
        elif "statuses" in value:
            return await self.process_status_webhook(value)
        elif "contacts" in value:
            return await self.process_contact_webhook(value)
        else:
            return WebhookResult(
                processed=False,
                type="unknown",
                data=payload
            )
```

```python
class WhatsAppComplianceManager:
    """
    Ensures compliance with WhatsApp Business Policy
    """
    
    async def validate_message(self, message: OutgoingMessage) -> ComplianceResult:
        """
        Validate message against WhatsApp policies
        """
        violations = []
        
        # 1. Check message type
        if message.type == "promotional":
            # Promotional messages require template
            if not message.template_name:
                violations.append("Promotional messages must use approved templates")
            
            # Check opt-in status
            opt_in = await self.check_opt_in(message.to)
            if not opt_in:
                violations.append("Recipient has not opted in for promotional messages")
        
        # 2. Check content restrictions
        if self.contains_restricted_content(message.content):
            violations.append("Message contains restricted content")
        
        # 3. Check sending hours (if configured)
        if not self.within_allowed_hours():
            violations.append("Outside of allowed sending hours")
        
        # 4. Check rate limits
        rate_limit = await self.check_rate_limit(message.to)
        if rate_limit.exceeded:
            violations.append(f"Rate limit exceeded: {rate_limit.message}")
        
        return ComplianceResult(
            valid=len(violations) == 0,
            violations=violations,
            requires_template=message.type == "promotional"
        )
    
    async def process_opt_out(self, phone_number: str):
        """
        Process opt-out and update suppression lists
        """
        # Add to global suppression list
        await self.suppression_list.add(phone_number)
        
        # Update all active campaigns
        await self.update_campaign_suppressions(phone_number)
        
        # Log for compliance
        await self.audit_log.log_opt_out(phone_number)
        
        # Notify agents if they were in conversation
        await self.notify_agents_of_opt_out(phone_number)
```

### **9.1 Key Performance Indicators**
```sql
-- Real-time dashboard queries
WITH agent_performance AS (
    SELECT 
        a.name as agent_name,
        COUNT(DISTINCT c.id) as total_conversations,
        COUNT(DISTINCT CASE WHEN c.status = 'resolved' THEN c.id END) as resolved_conversations,
        AVG(
            EXTRACT(EPOCH FROM (m2.created_at - m1.created_at))
        ) as avg_response_time_seconds,
        COUNT(DISTINCT CASE 
            WHEN m.direction = 'outgoing' AND m.sent_by_ai = true 
            THEN m.id 
        END) as ai_assisted_messages
    FROM agents a
    LEFT JOIN conversations c ON c.assigned_agent_id = a.id
    LEFT JOIN messages m ON m.conversation_id = c.id
    LEFT JOIN messages m1 ON m1.conversation_id = c.id 
        AND m1.direction = 'incoming'
    LEFT JOIN messages m2 ON m2.conversation_id = c.id 
        AND m2.direction = 'outgoing'
        AND m2.created_at > m1.created_at
    WHERE a.status != 'offline'
    GROUP BY a.id, a.name
),
campaign_performance AS (
    SELECT 
        bc.name as campaign_name,
        bc.type,
        bc.sent_at,
        (bc.stats->>'sent')::int as sent,
        (bc.stats->>'delivered')::int as delivered,
        (bc.stats->>'read')::int as read,
        (bc.stats->>'replied')::int as replied,
        ROUND(
            ((bc.stats->>'replied')::numeric / NULLIF((bc.stats->>'delivered')::numeric, 0)) * 100, 2
        ) as reply_rate
    FROM broadcast_campaigns bc
    WHERE bc.status = 'completed'
    ORDER BY bc.sent_at DESC
    LIMIT 10
)
SELECT * FROM agent_performance, campaign_performance;
```
